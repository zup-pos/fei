local function isPointInGui(g, x, y)
    if not g or not g.Parent then return false end
    local tl, sz = g.AbsolutePosition, g.AbsoluteSize
    if sz.X <= 0 or sz.Y <= 0 then return false end
    return x >= tl.X and x <= tl.X + sz.X and y >= tl.Y and y <= tl.Y + sz.Y
end

local function bindSafeClick(btn, callback)
    local pressInput, pressPos
    btn.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
        if input.UserInputState ~= Enum.UserInputState.Begin then return end
        pressInput = input
        pressPos = Vector2.new(input.Position.X, input.Position.Y)
    end)
    btn.InputEnded:Connect(function(input)
        if input ~= pressInput then return end
        local pos = Vector2.new(input.Position.X, input.Position.Y)
        local d = pressPos and (pos - pressPos).Magnitude or 999
        local wasInside = isPointInGui(btn, pos.X, pos.Y)
        pressInput = nil
        pressPos = nil
        if wasInside and d < 30 then
            local ok, err = pcall(callback)
            if not ok then warn("[bindSafeClick] " .. tostring(err)) end
        end
    end)
end

local function isKeywordDrawEnabled(kw)
    if not kw then return true end
    local v = state.keywordDrawEnabled[kw]
    if v == nil then return true end
    return v
end

local function isProtected(target)
    if not target then return false end
    if state.protectedTargets[target] then return true end
    local cur = target
    local depth = 0
    while cur and cur ~= workspace and depth < 30 do
        if state.protectedTargets[cur] then return true end
        cur = cur.Parent
        depth = depth + 1
    end
    return false
end

local function isProtectedTree(target)
    if not target then return false end
    if isProtected(target) then return true end
    if target:IsA("Model") then
        local ok, descendants = pcall(function() return target:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                if state.protectedTargets[d] then return true end
            end
        end
    end
    return false
end

local function extractKeyword(name)
    if not name or name == "" then return "" end
    local s = name
    s = s:gsub("%s*%(%d+%)%s*$", "")
    s = s:gsub("[_%s%-]+%d+%s*$", "")
    s = s:gsub("%d+$", "")
    s = s:gsub("[_%s%-]+$", "")
    s = s:match("^%s*(.-)%s*$") or s
    if s == "" then s = name end
    return string.lower(s)
end

    door = true, doors = true,
    wall = true, walls = true,
    floor = true, floors = true,
    roof = true, roofs = true,
}

local function isGenericKeyword(kw)
    if not kw or kw == "" then return true end
    if #kw < 3 then return true end
    return GENERIC_WORDS[kw] == true
end

local function tryExtractKeyword(name)
    local k = extractKeyword(name)
    if k == "" then return nil end
    if isGenericKeyword(k) then return nil end
    return k
end

-- ★ findBestKeyword 保留后代遍历：只在你「主动添加」时才调用
--   帮你从不具名的模型（比如整个模型叫 Model）中挖掘出有意义的关键词
local function findBestKeyword(root)
    if not root then return nil end
    local kw = tryExtractKeyword(root.Name)
    if kw then return kw end
    local ok, descendants = pcall(function() return root:GetDescendants() end)
    if ok and descendants then
        for _, d in ipairs(descendants) do
            if d:IsA("BasePart") then
                local k2 = tryExtractKeyword(d.Name)
                if k2 then return k2 end
            end
        end
    end
    local fallback = extractKeyword(root.Name)
    if fallback == "" or isGenericKeyword(fallback) then return nil end
    return fallback
end

local function sanitizeNote(val)
    if val == nil then return "" end
    val = tostring(val)
    val = val:gsub("[\r\n]+", " ")
    if #val > 80 then val = string.sub(val, 1, 80) end
    return val
end

local dragging, dragInput, dragLastPos, dragStartScreen
local dragStartFrameX, dragStartFrameY = 0, 0
local TOUCH_MATCH_THRESHOLD = 150

local function isStartInput(i)
    return i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch
end

titleBar.InputBegan:Connect(function(input)
    if not isStartInput(input) then return end
    if input.UserInputState ~= Enum.UserInputState.Begin then return end
    dragging = true; dragInput = input; dragLastPos = input.Position
    dragStartScreen = Vector2.new(input.Position.X, input.Position.Y)
    dragStartFrameX = frameX; dragStartFrameY = frameY
end)

addConn(UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    local matched = false
    if input == dragInput then matched = true
    elseif dragInput and dragInput.UserInputType == Enum.UserInputType.MouseButton1
        and input.UserInputType == Enum.UserInputType.MouseMovement then matched = true
    elseif dragInput and dragInput.UserInputType == Enum.UserInputType.Touch
        and input.UserInputType == Enum.UserInputType.Touch then
        if dragLastPos then
            local d = (input.Position - dragLastPos).Magnitude
            if d < TOUCH_MATCH_THRESHOLD then matched = true end
        end
    end
    if not matched then return end
    dragLastPos = input.Position
    local cur = Vector2.new(input.Position.X, input.Position.Y)
    local d = cur - dragStartScreen
    local newX, newY = clampFramePos(dragStartFrameX + d.X, dragStartFrameY + d.Y)
    setFramePos(newX, newY)
end))

addConn(UserInputService.InputEnded:Connect(function(input)
    if not dragging then return end
    if input == dragInput then
        dragging = false; dragInput = nil; dragLastPos = nil; dragStartScreen = nil
    elseif dragInput and dragInput.UserInputType == Enum.UserInputType.Touch
        and input.UserInputType == Enum.UserInputType.Touch then
        if dragLastPos and (input.Position - dragLastPos).Magnitude < TOUCH_MATCH_THRESHOLD then
            dragging = false; dragInput = nil; dragLastPos = nil; dragStartScreen = nil
        end
    end
end))

-- ==================== 黑洞吸附核心 ====================
local anchoredSaved = {}

local function unfreezeAll()
    for part, orig in pairs(anchoredSaved) do
        if part and part.Parent then
            pcall(function() part.Anchored = orig end)
        end
    end
    anchoredSaved = {}
end

local function getBlackholeTargetCF()
    local vp = camera.ViewportSize
    local ray = camera:ViewportPointToRay(vp.X / 2, vp.Y / 2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filterList = {}
    if player.Character then table.insert(filterList, player.Character) end
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            table.insert(filterList, target)
        end
    end
    params.FilterDescendantsInstances = filterList
    params.IgnoreWater = true
    local hit = workspace:Raycast(ray.Origin, ray.Direction * 100, params)
    if hit then
        return CFrame.new(hit.Position - ray.Direction.Unit * 2)
    else
        return camera.CFrame * CFrame.new(0, 0, -15)
    end
end

local function blackholeSnapOnce()
    if next(drawGroups) == nil then return 0 end
    local targetCF = getBlackholeTargetCF()
    local touched = {}
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            if target:IsA("Model") then
                local dOk, descendants = pcall(function() return target:GetDescendants() end)
                if dOk and descendants then
                    for _, d in ipairs(descendants) do
                        if d:IsA("BasePart") then
                            if anchoredSaved[d] == nil then anchoredSaved[d] = d.Anchored end
                            if d.Anchored then pcall(function() d.Anchored = false end) end
                            table.insert(touched, d)
                        end
                    end
                end
            elseif target:IsA("BasePart") then
                if anchoredSaved[target] == nil then anchoredSaved[target] = target.Anchored end
                if target.Anchored then pcall(function() target.Anchored = false end) end
                table.insert(touched, target)
            end
        end
    end
    if #touched == 0 then return 0 end
    task.wait(0.05)
    local movedCount = 0
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            if target:IsA("Model") then
                pcall(function() target:PivotTo(targetCF) end)
                movedCount = movedCount + 1
            elseif target:IsA("BasePart") then
                pcall(function() target.CFrame = targetCF end)
                movedCount = movedCount + 1
            end
        end
    end
    task.wait(0.05)
    for _, d in ipairs(touched) do
        if d and d.Parent then
            local orig = anchoredSaved[d]
            if orig ~= nil then pcall(function() d.Anchored = orig end) end
        end
    end
    for _, d in ipairs(touched) do anchoredSaved[d] = nil end
    return movedCount
end

local function blackholeSnapContinuous()
    if next(drawGroups) == nil then return 0 end
    local targetCF = getBlackholeTargetCF()
    local movedCount = 0
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            if target:IsA("Model") then
                pcall(function() target:PivotTo(targetCF) end)
                movedCount = movedCount + 1
            elseif target:IsA("BasePart") then
                pcall(function() target.CFrame = targetCF end)
                movedCount = movedCount + 1
            end
        end
    end
    return movedCount
end

-- ==================== 射线检测 ====================
local function isVisibleForRaycast(part)
    if not part or not part:IsA("BasePart") then return false end
    if part:IsA("Terrain") then return false end
    if part.Transparency >= 0.95 then return false end
    if part.LocalTransparencyModifier >= 0.95 then return false end
    local n = string.lower(part.Name)
    if string.find(n, "hitbox", 1, true) then return false end
    if string.find(n, "trigger", 1, true) then return false end
    if string.find(n, "zone", 1, true) then return false end
    if string.find(n, "invisible", 1, true) then return false end
    if string.find(n, "collision", 1, true) then return false end
    return true
end

local function resolveHitPart(part, hitPos)
    if not part or not part:IsA("BasePart") then return nil end
    if part:IsA("Terrain") then return nil end
    if isVisibleForRaycast(part) then return part end
    local cur = part
    local depth = 0
    while cur and cur ~= workspace and depth < 5 do
        if cur:IsA("Model") then
            local bestPart = nil
            local bestDist = math.huge
            local count = 0
            local ok, descendants = pcall(function() return cur:GetDescendants() end)
            if ok and descendants then
                for _, d in ipairs(descendants) do
                    count = count + 1
                    if count > 200 then break end
                    if d:IsA("BasePart") and not d:IsA("Terrain") and isVisibleForRaycast(d) then
                        local dist = (d.Position - hitPos).Magnitude
                        if dist < bestDist then
                            bestDist = dist
                            bestPart = d
                        end
                    end
                end
            end
            if bestPart then return bestPart end
        end
        cur = cur.Parent
        depth = depth + 1
    end
    return nil
end

local function getTargetInCenter(includeProtected)
    local vp = camera.ViewportSize
    local ray = camera:ViewportPointToRay(vp.X / 2, vp.Y / 2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filterList = {}
    if player.Character then table.insert(filterList, player.Character) end
    if not includeProtected then
        for t in pairs(state.protectedTargets) do
            if t then table.insert(filterList, t) end
        end
    end
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and not isKeywordDrawEnabled(entry.keyword) then
            table.insert(filterList, target)
        end
    end
    params.FilterDescendantsInstances = filterList
    params.IgnoreWater = true
    local origin = ray.Origin
    local direction = ray.Direction.Unit
    local remainingDist = 1000
    local attempts = 0
    while attempts < 15 and remainingDist > 0.5 do
        attempts = attempts + 1
        local r = workspace:Raycast(origin, direction * remainingDist, params)
        if not r then break end
        local part = r.Instance
        if part and part:IsA("BasePart") then
            local resolved = resolveHitPart(part, r.Position)
            if resolved then return resolved end
        end
        local dist = (r.Position - origin).Magnitude
        if dist <= 0 then break end
        origin = r.Position + direction * 0.1
        remainingDist = remainingDist - dist - 0.1
    end
    return nil
end

local function findModelRoot(part)
    if not part then return nil end
    if part:IsA("Model") then return part end
    local cur = part
    while cur and cur ~= workspace do
        if cur:IsA("Model") then return cur end
        cur = cur.Parent
    end
    return part
end

-- ==================== 坐标工具 ====================
local function isVisiblePart(part)
    if not part:IsA("BasePart") then return false end
    if part.Transparency >= 0.95 then return false end
    if part.LocalTransparencyModifier >= 0.95 then return false end
    local n = string.lower(part.Name)
    if string.find(n, "hitbox", 1, true) then return false end
    if string.find(n, "trigger", 1, true) then return false end
    if string.find(n, "zone", 1, true) then return false end
    if string.find(n, "invisible", 1, true) then return false end
    return true
end

local function getModelCenter(target)
    if target:IsA("BasePart") then return target.Position end
    if target:IsA("Model") then
        local hrp = target:FindFirstChild("HumanoidRootPart")
        if hrp and hrp:IsA("BasePart") then return hrp.Position end
        local sum = Vector3.new(0, 0, 0)
        local n = 0
        local ok, descendants = pcall(function() return target:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                if d:IsA("BasePart") and isVisiblePart(d) then
                    sum = sum + d.Position; n = n + 1
                end
            end
        end
        if n > 0 then return sum / n end
        local ok2, cf = pcall(function() return target:GetBoundingBox() end)
        if ok2 and cf then return cf.Position end
    end
    return nil
end

local function getBBox2D(target)
    local parts = {}
    if target:IsA("BasePart") then
        parts = { target }
    elseif target:IsA("Model") then
        local ok, descendants = pcall(function() return target:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                if d:IsA("BasePart") and isVisiblePart(d) then
                    table.insert(parts, d)
                end
            end
        end
    end
    if #parts == 0 then return nil end
    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local any = false
    for _, part in ipairs(parts) do
        local cf, sz = part.CFrame, part.Size
        local hx, hy, hz = sz.X / 2, sz.Y / 2, sz.Z / 2
        local corners = {
            cf * Vector3.new(-hx, -hy, -hz), cf * Vector3.new(hx, -hy, -hz),
            cf * Vector3.new(-hx, hy, -hz), cf * Vector3.new(hx, hy, -hz),
            cf * Vector3.new(-hx, -hy, hz), cf * Vector3.new(hx, -hy, hz),
            cf * Vector3.new(-hx, hy, hz), cf * Vector3.new(hx, hy, hz),
        }
        for _, v in ipairs(corners) do
            local sp, on = camera:WorldToScreenPoint(v)
            if on and sp.Z > 0 then
                any = true
                local safeX = math.clamp(sp.X, -10000, 10000)
                local safeY = math.clamp(sp.Y, -10000, 10000)
                if safeX < minX then minX = safeX end
                if safeX > maxX then maxX = safeX end
                if safeY < minY then minY = safeY end
                if safeY > maxY then maxY = safeY end
            end
        end
    end
    if not any then return nil end
    return minX, minY, maxX, maxY
end

-- ==================== 关键词匹配工具 ====================
local function buildKws()
    local kws, hasAny = {}, false
    for kw, _ in pairs(state.keywords) do
        kws[kw] = true
        hasAny = true
    end
    return hasAny and kws or nil
end

-- ★ 收紧匹配：只保留正向（物体名包含关键词），去掉危险的反向匹配
local function matchKeywordByName(name, kws)
    if not name then return nil end
    local extracted = extractKeyword(name)
    if extracted == "" then return nil end
    if kws[extracted] then return extracted end
    if isGenericKeyword(extracted) then return nil end
    if #extracted < 4 then return nil end
    -- ★ 只做正向匹配：物体名包含关键词
    for kw, _ in pairs(kws) do
        if not isGenericKeyword(kw) and #kw >= 4 then
            if string.find(extracted, kw, 1, true) then return kw end
        end
    end
    return nil
end

-- ★ 收紧 matchRoot：只检查 root 名字和 inst 的祖先链，不再遍历后代
--   findBestKeyword（用户主动添加）依然保留后代遍历，不影响用户主动添加能力
local function matchRoot(root, inst, kws)
    local cur = inst
    local depth = 0
    while cur and cur ~= workspace and depth < 6 do
        local m = matchKeywordByName(cur.Name, kws)
        if m then return m end
        cur = cur.Parent
        depth = depth + 1
    end
    return nil
end

-- ==================== 近距离同步扫描 ====================
local function quickNearbyScan()
    if state.scanPaused then return 0 end
    local kws = buildKws()
    if not kws then return 0 end

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local origin = hrp and hrp.Position or camera.CFrame.Position

    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if char then params.FilterDescendantsInstances = { char } end
    params.MaxParts = NEARBY_MAX_PARTS

    local ok, parts = pcall(function()
        return workspace:GetPartBoundsInRadius(origin, NEARBY_RADIUS, params)
    end)
    if not ok or not parts then return 0 end

    local added = 0
    local seenRoot = {}
    for _, inst in ipairs(parts) do
        if inst:IsA("BasePart") and not inst:IsA("Terrain") then
            local root = findModelRoot(inst)
            if root and not seenRoot[root] and not drawGroups[root] and not isProtectedTree(root) then
                seenRoot[root] = true
                local matchedKw = matchRoot(root, inst, kws)
                if matchedKw then
                    drawGroups[root] = {
                        g = createGroup(),
                        label = createNameLabel(),
                        name = root.Name,
                        keyword = matchedKw,
                    }
                    added = added + 1
                end
            end
        end
    end
    return added
end

-- ==================== 单目标尝试添加（事件驱动用） ====================
local function tryAddRoot(root)
    if state.scanPaused then return false end
    if not root then return false end
    local ok, parent = pcall(function() return root.Parent end)
    if not ok or not parent then return false end
    if drawGroups[root] then return false end
    if isProtectedTree(root) then return false end

    local kws = buildKws()
    if not kws then return false end

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local origin = hrp and hrp.Position or camera.CFrame.Position
    local center
    if root:IsA("Model") then
        local ok2, box = pcall(function() return root:GetBoundingBox() end)
        if ok2 and box then center = box.Position end
    elseif root:IsA("BasePart") then
        center = root.Position
    end
    if center and (center - origin).Magnitude > MAX_DRAW_DISTANCE then return false end

    local inst = root:IsA("BasePart") and root or root:FindFirstChildWhichIsA("BasePart")
    if not inst then
        local ok3, descendants = pcall(function() return root:GetDescendants() end)
        if ok3 and descendants then
            for _, d in ipairs(descendants) do
                if d:IsA("BasePart") then inst = d; break end
            end
        end
    end
    if not inst then return false end

    for _, pl in ipairs(Players:GetPlayers()) do
        local ok4, isDesc = pcall(function()
            return pl.Character and (pl.Character == root
                or root:IsDescendantOf(pl.Character)
                or pl.Character:IsDescendantOf(root))
        end)
        if ok4 and isDesc then return false end
    end

    local matchedKw = matchRoot(root, inst, kws)
    if not matchedKw then return false end

    drawGroups[root] = {
        g = createGroup(),
        label = createNameLabel(),
        name = root.Name,
        keyword = matchedKw,
    }
    return true
end

-- ==================== 分帧兜底扫描 ====================
local function startScanByKeywords()
    if state.scanPaused then return end
    local kws = buildKws()
    if not kws then return end

    if scanQueue then scanQueue = nil end

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local origin = hrp and hrp.Position or camera.CFrame.Position
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if char then params.FilterDescendantsInstances = { char } end
    params.MaxParts = 20000

    local ok, parts = pcall(function()
        return workspace:GetPartBoundsInRadius(origin, state.scanRadius, params)
    end)
    if not ok or not parts then return end

    scanQueue = {
        parts = parts,
        index = 1,
        added = 0,
        seenRoot = {},
        kws = kws,
    }
end

local function processScanQueue()
    if state.scanPaused then return end
    if not scanQueue then return end
    local parts = scanQueue.parts
    local kws = scanQueue.kws
    if not kws or not parts then scanQueue = nil; return end

    local stopIdx = math.min(scanQueue.index + SCAN_PARTS_PER_FRAME - 1, #parts)

    local playerChars = {}
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl.Character then playerChars[pl.Character] = true end
    end

    for i = scanQueue.index, stopIdx do
        local inst = parts[i]
        if inst and inst.Parent and inst:IsA("BasePart") and not inst:IsA("Terrain") then
            local root = findModelRoot(inst)
            if root and not scanQueue.seenRoot[root] and not drawGroups[root] and not isProtectedTree(root) then
                if not playerChars[root] then
                    local matchedKw = matchRoot(root, inst, kws)
                    if matchedKw then
                        scanQueue.seenRoot[root] = true
                        drawGroups[root] = {
                            g = createGroup(),
                            label = createNameLabel(),
                            name = root.Name,
                            keyword = matchedKw,
                        }
                        scanQueue.added = scanQueue.added + 1
                    end
                end
            end
        end
    end

    scanQueue.index = stopIdx + 1
    if scanQueue.index > #parts then
        local added = scanQueue.added or 0
        scanQueue = nil
        if added >= 2 then
            addLog("扫描补充 " .. added .. " 个目标")
        end
    end
end

local function onNewInstance(inst)
    if not inst:IsA("BasePart") then return end
    if inst:IsA("Terrain") then return end
    local root = findModelRoot(inst)
    if root then pendingRoots[root] = true end
end

addConn(workspace.DescendantAdded:Connect(onNewInstance))
addConn(workspace.ChildAdded:Connect(onNewInstance))

task.spawn(function()
    while true do
        task.wait(0.1)
        if not state.scanPaused and next(pendingRoots) then
            local batch = pendingRoots
            pendingRoots = {}
            local count = 0
            for root in pairs(batch) do
                if tryAddRoot(root) then count = count + 1 end
            end
            if count > 0 then addLog("事件捕获 " .. count .. " 个目标") end
        end
    end
end)

-- ==================== 添加目标 ====================
local function addTarget(target, autoScan)
    if not target then return false end
    local root = findModelRoot(target)
    if not root then return false end
    if isProtectedTree(root) then addLog("该目标已被保护"); return false end
    if drawGroups[root] then addLog("已存在: " .. root.Name); return false end

    local keyword = findBestKeyword(root)
    if not keyword then
        addLog("该目标名为通用词无法自动提取关键词: " .. root.Name)
        return false
    end

    state.keywords[keyword] = true
    drawGroups[root] = {
        g = createGroup(),
        label = createNameLabel(),
        name = root.Name,
        keyword = keyword,
    }
    addLog("添加: " .. root.Name .. " [关键词: " .. keyword .. "]", keyword, root)

    if autoScan == true then
        local nearbyAdded = quickNearbyScan()
        if nearbyAdded > 0 then
            addLog("近距离扫描补充 " .. nearbyAdded .. " 个目标")
        end
        startScanByKeywords()
    end
    return true
end

local function protectTarget(target)
    if not target then return false end
    if state.protectedTargets[target] then
        addLog("已在保护列表: " .. target.Name)
        return false
    end
    state.protectedTargets[target] = { note = "" }

    local check = target
    while check and check ~= workspace do
        if drawGroups[check] then
            destroyEntry(drawGroups[check])
            drawGroups[check] = nil
            break
        end
        check = check.Parent
    end

    updateProtectBtn()
    addLog("已加入单目标保护: " .. target.Name)
    return true
end

local function unprotectTarget(target)
    if not target then return false end
    if not state.protectedTargets[target] then
        addLog("不在保护列表: " .. target.Name)
        return false
    end
    state.protectedTargets[target] = nil
    updateProtectBtn()
    addLog("已取消保护: " .. target.Name)
    addTarget(target, false)
    return true
end

-- ==================== 主渲染循环 ====================
local lastBHFrame = 0
addConn(RunService.RenderStepped:Connect(function()
    local insetOk, inset = pcall(function() return GuiService:GetGuiInset() end)
    local insetY = (insetOk and inset) and inset.Y or 0

    cleanInvalidProtected()
    processScanQueue()

    if state.blackholeOn and next(drawGroups) then
        local now = tick()
        if now - lastBHFrame >= 1/30 then
            lastBHFrame = now
            blackholeSnapContinuous()
        end
    end

    if state.crosshairOn and crossH and crossV then
        local vp = camera.ViewportSize
        local cx, cy = vp.X / 2, vp.Y / 2
        crossH.Visible = true; crossV.Visible = true
        crossH.From = Vector2.new(cx - 10, cy); crossH.To = Vector2.new(cx + 10, cy)
        crossV.From = Vector2.new(cx, cy - 10); crossV.To = Vector2.new(cx, cy + 10)
    elseif crossH then
        crossH.Visible = false; crossV.Visible = false
    end

    if state.drawOn then
        ensureDrawObjects()
        local vp = camera.ViewportSize
        local cx = vp.X / 2
        local countTopY = 0 + insetY
        local camPos = camera.CFrame.Position

        local visible = {}
        local totalValid = 0
        local posSeen = {}
        local toRemove = {}

        for target, entry in pairs(drawGroups) do
            local ok, parent = pcall(function() return target.Parent end)
            local isValid = ok and parent and target:IsDescendantOf(workspace)

            if not isValid then
                table.insert(toRemove, target)
            else
                local skipByAncestor = false
                local anc = target.Parent
                local ancDepth = 0
                while anc and anc ~= workspace and ancDepth < 10 do
                    if drawGroups[anc] then skipByAncestor = true; break end
                    anc = anc.Parent
                    ancDepth = ancDepth + 1
                end

                local skip = skipByAncestor
                    or (not isKeywordDrawEnabled(entry.keyword))
                    or isProtectedTree(target)

                if skip then
                    for _, obj in pairs(entry.g) do obj.Visible = false end
                    if entry.label then entry.label.Visible = false end
                else
                    local center = getModelCenter(target)
                    if center then
                        local dist = (center - camPos).Magnitude

                        if dist > MAX_DRAW_DISTANCE then
                            table.insert(toRemove, target)
                        else
                            local posKey = string.format("%d_%d_%d",
                                math.floor(center.X * 10 + 0.5),
                                math.floor(center.Y * 10 + 0.5),
                                math.floor(center.Z * 10 + 0.5))
                            if posSeen[posKey] then
                                for _, obj in pairs(entry.g) do obj.Visible = false end
                                if entry.label then entry.label.Visible = false end
                            else
                                posSeen[posKey] = target
                                totalValid = totalValid + 1
                                local minX, minY, maxX, maxY = getBBox2D(target)

                                if minX then
                                    local isOutsideScreen = (maxX < -EDGE_MARGIN) or (minX > vp.X + EDGE_MARGIN) or (maxY < -EDGE_MARGIN) or (minY > vp.Y + EDGE_MARGIN)
                                    local centerSp, centerOn = camera:WorldToScreenPoint(center)
                                    local isBehindCamera = (not centerOn) or (centerSp.Z <= 0)

                                    if not isOutsideScreen and not isBehindCamera then
                                        table.insert(visible, {
                                            target = target, entry = entry,
                                            x1 = minX, y1 = minY + insetY,
                                            x2 = maxX, y2 = maxY + insetY,
                                            dist = dist,
                                            center = center,
                                        })
                                    else
                                        for _, obj in pairs(entry.g) do obj.Visible = false end
                                        if entry.label then entry.label.Visible = false end
                                    end
                                else
                                    for _, obj in pairs(entry.g) do obj.Visible = false end
                                    if entry.label then entry.label.Visible = false end
                                end
                            end
                        end
                    else
                        for _, obj in pairs(entry.g) do obj.Visible = false end
                        if entry.label then entry.label.Visible = false end
                    end
                end
            end
        end

        for _, target in ipairs(toRemove) do
            local entry = drawGroups[target]
            if entry then
                destroyEntry(entry)
                drawGroups[target] = nil
            end
        end

        local MERGE_DIST = 40
        local groups2D = {}
        for _, item in ipairs(visible) do
            local cx1 = (item.x1 + item.x2) / 2
            local cy1 = (item.y1 + item.y2) / 2
            local merged = false
            for _, g in ipairs(groups2D) do
                local gcx = (g.x1 + g.x2) / 2
                local gcy = (g.y1 + g.y2) / 2
                if math.abs(cx1 - gcx) < MERGE_DIST and math.abs(cy1 - gcy) < MERGE_DIST
                    and g.keyword == item.entry.keyword then
                    g.x1 = math.min(g.x1, item.x1)
                    g.y1 = math.min(g.y1, item.y1)
                    g.x2 = math.max(g.x2, item.x2)
                    g.y2 = math.max(g.y2, item.y2)
                    g.count = g.count + 1
                    table.insert(g.entries, item)
                    merged = true
                    break
                end
            end
            if not merged then
                table.insert(groups2D, {
                    x1 = item.x1, y1 = item.y1,
                    x2 = item.x2, y2 = item.y2,
                    count = 1, entries = { item },
                    keyword = item.entry.keyword,
                })
            end
        end

        local visibleGroups = {}
        local visibleLabels = {}
        local placedLabels = {}
        local LABEL_W = 130
        local LABEL_H = 20

        for _, g in ipairs(groups2D) do
            local first = g.entries[1]
            local entry = first.entry
            local group = entry.g
            visibleGroups[group] = true
            if entry.label then visibleLabels[entry.label] = true end
            for i = 2, #g.entries do
                local other = g.entries[i].entry
                for _, obj in pairs(other.g) do obj.Visible = false end
                if other.label then other.label.Visible = false end
            end
            local x1, y1, x2, y2 = g.x1, g.y1, g.x2, g.y2
            group.boxT.From = Vector2.new(x1, y1); group.boxT.To = Vector2.new(x2, y1)
            group.boxB.From = Vector2.new(x1, y2); group.boxB.To = Vector2.new(x2, y2)
            group.boxL.From = Vector2.new(x1, y1); group.boxL.To = Vector2.new(x1, y2)
            group.boxR.From = Vector2.new(x2, y1); group.boxR.To = Vector2.new(x2, y2)
            group.boxT.Visible = true; group.boxB.Visible = true
            group.boxL.Visible = true; group.boxR.Visible = true

            local centerPos = first.center
            if centerPos then
                local sp, on = camera:WorldToScreenPoint(centerPos)
                if on and sp.Z > 0 then
                    group.line.From = Vector2.new(cx, countTopY + 24)
                    group.line.To = Vector2.new(sp.X, sp.Y + insetY)
                    group.line.Visible = true
                else
                    group.line.Visible = false
                end
            else
                group.line.Visible = false
            end

            if entry.label then
                local keywordMap = {}
                local keywordOrder = {}
                for _, item in ipairs(g.entries) do
                    local e = item.entry
                    local kw = e.keyword or "?"
                    local noteStr = state.keywordNotes[kw]
                    local displayName
                    if noteStr and noteStr ~= "" then
                        displayName = noteStr
                    else
                        displayName = e.name or "?"
                    end
                    if not keywordMap[kw] then
                        keywordMap[kw] = { count = 0, name = displayName, minDist = math.huge }
                        table.insert(keywordOrder, kw)
                    end
                    keywordMap[kw].count = keywordMap[kw].count + 1
                    if item.dist and item.dist < keywordMap[kw].minDist then
                        keywordMap[kw].minDist = item.dist
                    end
                end
                table.sort(keywordOrder)
                local lines = {}
                for _, kw in ipairs(keywordOrder) do
                    local d = keywordMap[kw]
                    local line = d.name .. " ×" .. d.count
                    if state.showDistance and d.minDist ~= math.huge and d.minDist > 5 then
                        line = line .. " [" .. math.floor(d.minDist) .. "m]"
                    end
                    table.insert(lines, line)
                end
                entry.label.Text = table.concat(lines, "\n")
                local baseX = (x1 + x2) / 2
                local baseY = y1 - 4
                local actualY = baseY
                for _ = 1, 8 do
                    local collided = false
                    for _, p in ipairs(placedLabels) do
                        if math.abs(baseX - p.x) < LABEL_W
                            and math.abs(actualY - p.y) < LABEL_H then
                            actualY = p.y - LABEL_H
                            collided = true
                            break
                        end
                    end
                    if not collided then break end
                end
                table.insert(placedLabels, { x = baseX, y = actualY })
                entry.label.Position = UDim2.fromOffset(baseX, actualY)
                entry.label.Visible = true
            end
        end

        for _, entry in pairs(drawGroups) do
            if not visibleGroups[entry.g] then
                for _, obj in pairs(entry.g) do obj.Visible = false end
            end
            if entry.label and not visibleLabels[entry.label] then
                entry.label.Visible = false
            end
        end

        topCountText.Center = true
        topCountText.Position = Vector2.new(cx, countTopY)
        topCountText.Text = tostring(totalValid)
        topCountText.Visible = true
    end
end))

    if overlayGui then overlayGui:Destroy() end
    if logWindow then logWindow:Destroy() end
    if targetWindow then targetWindow:Destroy() end
    if protectWindow then protectWindow:Destroy() end
    if blackholeMenu then blackholeMenu:Destroy() end
end
