-- 关键词去尾数
local function extractKeyword(name)
    if not name or name == "" then return "" end
    local s = name
    s = s:gsub("%s*%(%d+%)%s*$", "")
    s = s:gsub("[_%s]+%d+%s*$", "")
    s = s:gsub("%d+$", "")
    s = s:gsub("[_%s]+$", "")
    s = s:match("^%s*(.-)%s*$") or s
    if s == "" then s = name end
    return string.lower(s)
end

-- 序列化 / 反序列化
local function serializeKeywords(keywords)
    local lines = {}
    for _, item in ipairs(keywords) do
        local prefix = item.prefix or ""
        local exact = item.exact or ""
        table.insert(lines, prefix .. "," .. exact)
    end
    return table.concat(lines, "\n")
end

local function deserializeKeywords(content)
    local keywords = {}
    for line in string.gmatch(content, "[^\n]+") do
        local firstComma = string.find(line, ",")
        if firstComma then
            local prefix = string.sub(line, 1, firstComma - 1)
            local exact = string.sub(line, firstComma + 1)
            if prefix ~= "" then
                table.insert(keywords, { prefix = prefix, exact = (exact ~= "" and exact) or nil })
            end
        else
            if line ~= "" then
                table.insert(keywords, { prefix = line, exact = nil })
            end
        end
    end
    return keywords
end

        end

                targetList[obj] = rec

            obj.ColorMap = ""; obj.MetalnessMap = ""; obj.RoughnessMap = ""; obj.NormalMap = ""
            return

            local rec = guiStateMap[obj]
            if not rec then rec = { enabled = obj.Enabled } end

                    end
                end
                if not tooBig then return cur end
            end
            cur = cur.Parent
        end
        return part
    end

    -- 【修改】collectTargetParts 里改用 isNameProtected 过滤
    local function collectTargetParts(root)
        local list = {}
        if not root then return list end
        local function consider(obj)
            -- 【关键】名字类保护直接跳过（覆盖 BasePart + 非 BasePart）
            if isNameProtected(obj) then return end
            if obj:IsA("BasePart") and not obj:IsA("Terrain") then
                if not isInvisibleWall(obj) then table.insert(list, obj) end
            elseif obj:IsA("Decal") or obj:IsA("Texture") or obj:IsA("SurfaceAppearance") then
                table.insert(list, obj)
            elseif obj:IsA("SurfaceGui") or obj:IsA("BillboardGui") then
                table.insert(list, obj)
            elseif obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail")
                or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                table.insert(list, obj)
            end
        end
        consider(root)
        for _, d in ipairs(root:GetDescendants()) do consider(d) end
        return list
    end

    local function makeDraggable(frame, handle)
        handle = handle or frame
        local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
        handle.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
            dragging = true; dragInput = input
            dragStart = Vector2.new(input.Position.X, input.Position.Y)
            startPos = frame.Position
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not dragging then return end
            local matched = (input == dragInput)
                or (dragInput and dragInput.UserInputType == Enum.UserInputType.MouseButton1
                    and input.UserInputType == Enum.UserInputType.MouseMovement)
            if not matched then return end
            if input.UserInputType ~= Enum.UserInputType.MouseMovement
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
            local cur = Vector2.new(input.Position.X, input.Position.Y)
            local d = cur - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input == dragInput then dragging = false; dragInput = nil end
        end)
    end

    SB.getTargetInCenter = function()

    SB.smartScan = function()
        if scanRunning then return end
        scanRunning = true
        if ui.control then showNotification("扫描中...") end
        task.spawn(function()
            local t0 = tick()
            local cam = workspace.CurrentCamera
            if not cam then scanRunning = false return end
            local range = CONFIG.scanRange
            local camPos = cam.CFrame.Position
            local look = cam.CFrame.LookVector
            local boxCenter = camPos + look * (range * 0.35)
            local boxSize = Vector3.new(range, range * 0.6, range)
            local overlapParams = OverlapParams.new()
            overlapParams.FilterType = Enum.RaycastFilterType.Exclude
            overlapParams.FilterDescendantsInstances = { player.Character }
            overlapParams.MaxParts = CONFIG.maxPartsPerScan
            local parts = workspace:GetPartBoundsInBox(CFrame.new(boxCenter), boxSize, overlapParams)
            local rootSet, roots = {}, {}
            local total = #parts
            for i, part in ipairs(parts) do
                if part:IsA("BasePart") and not part:IsA("Terrain")
                    and not targetList[part] and not isInvisibleWall(part) then
                    if not isIgnored(part) then
                        local root = getTargetRoot(part)
                        if root and not rootSet[root] and not isNameProtected(root) then
                            rootSet[root] = true
                            table.insert(roots, root)
                        end
                    end
                end
                if i % CONFIG.batchClassify == 0 then
                    task.wait()
                    if ui.control then
                        showNotification(string.format("分类 %d%%", math.floor(i / total * 100)))
                    end
                end
            end
            local applied, processedRoots = 0, 0
            for _, root in ipairs(roots) do
                local subParts = collectTargetParts(root)
                for _, p in ipairs(subParts) do
                    if targetList[p] == nil and guiStateMap[p] == nil then
                        local blocked = false
                        if p:IsA("BasePart") then
                            blocked = isIgnored(p)
                        else
                            blocked = isNameProtected(p)
                        end
                        if not blocked then
                            setObjTransparency(p, CONFIG.transparency)
                            partRootMap[p] = root
                            applied = applied + 1
                        end
                    end
                end
                processedRoots = processedRoots + 1
                if processedRoots % CONFIG.batchApply == 0 then
                    task.wait()
                    if ui.control then
                        showNotification(string.format("应用 %d/%d", processedRoots, #roots))
                    end
                end
            end
            SB.refreshListUI()
            local elapsed = tick() - t0
            if ui.control then
                showNotification(string.format("完成 %d (%.2fs)", applied, elapsed))
            end
            tanchuangxiaoxi(string.format("共修改 %d 个部件  (%.2fs)", applied, elapsed), "修改模式")
            scanRunning = false
        end)
    end

    SB.addTarget = function(target)
        if not target then return false end
        if not (target:IsA("BasePart") or target:IsA("Model")) then
            showNotification("无效目标"); return false
        end
        local root = getTargetRoot(target)
        if not root then return false end

        -- 整体保护检查：root 本身被保护则直接拒绝
        if isNameProtected(root) then
            showNotification("该目标已被保护")
            tanchuangxiaoxi("已被保护，无法添加: " .. root.Name, "保护")
            return false
        end

        local parts = collectTargetParts(root)
        if #parts == 0 then showNotification("无有效部件"); return false end

        local addedCount = 0
        for _, p in ipairs(parts) do
            if targetList[p] ~= nil or guiStateMap[p] ~= nil then
                addedCount = addedCount + 1
            end
        end
        if addedCount == #parts and addedCount > 0 then
            SB.removeTarget(root)
            showNotification("已取消: " .. root.Name)
            return false
        end

        local added = 0
        for _, p in ipairs(parts) do
            if player.Character and p:IsDescendantOf(player.Character) then continue end
            local blocked = false
            if p:IsA("BasePart") then
                blocked = isIgnored(p)
            else
                blocked = isNameProtected(p)
            end
            if blocked then continue end
            if targetList[p] == nil and guiStateMap[p] == nil then
                setObjTransparency(p, CONFIG.transparency)
                partRootMap[p] = root
                added = added + 1
            end
        end
        SB.refreshListUI()
        if added > 0 then
            showNotification("已添加 " .. root.Name .. " (" .. added .. ")")
        else
            showNotification("全部被拦，未添加")
        end
        return added > 0
    end

    SB.removeTarget = function(root)
        if not root then return end
        local parts = collectTargetParts(root)
        for _, p in ipairs(parts) do
            restoreObj(p)
            targetList[p] = nil; guiStateMap[p] = nil; partRootMap[p] = nil
        end
        SB.refreshListUI()
    end

    SB.toggleProtect = function(target)
        if not target then return end
        local root = getTargetRoot(target)
        if not root then return end

        if protectedList[root] then
            -- 取消保护
            protectedList[root] = nil
            tanchuangxiaoxi("已取消保护: " .. root.Name, "保护")
        else
            -- 开启保护
            protectedList[root] = true
            -- 恢复所有该 root 下已被修改的部件（含非 BasePart）
            local restored = 0
            local allObjs = { root }
            for _, d in ipairs(root:GetDescendants()) do
                table.insert(allObjs, d)
            end
            for _, p in ipairs(allObjs) do
                if targetList[p] ~= nil or guiStateMap[p] ~= nil then
                    restoreObj(p)
                    targetList[p] = nil; guiStateMap[p] = nil; partRootMap[p] = nil
                    restored = restored + 1
                end
            end
            SB.refreshListUI()
            local extra = (restored > 0) and (" (已恢复 " .. restored .. " 部件)") or ""
            tanchuangxiaoxi("已保护: " .. root.Name .. extra, "保护")
        end
    end

    SB.restoreAll = function()
        local n = 0
        for part, _ in pairs(targetList) do
            if part and part.Parent then
                restoreObj(part); n = n + 1
            end
        end
        for obj, _ in pairs(guiStateMap) do
            if obj and obj.Parent then
                restoreObj(obj); n = n + 1
            end
        end
        targetList = {}; guiStateMap = {}; partRootMap = {}
        SB.refreshListUI()
        return n
    end

    SB.getAlpha = function() return CONFIG.transparency end
    SB.setAlpha = function(v)
        CONFIG.transparency = math.clamp(v, 0, 1)
        for part, _ in pairs(targetList) do
            setObjTransparency(part, CONFIG.transparency)
        end
        if ui.control and ui.control.tBox then
            ui.control.tBox.Text = string.format("%.2f", CONFIG.transparency)
            ui.control.tLbl.Text = "透明度: " .. ui.control.tBox.Text
        end
        if ui.control and ui.control.alphaKnob then
            ui.control.alphaKnob.Position = UDim2.new(CONFIG.transparency, -7, 0, -4)
        end
    end

    SB.openUI = function()
        if active then return end
        active = true
        containerGui.Enabled = true
        SB.createCrosshair()
        openControlWindow()
    end

    SB.closeUI = function()
        local n = SB.restoreAll()
        active = false
        containerGui.Enabled = false
        SB.removeCrosshair()
        if controlAlphaConn then controlAlphaConn:Disconnect() controlAlphaConn = nil end
        if ui.controlWin and ui.controlWin.frame then ui.controlWin.frame:Destroy() end
        if ui.infoWin and ui.infoWin.frame then ui.infoWin.frame:Destroy() end
        ui.controlWin = nil; ui.infoWin = nil; ui.info = nil; ui.control = nil
        return n
    end

    SB.isActive = function() return active end
    SB.setActive = function(v) active = v end

    SB.shutdown = function()
        SB.restoreAll()
        active = false
        containerGui.Enabled = false
        SB.removeCrosshair()
        if controlAlphaConn then controlAlphaConn:Disconnect() controlAlphaConn = nil end
        if ui.controlWin and ui.controlWin.frame then ui.controlWin.frame:Destroy() end
        if ui.infoWin and ui.infoWin.frame then ui.infoWin.frame:Destroy() end
        ui.controlWin = nil; ui.infoWin = nil; ui.info = nil; ui.control = nil
    end

    return SB
end)()

local isDragging = false
local dragInput = nil
local handleTouches = {}
local function isDragStartInput(input)
    return input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseButton1
end
local function beginExpandDrag(input) isDragging = true; dragInput = input end
UserInputService.InputBegan:Connect(function(input, _gpe)
    if not isDragStartInput(input) then return end
    if not isPointInGui(uiLabel, input.Position.X, input.Position.Y) then return end
    handleTouches[input] = true
    if isDragging then return end
    beginExpandDrag(input)
end)
UserInputService.InputChanged:Connect(function(input)
    if not isDragging then return end
    local matched = (input == dragInput)
        or (dragInput ~= nil
            and dragInput.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement)
    if not matched then return end
    if input.UserInputType ~= Enum.UserInputType.Touch
    and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
    local d = Vector2.new(input.Delta.X, input.Delta.Y)
    if d.Magnitude == 0 then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local vp = cam.ViewportSize
    local fw = Frame.Size.X.Offset
    local fh = Frame.Size.Y.Offset
    setFramePos(clamp(frameX + d.X, 0, vp.X - fw), clamp(frameY + d.Y, 0, vp.Y - fh))
end)
UserInputService.InputEnded:Connect(function(input)
    if handleTouches[input] then handleTouches[input] = nil end
    if not isDragging or input ~= dragInput then return end
    if not isDragStartInput(input) then return end
    local nextInput = nil
    for t in pairs(handleTouches) do nextInput = t; break end
    if nextInput then beginExpandDrag(nextInput)
    else isDragging = false; dragInput = nil end
end)

local function applyFly()
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hum then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    char.Animate.Disabled = true
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        pcall(function() track:AdjustSpeed(0) end)
    end
    for _, state in ipairs(VALID_HUMANOD_STATES) do
        pcall(function() hum:SetStateEnabled(state, false) end)
    end
    local isFreeMode = (flyMode == "自由")
    if isFreeMode then
        hum.PlatformStand = true; hum.AutoRotate = false
    else
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.Swimming)
            hum.PlatformStand = true
        end)
    end
    local startY = hrp.Position.Y
    local bg = Instance.new("BodyGyro")
    bg.P = 9e4; bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.CFrame = hrp.CFrame; bg.Parent = hrp
    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.new(0, 0.1, 0)
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Parent = hrp
    _G._flyData = { bg = bg, bv = bv, hrp = hrp, startY = startY }
    task.spawn(function()
        while isFlying and player.Character and hum and hum.Parent and hum.Health > 0 do
            RunService.Heartbeat:Wait()
            local cam = workspace.CurrentCamera
            if cam then
                local moveDir = hum.MoveDirection
                local maxspeed = 50 * speeds
                local curHrp = char:FindFirstChild("HumanoidRootPart")
                if not curHrp then break end
                local anchorPos = curHrp.Position
                if isFreeMode then
                    local camLook = cam.CFrame.LookVector
                    local camRight = cam.CFrame.RightVector
                    local flatLook = Vector3.new(camLook.X, 0, camLook.Z)
                    local flatRight = Vector3.new(camRight.X, 0, camRight.Z)
                    if flatLook.Magnitude > 0.001 then flatLook = flatLook.Unit end
                    if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit end
                    local fi = moveDir:Dot(flatLook)
                    local ri = moveDir:Dot(flatRight)
                    local combined = camLook * fi + camRight * ri
                    if combined.Magnitude > 1 then combined = combined.Unit end
                    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                    bv.Velocity = combined * maxspeed
                    if flyFaceMode == "水平" then
                        local look = cam.CFrame.LookVector
                        local yaw = math.atan2(-look.X, -look.Z)
                        bg.CFrame = CFrame.new(anchorPos) * CFrame.Angles(0, yaw, 0)
                    else bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation end
                else
                    if flyMode == "悬空" then
                        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                        local hd = Vector3.new(moveDir.X, 0, moveDir.Z)
                        bv.Velocity = (hd.Magnitude > 0) and (hd.Unit * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation
                    elseif flyMode == "绝对锁高" then
                        bv.MaxForce = Vector3.new(9e9, 0, 9e9)
                        local hd = Vector3.new(moveDir.X, 0, moveDir.Z)
                        bv.Velocity = (hd.Magnitude > 0) and (hd.Unit * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation
                        local pos = curHrp.Position
                        curHrp.CFrame = CFrame.new(pos.X, startY, pos.Z) * (curHrp.CFrame - curHrp.Position)
                    else
                        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                        bv.Velocity = (moveDir.Magnitude > 0) and (moveDir * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation
                    end
                end
            end
        end
        removeFly()
    end)
    return true
end

local function startVectorForce()
    local char = player.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local attach = Instance.new("Attachment"); attach.Parent = hrp; accelAttach = attach
    local vf = Instance.new("VectorForce")
    vf.Attachment0 = attach; vf.RelativeTo = Enum.ActuatorRelativeTo.World
    vf.ApplyAtCenterOfMass = true; vf.Force = Vector3.new(0, 0, 0)
    vf.Parent = hrp; accelObj = vf; accelLastMethod = "VectorForce"
    accelConn = RunService.Heartbeat:Connect(function()
        if not speedModeEnabled then return end
        if not vf or not vf.Parent then return end
        local h = player.Character and player.Character:FindFirstChildWhichIsA("Humanoid")
        if not h then return end
        local dir = h.MoveDirection
        if dir.Magnitude > 0.1 then
            local dxz = Vector3.new(dir.X, 0, dir.Z).Unit
            local mag = lockedSpeed * accelForceScale
            vf.Force = Vector3.new(dxz.X * mag, 0, dxz.Z * mag)
        else vf.Force = Vector3.new(0, 0, 0) end
    end)
    return true
end

    end)

    return false
end

local function saveOriginalLighting()
    local Lighting = game:GetService("Lighting")
    originalLighting = {
        Brightness = Lighting.Brightness,
        Ambient = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        ColorShift_Bottom = Lighting.ColorShift_Bottom,
        ColorShift_Top = Lighting.ColorShift_Top,
        FogEnd = Lighting.FogEnd, FogStart = Lighting.FogStart,
        GlobalShadows = Lighting.GlobalShadows,
        EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
        EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
    }
    atmosphereInstance = Lighting:FindFirstChildOfClass("Atmosphere")
    if atmosphereInstance then
        originalAtmosphere = {
            Density = atmosphereInstance.Density,
            Offset = atmosphereInstance.Offset,
            Color = atmosphereInstance.Color,
            Decay = atmosphereInstance.Decay,
            Glare = atmosphereInstance.Glare,
            Haze = atmosphereInstance.Haze
        }
    end
end
local function restoreLighting()
    local Lighting = game:GetService("Lighting")
    for prop, val in pairs(originalLighting) do
        pcall(function() Lighting[prop] = val end)
    end
    if atmosphereInstance and atmosphereInstance.Parent then
        for prop, val in pairs(originalAtmosphere) do
            pcall(function() atmosphereInstance[prop] = val end)
        end
    end
    for _, conn in pairs(lightingFrozenConnections) do
        pcall(function() conn:Disconnect() end)
    end
    lightingFrozenConnections = {}
end
local function freezeLightingProperty(instance, propName, targetValue, isFrozen)
    local connKey = instance:GetFullName() .. "_" .. propName
    if lightingFrozenConnections[connKey] then
        lightingFrozenConnections[connKey]:Disconnect()
        lightingFrozenConnections[connKey] = nil
    end
    pcall(function() instance[propName] = targetValue end)
    if isFrozen then
        lightingFrozenConnections[connKey] = instance:GetPropertyChangedSignal(propName):Connect(function()
            pcall(function()
                if instance[propName] ~= targetValue then instance[propName] = targetValue end
            end)
        end)
    end
end
local function applyNightVision()
    local Lighting = game:GetService("Lighting")
    Lighting.Brightness = nightVisionBrightness
    if nightVisionSettings.ambient then
        freezeLightingProperty(Lighting, "Ambient", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "OutdoorAmbient", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "ColorShift_Bottom", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "ColorShift_Top", Color3.new(1, 1, 1), true)
    else
        if originalLighting.Ambient then freezeLightingProperty(Lighting, "Ambient", originalLighting.Ambient, false) end
        if originalLighting.OutdoorAmbient then freezeLightingProperty(Lighting, "OutdoorAmbient", originalLighting.OutdoorAmbient, false) end
        if originalLighting.ColorShift_Bottom then freezeLightingProperty(Lighting, "ColorShift_Bottom", originalLighting.ColorShift_Bottom, false) end
        if originalLighting.ColorShift_Top then freezeLightingProperty(Lighting, "ColorShift_Top", originalLighting.ColorShift_Top, false) end
    end
    if nightVisionSettings.shadows then freezeLightingProperty(Lighting, "GlobalShadows", false, true)
    else if originalLighting.GlobalShadows ~= nil then freezeLightingProperty(Lighting, "GlobalShadows", originalLighting.GlobalShadows, false) end end
    if nightVisionSettings.envAo then
        freezeLightingProperty(Lighting, "EnvironmentDiffuseScale", 0, true)
        freezeLightingProperty(Lighting, "EnvironmentSpecularScale", 0, true)
    else
        if originalLighting.EnvironmentDiffuseScale then freezeLightingProperty(Lighting, "EnvironmentDiffuseScale", originalLighting.EnvironmentDiffuseScale, false) end
        if originalLighting.EnvironmentSpecularScale then freezeLightingProperty(Lighting, "EnvironmentSpecularScale", originalLighting.EnvironmentSpecularScale, false) end
    end
    if nightVisionSettings.fog then
        freezeLightingProperty(Lighting, "FogEnd", 100000, true)
        freezeLightingProperty(Lighting, "FogStart", 0, true)
    else
        if originalLighting.FogEnd then freezeLightingProperty(Lighting, "FogEnd", originalLighting.FogEnd, false) end
        if originalLighting.FogStart then freezeLightingProperty(Lighting, "FogStart", originalLighting.FogStart, false) end
    end
    if atmosphereInstance and atmosphereInstance.Parent then
        if nightVisionSettings.atmosphere then
            freezeLightingProperty(atmosphereInstance, "Density", 0, true)
            freezeLightingProperty(atmosphereInstance, "Haze", 0, true)
        else
            if originalAtmosphere.Density then freezeLightingProperty(atmosphereInstance, "Density", originalAtmosphere.Density, false) end
            if originalAtmosphere.Haze then freezeLightingProperty(atmosphereInstance, "Haze", originalAtmosphere.Haze, false) end
        end
    end
end

local function makeDeltaDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragInput = false, nil
    local handleTouches = {}
    local function isStartInput(i)
        return i.UserInputType == Enum.UserInputType.Touch
            or i.UserInputType == Enum.UserInputType.MouseButton1
    end
    local function beginDrag(i) dragging = true; dragInput = i end
    handle.InputBegan:Connect(function(input)
        if not isStartInput(input) then return end
        handleTouches[input] = true
        if dragging then return end
        beginDrag(input)
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        local matched = (input == dragInput)
            or (dragInput ~= nil
                and dragInput.UserInputType == Enum.UserInputType.MouseButton1
                and input.UserInputType == Enum.UserInputType.MouseMovement)
        if not matched then return end
        if input.UserInputType ~= Enum.UserInputType.Touch
        and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
        local d = Vector2.new(input.Delta.X, input.Delta.Y)
        if d.Magnitude == 0 then return end
        frame.Position = frame.Position + UDim2.fromOffset(d.X, d.Y)
    end)
    UserInputService.InputEnded:Connect(function(input)
        if handleTouches[input] then handleTouches[input] = nil end
        if input ~= dragInput then return end
        if not isStartInput(input) then return end
        local nextInput = nil
        for t in pairs(handleTouches) do nextInput = t; break end
        if nextInput then beginDrag(nextInput)
        else dragging = false; dragInput = nil end
    end)
end

        for _, item in ipairs(windowButtons) do item.updateUI() end
        if nightVisionEnabled then applyNightVision() end
        tanchuangxiaoxi("已重置所有渲染属性", "透视")
    end)
    local dragging = false
    knob.MouseButton1Down:Connect(function() dragging = true end)
    local inputEndConn = UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if dragging then
            local mp = UserInputService:GetMouseLocation()
            local ap = sliderBg.AbsolutePosition
            local as = sliderBg.AbsoluteSize.X
            local relX = clamp(mp.X - ap.X, 0, as)
            local cp = relX / as
            local nb = minBright + cp * (maxBright - minBright)
            nb = math.floor(nb * 100) / 100
            nightVisionBrightness = nb
            if nightVisionEnabled then applyNightVision() end
        end
        local sp = (nightVisionBrightness - minBright) / (maxBright - minBright)
        knob.Position = UDim2.new(sp, -6, 0, -4.5)
        valueLabel.Text = "亮度: " .. string.format("%.2f", nightVisionBrightness)
        if modeIndex == 4 then updateSpeedButtonText() end
    end)
    sg.Destroying:Connect(function()
        if conn then conn:Disconnect() end
        if inputEndConn then inputEndConn:Disconnect() end
    end)
    return sg
end

local function enableNoclip()
    if not player.Character then return end
    if noclipEnabled then return end
    noclipEnabled = true
    applyNoclip()
    local char, hrp, hum = getCharacterParts()
    if hum then
        if noclipStateConnection then noclipStateConnection:Disconnect() end
        noclipStateConnection = hum.StateChanged:Connect(function(oldState, newState)
            if not noclipEnabled then return end
            if newState == Enum.HumanoidStateType.Jumping
            or newState == Enum.HumanoidStateType.Freefall
            or newState == Enum.HumanoidStateType.FallingDown
            or newState == Enum.HumanoidStateType.Landed then
                applyNoclip()
            end
        end)
    end
    if noclipConnection then noclipConnection:Disconnect() end
    noclipConnection = RunService.Stepped:Connect(function()

    end)

local function reposition()
    local y = startY
    for _, n in ipairs(notifs) do
        local f = n.frame
        TweenService:Create(f, TweenInfo.new(0.2), {Position = UDim2.new(1, -f.Size.X.Offset - 10, 0, y)}):Play()
        y = y + n.height + spacing
    end
end
local function remove(f)
    for i, n in ipairs(notifs) do
        if n.frame == f then table.remove(notifs, i) break end
    end
end
local function processReady()
    while #readyQueue > 0 do
        local n = table.remove(readyQueue, 1)
        if n and n.frame and n.frame.Parent then
            local f, sg = n.frame, n.sg
            local fadeOut = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            local t1 = TweenService:Create(f, fadeOut, {BackgroundTransparency = 1})
            TweenService:Create(n.title, fadeOut, {TextTransparency = 1}):Play()
            TweenService:Create(n.line, fadeOut, {BackgroundTransparency = 1}):Play()
            TweenService:Create(n.msg, fadeOut, {TextTransparency = 1}):Play()
            t1:Play(); t1.Completed:Wait()
            remove(f); sg:Destroy()
        end
    end
    reposition()
end

                    end
                end
            end)
        else

function enableVolumeKey()
    local hasVD = pcall(function() return Enum.KeyCode.VolumeDown end)
    local hasVU = pcall(function() return Enum.KeyCode.VolumeUp end)
    if not hasVD or not hasVU then
        tanchuangxiaoxi("您的设备不支持音量键", "提示"); return
    end
    if volumeKeyEnabled then return end
    volumeKeyConnection = UserInputService.InputBegan:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.VolumeDown then setUIVisible(false)
        elseif input.KeyCode == Enum.KeyCode.VolumeUp then setUIVisible(true) end
    end)
    volumeKeyEnabled = true
    tanchuangxiaoxi("音量键已启用", "设置")
end

function disableVolumeKey()
    if not volumeKeyEnabled then return end
    if volumeKeyConnection then volumeKeyConnection:Disconnect() volumeKeyConnection = nil end
    volumeKeyEnabled = false
    tanchuangxiaoxi("音量键已禁用", "设置")
end

-- ==================== 保护设置菜单 ====================
showProtectMenu = function()

                if not val or val == "" then return end
                val = val:match("^%s*(.-)%s*$")
                if val == "" then return end
                local keywords = {}
                for kw in string.gmatch(val, "([^,]+)") do
                    local trimmed = kw:match("^%s*(.-)%s*$")
                    if trimmed ~= "" then table.insert(keywords, trimmed) end
                end
                if #keywords == 0 then
                    tanchuangxiaoxi("未检测到有效关键词", "保护"); return
                end
                local addedCount = 0
                for _, kw in ipairs(keywords) do
                    local isDuplicate = false
                    for _, item in ipairs(userProtectKeywords) do
                        if item.prefix and string.lower(item.prefix) == string.lower(kw) then
                            isDuplicate = true; break
                        end
                    end
                    if not isDuplicate then
                        table.insert(userProtectKeywords, { prefix = kw, exact = nil })
                        addedCount = addedCount + 1
                    end
                end
                if addedCount > 0 then tanchuangxiaoxi("成功导入 " .. addedCount .. " 个关键词", "保护")
                else tanchuangxiaoxi("没有新增关键词（可能已存在）", "保护") end
            end)
        end
    })
    table.insert(buttons, {
        text = "🎯 从准心拾取关键词",
        bgColor = Color3.fromRGB(80, 180, 100),
        callback = function(menu)
            menu:Destroy()
            if not SmartBuilding.getTargetInCenter or not SmartBuilding.getTargetRoot then
                tanchuangxiaoxi("模块未就绪", "保护"); return
            end
            local t = SmartBuilding.getTargetInCenter()
            if not t then tanchuangxiaoxi("未对准目标", "保护"); return end
            local root = SmartBuilding.getTargetRoot(t)
            if not root then tanchuangxiaoxi("无法获取根对象", "保护"); return end
            local rawName = root.Name
            local prefix = extractKeyword(rawName)
            for _, item in ipairs(userProtectKeywords) do
                if item.exact == rawName then tanchuangxiaoxi("已存在: " .. rawName, "保护"); return end
            end
            table.insert(userProtectKeywords, { prefix = prefix, exact = rawName })
            if prefix == string.lower(rawName) then tanchuangxiaoxi("已添加: " .. rawName, "保护")
            else tanchuangxiaoxi("已添加: " .. prefix .. " | " .. rawName, "保护") end
        end
    })
    table.insert(buttons, {
        text = "📋 管理保护关键词 (" .. #userProtectKeywords .. ")",
        bgColor = Color3.fromRGB(150, 100, 200),
        callback = function(menu)
            menu:Destroy()
            if showKeywordManageMenu then showKeywordManageMenu() end
        end
    })
    table.insert(buttons, {
        text = "🔧 默认保护管理",
        bgColor = Color3.fromRGB(90, 130, 170),
        callback = function(menu)
            menu:Destroy()
            if showDefaultProtectMenu then showDefaultProtectMenu() end
        end
    })
    table.insert(buttons, {
        text = "📥 从文件导入关键词",
        bgColor = Color3.fromRGB(46, 139, 87),
        callback = function(menu)
            menu:Destroy()
            if not isfolder("ProtectKeywords") then
                tanchuangxiaoxi("文件夹 ProtectKeywords 不存在，请先导出一次", "保护"); return
            end
            local files = listfiles("ProtectKeywords")
            local btnList = {}
            for _, file in ipairs(files) do
                if string.sub(file, -4) == ".txt" then
                    local fileName = string.match(file, "([^/\\]+)%.txt$") or file
                    local function renameFile(oldFile, oldName)
                        showInputDialog("输入新文件名", oldName, function(newName)
                            if not newName or newName == "" then return end
                            newName = newName:match("^%s*(.-)%s*$")
                            if newName == "" then return end
                            local newFilePath = "ProtectKeywords/" .. newName .. ".txt"
                            if newFilePath == oldFile then tanchuangxiaoxi("不能覆盖自身！", "保护"); return end
                            if isfile(newFilePath) then
                                createMenu("文件已存在", {
                                    { text = "🔄 覆盖原文件", callback = function(subMenu)
                                        subMenu:Destroy()
                                        delfile(newFilePath)
                                        local content = readfile(oldFile)
                                        writefile(newFilePath, content)
                                        delfile(oldFile)
                                        tanchuangxiaoxi("已重命名并覆盖", "保护")
                                        if showProtectMenu then showProtectMenu() end
                                    end },
                                    { text = "✏️ 重命名", callback = function(subMenu)
                                        subMenu:Destroy()
                                        renameFile(oldFile, oldName)
                                    end },
                                    { text = "❌ 取消", callback = function(subMenu) subMenu:Destroy() end }
                                }, nil)
                            else
                                local content = readfile(oldFile)
                                writefile(newFilePath, content)
                                delfile(oldFile)
                                tanchuangxiaoxi("已重命名", "保护")
                                if showProtectMenu then showProtectMenu() end
                            end
                        end)
                    end

                    table.insert(btnList, {
                        text = "📄 " .. fileName,
                        bgColor = Color3.fromRGB(70, 130, 180),
                        callback = function(subMenu)
                            subMenu:Destroy()
                            local content = readfile(file)
                            local imported = deserializeKeywords(content)
                            local addedCount = 0
                            for _, item in ipairs(imported) do
                                local isDuplicate = false
                                for _, existing in ipairs(userProtectKeywords) do
                                    if item.exact and existing.exact == item.exact then isDuplicate = true; break end
                                    if not item.exact and existing.prefix == item.prefix and not existing.exact then isDuplicate = true; break end
                                end
                                if not isDuplicate then
                                    table.insert(userProtectKeywords, item)
                                    addedCount = addedCount + 1
                                end
                            end
                            tanchuangxiaoxi("已从 [" .. fileName .. "] 导入 " .. addedCount .. " 条", "保护")
                            if showProtectMenu then showProtectMenu() end
                        end,
                        onLongPress = function(subMenu)
                            subMenu:Destroy()
                            createMenu("文件操作: " .. fileName, {
                                { text = "✏️ 重命名", callback = function(m) m:Destroy(); renameFile(file, fileName) end },
                                { text = "🗑️ 删除文件", callback = function(m)
                                    m:Destroy()
                                    delfile(file)
                                    tanchuangxiaoxi("已删除文件: " .. fileName, "保护")
                                    if showProtectMenu then showProtectMenu() end
                                end },
                                { text = "❌ 取消", callback = function(m) m:Destroy() end }
                            }, nil)

        text = "🗑 清空所有关键词",
        bgColor = Color3.fromRGB(178, 34, 34),
        callback = function(menu)

            userProtectKeywords = {}
            local count = 0
            for _ in pairs(protectedList) do count = count + 1 end
            protectedList = {}
            tanchuangxiaoxi(string.format("已清空所有关键词 (%d) 和单目标保护 (%d)", kwCount, count), "保护")
        end
    })
    createMenu("🛡 保护设置", buttons, nil)

-- ==================== 单目标保护主菜单 ====================
showSingleProtectMenu = function()
    local buttons = {}

    -- 统计
    local protectedCount = 0
    for obj, _ in pairs(protectedList) do
        if obj and obj.Parent then protectedCount = protectedCount + 1 end
    end

    -- 1. 添加目标（按名字批量）
    table.insert(buttons, {
        text = "➕ 添加目标（按名字，逗号分隔批量）",
        bgColor = Color3.fromRGB(50, 130, 200),
        callback = function(menu)
            menu:Destroy()
            showInputDialog("输入对象名字（可用逗号分隔批量）", "", function(val)
                if not val or val == "" then return end
                local names = {}
                for n in string.gmatch(val, "([^,]+)") do
                    local trimmed = n:match("^%s*(.-)%s*$")
                    if trimmed ~= "" then table.insert(names, trimmed) end
                end
                if #names == 0 then
                    tanchuangxiaoxi("未检测到有效名字", "保护"); return
                end

                local added, scanned = 0, 0
                local nameSet = {}
                for _, n in ipairs(names) do nameSet[n] = true end

                for _, obj in ipairs(workspace:GetDescendants()) do
                    scanned = scanned + 1
                    if nameSet[obj.Name] and not protectedList[obj] then
                        protectedList[obj] = true
                        added = added + 1
                    end
                end
                tanchuangxiaoxi("已添加 " .. added .. " 个对象 (扫描 " .. scanned .. ")", "保护")
                if showSingleProtectMenu then showSingleProtectMenu() end
            end)
        end
    })

    -- 2. 从准心拾取目标
    table.insert(buttons, {
        text = "🎯 从准心拾取目标",
        bgColor = Color3.fromRGB(80, 180, 100),
        callback = function(menu)
            menu:Destroy()
            if not SmartBuilding.getTargetInCenter or not SmartBuilding.getTargetRoot then
                tanchuangxiaoxi("模块未就绪", "保护"); return
            end
            local t = SmartBuilding.getTargetInCenter()
            if not t then tanchuangxiaoxi("未对准目标", "保护"); return end
            local root = SmartBuilding.getTargetRoot(t)
            if not root then tanchuangxiaoxi("无法获取根对象", "保护"); return end

            if protectedList[root] then
                tanchuangxiaoxi("已保护: " .. root.Name, "保护")
            else
                SmartBuilding.toggleProtect(t)
            end
        end
    })

    -- 3. 管理单目标（进入子列表）
    table.insert(buttons, {
        text = "📋 管理单目标 (" .. protectedCount .. ")",
        bgColor = Color3.fromRGB(150, 100, 200),
        callback = function(menu)
            menu:Destroy()
            if showSingleManageMenu then showSingleManageMenu() end
        end
    })

    -- 4. 从文件导入
    table.insert(buttons, {
        text = "📥 从文件导入单目标",
        bgColor = Color3.fromRGB(46, 139, 87),
        callback = function(menu)
            menu:Destroy()
            if not isfolder("ProtectKeywords") then
                tanchuangxiaoxi("文件夹 ProtectKeywords 不存在，请先导出一次", "保护"); return
            end
            local files = listfiles("ProtectKeywords")
            local btnList = {}
            for _, file in ipairs(files) do
                if string.sub(file, -4) == ".txt" then
                    local fileName = string.match(file, "([^/\\]+)%.txt$") or file
                    table.insert(btnList, {
                        text = "📄 " .. fileName,
                        bgColor = Color3.fromRGB(70, 130, 180),
                        callback = function(subMenu)
                            subMenu:Destroy()
                            local content = readfile(file)
                            -- 解析每行：支持 "name" 或 "prefix,exact" 格式
                            local names = {}
                            for line in string.gmatch(content, "[^\n]+") do
                                local firstComma = string.find(line, ",")
                                if firstComma then
                                    local exact = string.sub(line, firstComma + 1)
                                    if exact ~= "" then
                                        table.insert(names, exact)
                                    else
                                        local prefix = string.sub(line, 1, firstComma - 1)
                                        if prefix ~= "" then table.insert(names, prefix) end
                                    end
                                else
                                    if line ~= "" then table.insert(names, line) end
                                end
                            end
                            local nameSet = {}
                            for _, n in ipairs(names) do nameSet[n] = true end

                            local added = 0
                            for _, obj in ipairs(workspace:GetDescendants()) do
                                if nameSet[obj.Name] and not protectedList[obj] then
                                    protectedList[obj] = true
                                    added = added + 1
                                end
                            end
                            tanchuangxiaoxi("已从 [" .. fileName .. "] 导入 " .. added .. " 个对象", "保护")
                            if showSingleProtectMenu then showSingleProtectMenu() end
                        end
                    })
                end
            end
            if #btnList == 0 then tanchuangxiaoxi("文件夹内暂无 .txt 文件", "保护")
            else createMenu("选择要导入的单目标文件", btnList, showSingleProtectMenu) end
        end
    })

    -- 5. 清空单目标
    table.insert(buttons, {
        text = "🗑 清空单目标 (" .. protectedCount .. ")",
        bgColor = Color3.fromRGB(178, 34, 34),
        callback = function(menu)
            menu:Destroy()
            protectedList = {}
            tanchuangxiaoxi("已清空 " .. protectedCount .. " 个单目标保护", "保护")
            if showSingleProtectMenu then showSingleProtectMenu() end
        end
    })

    createMenu("🎯 单目标保护管理", buttons, showKeywordManageMenu)

-- ==================== 管理单目标（子列表） ====================
showSingleManageMenu = function()
    local buttons = {}

    -- 收集有效单目标，顺便清理失效的
    local validTargets = {}
    for obj, _ in pairs(protectedList) do
        if obj and obj.Parent then
            table.insert(validTargets, obj)
        else
            protectedList[obj] = nil
        end
    end
    local total = #validTargets

    -- 1. 复制所有名字（跳过空值，逗号分隔）
    table.insert(buttons, {
        text = "📋 复制所有名字 (" .. total .. ")",
        bgColor = Color3.fromRGB(50, 130, 200),
        callback = function(menu)
            menu:Destroy()
            if total == 0 then tanchuangxiaoxi("没有可复制的对象", "保护"); return end
            local names = {}
            for _, obj in ipairs(validTargets) do
                local n = obj.Name
                if n and n ~= "" then
                    table.insert(names, n)
                end
            end
            if #names == 0 then
                tanchuangxiaoxi("没有有效的名字可复制", "保护"); return
            end
            local text = table.concat(names, ",")
            if setclipboard then
                pcall(setclipboard, text)
                tanchuangxiaoxi("已复制 " .. #names .. " 个名字", "保护")
            else
                tanchuangxiaoxi("剪贴板不可用", "保护")
            end
        end
    })

    -- 2. 复制所有路径（跳过空值，逗号分隔）
    table.insert(buttons, {
        text = "📋 复制所有路径 (" .. total .. ")",
        bgColor = Color3.fromRGB(70, 140, 180),
        callback = function(menu)
            menu:Destroy()
            if total == 0 then tanchuangxiaoxi("没有可复制的对象", "保护"); return end
            local paths = {}
            for _, obj in ipairs(validTargets) do
                local p = obj:GetFullName()
                if p and p ~= "" then
                    table.insert(paths, p)
                end
            end
            if #paths == 0 then
                tanchuangxiaoxi("没有有效的路径可复制", "保护"); return
            end
            local text = table.concat(paths, ",")
            if setclipboard then
                pcall(setclipboard, text)
                tanchuangxiaoxi("已复制 " .. #paths .. " 条路径", "保护")
            else
                tanchuangxiaoxi("剪贴板不可用", "保护")
            end
        end
    })

    -- 3. 导出到文件
    table.insert(buttons, {
        text = "📤 导出到文件 (" .. total .. ")",
        bgColor = Color3.fromRGB(200, 130, 0),
        callback = function(menu)
            menu:Destroy()
            if total == 0 then tanchuangxiaoxi("没有可导出的对象", "保护"); return end
            if not isfolder("ProtectKeywords") then makefolder("ProtectKeywords") end

            -- 生成导出内容：每行 ",name"（prefix 空，exact=名字）；跳过空名字
            local lines = {}
            for _, obj in ipairs(validTargets) do
                local n = obj.Name
                if n and n ~= "" then
                    table.insert(lines, "," .. n)
                end
            end
            if #lines == 0 then
                tanchuangxiaoxi("没有有效的对象名字可导出", "保护"); return
            end
            local exportText = table.concat(lines, "\n")

            showInputDialog("输入导出文件名", "singles_" .. os.date("%Y%m%d_%H%M%S"), function(filename)
                if not filename or filename == "" then return end
                filename = filename:match("^%s*(.-)%s*$")
                if filename == "" then return end
                local filepath = "ProtectKeywords/" .. filename .. ".txt"
                local function doWrite()
                    writefile(filepath, exportText)
                    tanchuangxiaoxi("已导出 " .. #lines .. " 条到 " .. filepath, "保护")
                end
                if isfile(filepath) then
                    createMenu("文件已存在", {
                        { text = "🔄 覆盖原文件", callback = function(sm) sm:Destroy(); doWrite() end },
                        { text = "✏️ 重命名", callback = function(sm)
                            sm:Destroy()
                            showInputDialog("输入新文件名", filename .. "_1", function(newName)
                                if not newName or newName == "" then return end
                                newName = newName:match("^%s*(.-)%s*$")
                                if newName == "" then return end
                                local np = "ProtectKeywords/" .. newName .. ".txt"
                                if isfile(np) then tanchuangxiaoxi("该文件名依然存在！", "保护")
                                else writefile(np, exportText); tanchuangxiaoxi("已导出: " .. np, "保护") end
                            end)
                        end },
                        { text = "❌ 取消", callback = function(sm) sm:Destroy() end }
                    }, nil)
                else
                    doWrite()
                end
            end)
        end
    })

    -- 4. 对象列表（最多显示 100 个）
    local displayLimit = 100
    local index = 0
    for _, obj in ipairs(validTargets) do
        if index >= displayLimit then break end
        index = index + 1
        local objCopy = obj
        local displayName = obj.Name
        if not displayName or displayName == "" then displayName = "(无名)" end
        if #displayName > 30 then displayName = string.sub(displayName, 1, 27) .. "..." end
        table.insert(buttons, {
            text = "🎯 " .. displayName .. "（点击查看）",
            bgColor = Color3.fromRGB(60, 100, 140),
            callback = function(menu)
                menu:Destroy()
                if showSingleDetailDialog then showSingleDetailDialog(objCopy) end
            end
        })
    end

    if total > displayLimit then
        table.insert(buttons, {
            text = "（还有 " .. (total - displayLimit) .. " 个未显示）",
            bgColor = Color3.fromRGB(60, 60, 60),
            callback = function() end
        })
    end

    createMenu("📋 管理单目标 (" .. total .. ")", buttons, showSingleProtectMenu)
end

local function getMoveVector(dir, rootPart)
    local step = dir * moveStep
    if moveMode == "角色上下" then return rootPart.CFrame.UpVector * step
    elseif moveMode == "角色前后" then return rootPart.CFrame.LookVector * step
    elseif moveMode == "角色左右" then return -rootPart.CFrame.RightVector * step
    elseif moveMode == "屏幕上下" then
        local c = workspace.CurrentCamera
        if c then return c.CFrame.UpVector * step end
    elseif moveMode == "屏幕前后" then
        local c = workspace.CurrentCamera
        if c then return c.CFrame.LookVector * step end
    elseif moveMode == "屏幕左右" then
        local c = workspace.CurrentCamera
        if c then return -c.CFrame.RightVector * step end
    elseif moveMode == "水平上下" then return Vector3.new(0, step, 0)
    elseif moveMode == "水平前后(屏幕)" then
        local c = workspace.CurrentCamera
        if c then
            local l = c.CFrame.LookVector
            local h = Vector3.new(l.X, 0, l.Z)
            if h.Magnitude > 0 then return h.Unit * step end
        end
    elseif moveMode == "水平左右(屏幕)" then
        local c = workspace.CurrentCamera
        if c then
            local r = c.CFrame.RightVector
            local h = Vector3.new(r.X, 0, r.Z)
            if h.Magnitude > 0 then return h.Unit * step end
        end
    end
    return Vector3.new()
end

-- ==================== 按钮绑定：上升 ====================
do
    local pressStartTime = 0; local isPressing = false; local lastInterval = 0
    up.MouseButton1Down:Connect(function()
        if isPressing then return end
        isPressing = true; pressStartTime = tick(); lastInterval = tick()
        local char = player.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local r = char.HumanoidRootPart
            r.CFrame = r.CFrame + getMoveVector(1, r)
        end
    end)
    up.MouseButton1Up:Connect(function() isPressing = false end)
    up.MouseLeave:Connect(function() isPressing = false end)
    RunService.Heartbeat:Connect(function()
        if not isPressing then return end
        if tick() - pressStartTime > 10 then isPressing = false; return end
        local e = tick() - pressStartTime
        if e < 0.3 then return end
        local interval = longPressSpeed * math.pow(0.9, (e - 0.3) / longPressSpeed)
        interval = math.max(0.001, interval)
        if tick() - lastInterval >= interval then
            lastInterval = tick()
            local char = player.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local r = char.HumanoidRootPart
                r.CFrame = r.CFrame + getMoveVector(1, r)
            end
        end
    end)
end

do
    local pressStartTime = 0; local isPressing = false
    local longPressTriggered = false; local longPressDelay = 0.3
    local function doLongPressAction() showMainMenu() end
    local function doShortPressAction()
        Frame.Visible = false
        local vp = getScreenSize()
        local miniX = clamp(frameX + 132, 0, vp.X - 58)
        local miniY = clamp(frameY, 0, vp.Y - 28)
        local mini = Instance.new("Frame")
        mini.Name = "MiniUI"; mini.Parent = main
        mini.BackgroundColor3 = Color3.fromRGB(163, 255, 137)
        mini.BorderColor3 = Color3.fromRGB(103, 221, 213)
        mini.Size = UDim2.new(0, 58, 0, 28)
        mini.Position = UDim2.fromOffset(miniX, miniY); mini.Active = true
        local miniBtn = Instance.new("TextButton")
        miniBtn.Name = "MiniButton"; miniBtn.Parent = mini
        miniBtn.Size = UDim2.new(1, 0, 1, 0)
        miniBtn.BackgroundColor3 = Color3.fromRGB(255, 249, 74)
        miniBtn.Text = "UI"; miniBtn.TextColor3 = Color3.fromRGB(0, 0, 0)
        miniBtn.TextScaled = true; miniBtn.Font = Enum.Font.SourceSans
        miniBtn.Active = true
        local miniX_var, miniY_var = miniX, miniY
        local miniIsDragging = false; local miniDragInput = nil
        local miniHandleTouches = {}; local miniTotalDelta = 0
        local miniPressStartPos = nil
        local function isMiniStartInput(input)
            return input.UserInputType == Enum.UserInputType.Touch
                or input.UserInputType == Enum.UserInputType.MouseButton1
        end
        local function beginMiniDrag(input) miniIsDragging = true; miniDragInput = input end
        UserInputService.InputBegan:Connect(function(input, _gpe)
            if not isMiniStartInput(input) then return end
            if not isPointInGui(miniBtn, input.Position.X, input.Position.Y) then return end
            miniHandleTouches[input] = true
            if miniIsDragging then return end
            beginMiniDrag(input)
            miniTotalDelta = 0
            miniPressStartPos = Vector2.new(input.Position.X, input.Position.Y)
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not miniIsDragging then return end
            local matched = (input == miniDragInput)
                or (miniDragInput ~= nil
                    and miniDragInput.UserInputType == Enum.UserInputType.MouseButton1
                    and input.UserInputType == Enum.UserInputType.MouseMovement)
            if not matched then return end
            if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
            local d = Vector2.new(input.Delta.X, input.Delta.Y)
            miniTotalDelta = miniTotalDelta + d.Magnitude
            if d.Magnitude == 0 then return end
            local cam = workspace.CurrentCamera
            if not cam then return end
            local vp3 = cam.ViewportSize
            local mw = mini.Size.X.Offset; local mh = mini.Size.Y.Offset
            miniX_var = clamp(miniX_var + d.X, 0, vp3.X - mw)
            miniY_var = clamp(miniY_var + d.Y, 0, vp3.Y - mh)
            mini.Position = UDim2.fromOffset(miniX_var, miniY_var)
        end)
        UserInputService.InputEnded:Connect(function(input)
            if miniHandleTouches[input] then miniHandleTouches[input] = nil end
            if not miniIsDragging or input ~= miniDragInput then return end
            if not isMiniStartInput(input) then return end
            local nextInput = nil
            for t in pairs(miniHandleTouches) do nextInput = t; break end
            if nextInput then beginMiniDrag(nextInput); return end
            local endPos = Vector2.new(input.Position.X, input.Position.Y)
            local straightDist = miniPressStartPos and (endPos - miniPressStartPos).Magnitude or 0
            if straightDist < 10 and miniTotalDelta < 15 then
                local vp2 = getScreenSize()
                local fw = Frame.Size.X.Offset; local fh = Frame.Size.Y.Offset
                local nx = clamp(miniX_var - 132, 0, vp2.X - fw)
                local ny = clamp(miniY_var, 0, vp2.Y - fh)
                setFramePos(nx, ny)
                Frame.Visible = true
                mini:Destroy(); miniWindow = nil
            end
            miniIsDragging = false; miniDragInput = nil; miniPressStartPos = nil
        end)
        miniWindow = mini
    end
    hide.MouseButton1Down:Connect(function()
        if isPressing then return end
        isPressing = true; longPressTriggered = false; pressStartTime = tick()
    end)
    hide.MouseButton1Up:Connect(function()
        if not isPressing then return end
        isPressing = false
        local e = tick() - pressStartTime
        if e < longPressDelay then doShortPressAction() end
    end)
    hide.MouseLeave:Connect(function() if isPressing then isPressing = false end end)
    RunService.Heartbeat:Connect(function()
        if not isPressing or longPressTriggered then return end
        if tick() - pressStartTime > 10 then
            isPressing = false; longPressTriggered = false; return
        end
        local e = tick() - pressStartTime
        if e >= longPressDelay then
            longPressTriggered = true; doLongPressAction()
        end
    end)
end

-- ==================== 彻底关闭 ====================
function fullShutdown()
    if isFlying then pcall(toggleFly, false) end
    if speedModeEnabled then pcall(applySpeedMode, false) end
    if noclipEnabled then pcall(disableNoclip) end
    if nightVisionEnabled then pcall(disableNightVision) end
    if SmartBuilding.isActive() then pcall(SmartBuilding.closeUI) end

    pcall(stopCamFreeze)
    pcall(disableCamNoClip)
    if camOffsetConn then camOffsetConn:Disconnect() camOffsetConn = nil end
    camOffsetX, camOffsetY, camOffsetZ = 0, 0, 0
    local char = player.Character
    if char then
        local hum = char:FindFirstChildWhichIsA("Humanoid")
        if hum then pcall(function() hum.CameraOffset = Vector3.zero end) end
    end

    if speedModeConnection then speedModeConnection:Disconnect() speedModeConnection = nil end
    if speedUpdateConnection then speedUpdateConnection:Disconnect() speedUpdateConnection = nil end
    if accelConn then accelConn:Disconnect() accelConn = nil end
    if _G._flyData then removeFly() end
    stopPhysAccel()
    tpwalking = false

    if volumeKeyConnection then volumeKeyConnection:Disconnect() volumeKeyConnection = nil end
    volumeKeyEnabled = false

    if _activeInputDialog then
        pcall(function() _activeInputDialog:Destroy() end)
        _activeInputDialog = nil
    end
    if _activeMenuDialog then
        pcall(function() _activeMenuDialog:Destroy() end)
        _activeMenuDialog = nil
    end
end

    if noclipWindow then noclipWindow:Destroy() noclipWindow = nil end
