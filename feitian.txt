local HTTP_TIMEOUT = 5
local _rawHttpFn = nil
pcall(function() if request then _rawHttpFn = request end end)
pcall(function() if not _rawHttpFn and http_request then _rawHttpFn = http_request end end)
pcall(function() if not _rawHttpFn and syn and syn.request then _rawHttpFn = syn.request end end)
pcall(function() if not _rawHttpFn and http and http.request then _rawHttpFn = http.request end end)

if _rawHttpFn then
    local function safeRequest(opts)
        local result, finished = nil, false
        task.spawn(function()
            local ok, res = pcall(_rawHttpFn, opts)
            if ok and type(res) == "table" then result = res end
            finished = true
        end)
        local t0 = tick()
        while not finished and (tick() - t0) < HTTP_TIMEOUT do
            task.wait(0.1)
        end
        if not finished then
            return {StatusCode = 0, Body = "", Headers = {}}
        end
        return result
    end
    if request then request = function(opts) return safeRequest(opts) end end
    if http_request then http_request = function(opts) return safeRequest(opts) end end
    if syn and syn.request then syn.request = function(opts) return safeRequest(opts) end end
    if http and http.request then http.request = function(opts) return safeRequest(opts) end end
end

local function copyBeaconConfig(src)
    local c = {}
    local base = src or BEACON_DEFAULT
    c.enabled = base.enabled or false
    c.showHighlight = (base.showHighlight ~= false)
    c.showBeam = (base.showBeam ~= false)
    c.showName = (base.showName ~= false)
    c.customName = base.customName or ""
    c.color = base.color or Color3.fromRGB(255, 200, 0)
    c.beamWidth = base.beamWidth or 0.5
    c.beamHeight = base.beamHeight or 250
    c.hlTransparency = base.hlTransparency or 0.9
    c.periodicScan = (base.periodicScan == true)
    c.scanInterval = base.scanInterval or 2.5
    return c
end

local function makeBeaconConfig(overrides)
    local cfg = copyBeaconConfig(BEACON_DEFAULT)
    if overrides then
        for k, v in pairs(overrides) do
            cfg[k] = v
        end
    end
    return cfg
end

local function extractKeyword(name)
    if not name or name == "" then return "" end
    local s = name
    s = s:gsub("%s*%(%d+%)%s*$", "")
    s = s:gsub("[_%s]+%d+%s*$", "")
    s = s:gsub("%d+$", "")
    s = s:gsub("[_%s]+$", "")
    s = s:match("^%s*(.-)%s*$") or s
    if s == "" then s = name end
    return string.lower(s)
end

-- ==================== 序列化 / 反序列化（16字段 "|" 分隔） ====================
local BEACON_FIELD_COUNT = 16

local function serializeKeywords(keywords)
    local lines = {}
    for _, item in ipairs(keywords) do
        local prefix = item.prefix or ""
        local exact = item.exact or ""
        local b = item.beacon
        if b and b.enabled then
            local c = b.color
            table.insert(lines, table.concat({
                prefix, exact,
                "1",
                b.enabled and "1" or "0",
                b.showHighlight and "1" or "0",
                b.showBeam and "1" or "0",
                b.showName and "1" or "0",
                (b.customName or ""):gsub("|", "/"),
                tostring(math.floor(c.R * 255 + 0.5)),
                tostring(math.floor(c.G * 255 + 0.5)),
                tostring(math.floor(c.B * 255 + 0.5)),
                string.format("%.3f", b.beamWidth or 0.5),
                string.format("%.3f", b.beamHeight or 250),
                string.format("%.3f", b.hlTransparency or 0.9),
                b.periodicScan and "1" or "0",
                string.format("%.3f", b.scanInterval or 2.5),
            }, "|"))
        else
            table.insert(lines, prefix .. "," .. exact)
        end
    end
    return table.concat(lines, "\n")
end

local function isBeaconLine(line)
    local _, count = line:gsub("|", "")
    return count >= (BEACON_FIELD_COUNT - 1)
end

local function deserializeKeywords(content)
    local keywords = {}
    for line in string.gmatch(content, "[^\n]+") do
        if isBeaconLine(line) then
            local fields = {}
            for f in string.gmatch(line .. "|", "([^|]*)|") do
                table.insert(fields, f)
            end
            if #fields >= BEACON_FIELD_COUNT then
                local prefix = fields[1]
                local exact = fields[2]
                if prefix ~= "" or exact ~= "" then
                    local item = { prefix = prefix, exact = (exact ~= "" and exact) or nil }
                    local b = makeBeaconConfig()
                    b.enabled = (fields[4] == "1")
                    b.showHighlight = (fields[5] == "1")
                    b.showBeam = (fields[6] == "1")
                    b.showName = (fields[7] == "1")
                    b.customName = fields[8] or ""
                    local r = tonumber(fields[9]) or 255
                    local g = tonumber(fields[10]) or 200
                    local bl = tonumber(fields[11]) or 0
                    b.color = Color3.fromRGB(r, g, bl)
                    b.beamWidth = tonumber(fields[12]) or 0.5
                    b.beamHeight = tonumber(fields[13]) or 250
                    b.hlTransparency = tonumber(fields[14]) or 0.9
                    b.periodicScan = (fields[15] == "1")
                    b.scanInterval = tonumber(fields[16]) or 2.5
                    item.beacon = b
                    table.insert(keywords, item)
                end
            end
        else
            local firstComma = string.find(line, ",")
            if firstComma then
                local prefix = string.sub(line, 1, firstComma - 1)
                local exact = string.sub(line, firstComma + 1)
                if prefix ~= "" then
                    table.insert(keywords, { prefix = prefix, exact = (exact ~= "" and exact) or nil })
                end
            else
                if line ~= "" then
                    table.insert(keywords, { prefix = line, exact = nil })
                end
            end
        end
    end
    return keywords
end

-- ==================== 通用滑块拖动（多指隔离） ====================
local function makeSliderDraggable(knob, sliderBg, knobOffsetX, knobOffsetY, onUpdate, onEnd)
    local dragging = false
    local dragInput = nil
    local lastCp = 0

    local function updateFromInput(input)
        local ap = sliderBg.AbsolutePosition
        local as = sliderBg.AbsoluteSize.X
        if as <= 0 then return end
        local relX = math.clamp(input.Position.X - ap.X, 0, as)
        local cp = relX / as
        lastCp = cp
        knob.Position = UDim2.new(cp, knobOffsetX, 0, knobOffsetY)
        if onUpdate then onUpdate(cp) end
    end

    local connKnobBegan = knob.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
        dragging = true
        dragInput = input
        updateFromInput(input)
    end)

    local connChanged = UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        local matched = (input == dragInput)
            or (dragInput ~= nil
                and dragInput.UserInputType == Enum.UserInputType.MouseButton1
                and input.UserInputType == Enum.UserInputType.MouseMovement)
        if not matched then return end
        if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
        updateFromInput(input)
    end)

    local connEnded = UserInputService.InputEnded:Connect(function(input)
        if input ~= dragInput then return end
        dragging = false
        dragInput = nil
        if onEnd then onEnd(lastCp) end
    end)

    return function()
        if connKnobBegan then connKnobBegan:Disconnect() end
        if connChanged then connChanged:Disconnect() end
        if connEnded then connEnded:Disconnect() end
    end
end

        if part.Transparency >= CONFIG.alreadyInvisibleThreshold then return true end
        return false
    end

    local function checkIgnored(part)
        if not part or not part:IsA("BasePart") then return true, "非 BasePart" end
        if part:IsA("Terrain") then return true, "Terrain" end
        if isAncestorProtected(part) then return true, "单目标保护" end
        if ancestorsMatchUserKw(part) then return true, "用户关键词" end
        if player.Character and part:IsDescendantOf(player.Character) then return true, "玩家角色" end
        if nameMatches(part, CONFIG.invisibleWallKeywords)
            and not nameMatches(part, CONFIG.buildingKeywords) then return true, "空气墙名" end
        if nameMatches(part, CONFIG.strongIgnore) then return true, "强忽略词" end
        if exactMatchAny(part.Name, CONFIG.exactIgnore) then return true, "忽略词" end
        if ancestorsMatchActiveProtect(part) then return true, "默认保护" end
        if ancestorsMatch(part, CONFIG.spawnKeywords) then return true, "刷新点/出生点" end
        if advancedProtect.classes then
            for _, cls in ipairs(CONFIG.protectClasses) do
                if part:IsA(cls) then return true, "保护类 " .. cls end
            end
        end
        if advancedProtect.prompt and ancestorsHavePrompt(part) then
            return true, "含交互组件"
        end
        if advancedProtect.humanoid and ancestorsHaveHumanoid(part) then
            return true, "含 Humanoid"
        end
        return false, nil
    end
    local function isIgnored(part) return (checkIgnored(part)) end

    local function setObjTransparency(obj, trans)
        if not obj or not obj.Parent then return end
        if obj:IsA("BasePart") and not obj:IsA("Terrain") then
            local rec = targetList[obj]
            if not rec then
                rec = { transparency = obj.Transparency, material = obj.Material,
                        castShadow = obj.CastShadow, reflectance = obj.Reflectance,
                        canCollide = obj.CanCollide }
                targetList[obj] = rec
            end
            obj.Transparency = trans
            obj.Material = Enum.Material.SmoothPlastic
            obj.CastShadow = false; obj.Reflectance = 0
            if collisionDeleteEnabled then
                obj.CanCollide = false
            end
            return
        end
        if obj:IsA("Decal") or obj:IsA("Texture") then
            local rec = targetList[obj]
            if not rec then rec = { transparency = obj.Transparency } end
            targetList[obj] = rec; obj.Transparency = trans
            return
        end
        if obj:IsA("SurfaceAppearance") then
            local rec = targetList[obj]
            if not rec then
                rec = { colorMap = obj.ColorMap, metalnessMap = obj.MetalnessMap,
                        roughnessMap = obj.RoughnessMap, normalMap = obj.NormalMap }
                targetList[obj] = rec
            end
            obj.ColorMap = ""; obj.MetalnessMap = ""; obj.RoughnessMap = ""; obj.NormalMap = ""
            return
        end
        if obj:IsA("SurfaceGui") or obj:IsA("BillboardGui") then
            local rec = guiStateMap[obj]
            if not rec then rec = { enabled = obj.Enabled } end
            guiStateMap[obj] = rec; obj.Enabled = false
            return
        end
        if obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail")
            or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
            local rec = guiStateMap[obj]
            if not rec then rec = { enabled = obj.Enabled } end
            guiStateMap[obj] = rec; obj.Enabled = false
            return
        end
    end

    local function restoreObj(obj)
        if not obj or not obj.Parent then return end
        if obj:IsA("BasePart") and not obj:IsA("Terrain") then
            local rec = targetList[obj]
            if rec then
                obj.Transparency = rec.transparency
                obj.Material = rec.material
                obj.CastShadow = rec.castShadow
                obj.Reflectance = rec.reflectance
                if rec.canCollide ~= nil then
                    obj.CanCollide = rec.canCollide
                end
            end
            return
        end
        if obj:IsA("Decal") or obj:IsA("Texture") then
            local rec = targetList[obj]
            if rec then obj.Transparency = rec.transparency end
            return
        end
        if obj:IsA("SurfaceAppearance") then
            local rec = targetList[obj]
            if rec then
                obj.ColorMap = rec.colorMap
                obj.MetalnessMap = rec.metalnessMap
                obj.RoughnessMap = rec.roughnessMap
                obj.NormalMap = rec.normalMap
            end
            return
        end
        if obj:IsA("SurfaceGui") or obj:IsA("BillboardGui") then
            local rec = guiStateMap[obj]
            if rec then obj.Enabled = rec.enabled end
            return
        end
        if obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail")
            or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
            local rec = guiStateMap[obj]
            if rec then obj.Enabled = rec.enabled end
            return
        end
    end

    local function getTargetRoot(part)
        if not part then return nil end
        local cur = part
        while cur and cur ~= workspace do
            if cur:IsA("Model") then
                local count, tooBig = 0, false
                for _, d in ipairs(cur:GetDescendants()) do
                    if d:IsA("BasePart") then
                        count = count + 1
                        if count > CONFIG.maxModelParts then tooBig = true break end
                    end
                end
                if not tooBig then return cur end
            end
            cur = cur.Parent
        end
        return part
    end
    SB.getTargetRoot = getTargetRoot

    local function collectTargetParts(root)
        local list = {}
        if not root then return list end
        local function consider(obj)
            if isNameProtected(obj) then return end
            if obj:IsA("BasePart") and not obj:IsA("Terrain") then
                if not isInvisibleWall(obj) then table.insert(list, obj) end
            elseif obj:IsA("Decal") or obj:IsA("Texture") or obj:IsA("SurfaceAppearance") then
                table.insert(list, obj)
            elseif obj:IsA("SurfaceGui") or obj:IsA("BillboardGui") then
                table.insert(list, obj)
            elseif obj:IsA("ParticleEmitter") or obj:IsA("Beam") or obj:IsA("Trail")
                or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                table.insert(list, obj)
            end
        end
        consider(root)
        for _, d in ipairs(root:GetDescendants()) do consider(d) end
        return list
    end

    local function makeDraggable(frame, handle)
        handle = handle or frame
        local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
        handle.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.MouseButton1
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
            dragging = true; dragInput = input
            dragStart = Vector2.new(input.Position.X, input.Position.Y)
            startPos = frame.Position
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not dragging then return end
            local matched = (input == dragInput)
                or (dragInput and dragInput.UserInputType == Enum.UserInputType.MouseButton1
                    and input.UserInputType == Enum.UserInputType.MouseMovement)
            if not matched then return end
            if input.UserInputType ~= Enum.UserInputType.MouseMovement
            and input.UserInputType ~= Enum.UserInputType.Touch then return end
            local cur = Vector2.new(input.Position.X, input.Position.Y)
            local d = cur - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input == dragInput then dragging = false; dragInput = nil end
        end)
    end

    SB.smartScan = function()
        if scanRunning then return end
        scanRunning = true
        if ui.control then showNotification("扫描中...") end
        task.spawn(function()
            local t0 = tick()
            local cam = workspace.CurrentCamera
            if not cam then scanRunning = false return end
            local range = CONFIG.scanRange
            local camPos = cam.CFrame.Position
            local look = cam.CFrame.LookVector
            local boxCenter = camPos + look * (range * 0.35)
            local boxSize = Vector3.new(range, range * 0.6, range)
            local overlapParams = OverlapParams.new()
            overlapParams.FilterType = Enum.RaycastFilterType.Exclude
            overlapParams.FilterDescendantsInstances = { player.Character }
            overlapParams.MaxParts = CONFIG.maxPartsPerScan
            local parts = workspace:GetPartBoundsInBox(CFrame.new(boxCenter), boxSize, overlapParams)
            local rootSet, roots = {}, {}
            local total = #parts
            for i, part in ipairs(parts) do
                if part:IsA("BasePart") and not part:IsA("Terrain")
                    and not targetList[part] and not isInvisibleWall(part) then
                    if not isIgnored(part) then
                        local root = getTargetRoot(part)
                        if root and not rootSet[root] and not isNameProtected(root) then
                            rootSet[root] = true
                            table.insert(roots, root)
                        end
                    end
                end
                if i % CONFIG.batchClassify == 0 then
                    task.wait()
                    if ui.control then
                        showNotification(string.format("分类 %d%%", math.floor(i / total * 100)))
                    end
                end
            end
            local applied, processedRoots = 0, 0
            for _, root in ipairs(roots) do
                local subParts = collectTargetParts(root)
                for _, p in ipairs(subParts) do
                    if targetList[p] == nil and guiStateMap[p] == nil then
                        local blocked = false
                        if p:IsA("BasePart") then
                            blocked = isIgnored(p)
                        else
                            blocked = isNameProtected(p)
                        end
                        if not blocked then
                            setObjTransparency(p, CONFIG.transparency)
                            partRootMap[p] = root
                            applied = applied + 1
                        end
                    end
                end
                processedRoots = processedRoots + 1
                if processedRoots % CONFIG.batchApply == 0 then
                    task.wait()
                    if ui.control then
                        showNotification(string.format("应用 %d/%d", processedRoots, #roots))
                    end
                end
            end
            SB.refreshListUI()
            local elapsed = tick() - t0
            if ui.control then
                showNotification(string.format("完成 %d (%.2fs)", applied, elapsed))
            end
            tanchuangxiaoxi(string.format("共修改 %d 个部件  (%.2fs)", applied, elapsed), "修改模式")
            scanRunning = false
        end)
    end

    SB.addTarget = function(target)
        if not target then return false end
        if not (target:IsA("BasePart") or target:IsA("Model")) then
            showNotification("无效目标"); return false
        end
        local root = getTargetRoot(target)
        if not root then return false end

        if isNameProtected(root) then
            showNotification("该目标已被保护")
            tanchuangxiaoxi("已被保护，无法添加: " .. root.Name, "保护")
            return false
        end

        local parts = collectTargetParts(root)
        if #parts == 0 then showNotification("无有效部件"); return false end

        local addedCount = 0
        for _, p in ipairs(parts) do
            if targetList[p] ~= nil or guiStateMap[p] ~= nil then
                addedCount = addedCount + 1
            end
        end
        if addedCount == #parts and addedCount > 0 then
            SB.removeTarget(root)
            showNotification("已取消: " .. root.Name)
            return false
        end

        local added = 0
        for _, p in ipairs(parts) do
            if player.Character and p:IsDescendantOf(player.Character) then continue end
            local blocked = false
            if p:IsA("BasePart") then
                blocked = isIgnored(p)
            else
                blocked = isNameProtected(p)
            end
            if blocked then continue end
            if targetList[p] == nil and guiStateMap[p] == nil then
                setObjTransparency(p, CONFIG.transparency)
                partRootMap[p] = root
                added = added + 1
            end
        end
        SB.refreshListUI()
        if added > 0 then
            showNotification("已添加 " .. root.Name .. " (" .. added .. ")")
        else
            showNotification("全部被拦，未添加")
        end
        return added > 0
    end

        if protectedList[root] then
            protectedList[root] = nil
            tanchuangxiaoxi("已取消保护: " .. root.Name, "保护")
        else
            protectedList[root] = true
            local restored = 0
            local allObjs = { root }
            for _, d in ipairs(root:GetDescendants()) do
                table.insert(allObjs, d)
            end
            for _, p in ipairs(allObjs) do
                if targetList[p] ~= nil or guiStateMap[p] ~= nil then
                    restoreObj(p)
                    targetList[p] = nil; guiStateMap[p] = nil; partRootMap[p] = nil
                    restored = restored + 1
                end
            end
            SB.refreshListUI()
            local extra = (restored > 0) and (" (已恢复 " .. restored .. " 部件)") or ""
            tanchuangxiaoxi("已保护: " .. root.Name .. extra, "保护")
        end
    end

local isDragging = false
local dragInput = nil
local handleTouches = {}
local function isDragStartInput(input)
    return input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseButton1
end
local function beginExpandDrag(input) isDragging = true; dragInput = input end
UserInputService.InputBegan:Connect(function(input, _gpe)
    if not isDragStartInput(input) then return end
    if not isPointInGui(uiLabel, input.Position.X, input.Position.Y) then return end
    handleTouches[input] = true
    if isDragging then return end
    beginExpandDrag(input)
end)
UserInputService.InputChanged:Connect(function(input)
    if not isDragging then return end
    local matched = (input == dragInput)
        or (dragInput ~= nil
            and dragInput.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement)
    if not matched then return end
    if input.UserInputType ~= Enum.UserInputType.Touch
    and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
    local d = Vector2.new(input.Delta.X, input.Delta.Y)
    if d.Magnitude == 0 then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local vp = cam.ViewportSize
    local fw = Frame.Size.X.Offset
    local fh = Frame.Size.Y.Offset
    setFramePos(clamp(frameX + d.X, 0, vp.X - fw), clamp(frameY + d.Y, 0, vp.Y - fh))
end)
UserInputService.InputEnded:Connect(function(input)
    if handleTouches[input] then handleTouches[input] = nil end
    if not isDragging or input ~= dragInput then return end
    if not isDragStartInput(input) then return end
    local nextInput = nil
    for t in pairs(handleTouches) do nextInput = t; break end
    if nextInput then beginExpandDrag(nextInput)
    else isDragging = false; dragInput = nil end
end)

local function applyFly()
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hum then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    char.Animate.Disabled = true
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        pcall(function() track:AdjustSpeed(0) end)
    end
    for _, state in ipairs(VALID_HUMANOD_STATES) do
        pcall(function() hum:SetStateEnabled(state, false) end)
    end
    local isFreeMode = (flyMode == "自由")
    if isFreeMode then
        hum.PlatformStand = true; hum.AutoRotate = false
    else
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.Swimming)
            hum.PlatformStand = true
        end)
    end
    local startY = hrp.Position.Y
    local bg = Instance.new("BodyGyro")
    bg.P = 9e4; bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.CFrame = hrp.CFrame; bg.Parent = hrp
    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.new(0, 0.1, 0)
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Parent = hrp
    _G._flyData = { bg = bg, bv = bv, hrp = hrp, startY = startY }
    task.spawn(function()
        while isFlying and player.Character and hum and hum.Parent and hum.Health > 0 do
            RunService.Heartbeat:Wait()
            local cam = workspace.CurrentCamera
            if cam then
                local moveDir = hum.MoveDirection
                local maxspeed = 50 * speeds
                local curHrp = char:FindFirstChild("HumanoidRootPart")
                if not curHrp then break end
                local anchorPos = curHrp.Position
                if isFreeMode then
                    local camLook = cam.CFrame.LookVector
                    local camRight = cam.CFrame.RightVector
                    local flatLook = Vector3.new(camLook.X, 0, camLook.Z)
                    local flatRight = Vector3.new(camRight.X, 0, camRight.Z)
                    if flatLook.Magnitude > 0.001 then flatLook = flatLook.Unit end
                    if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit end
                    local fi = moveDir:Dot(flatLook)
                    local ri = moveDir:Dot(flatRight)
                    local combined = camLook * fi + camRight * ri
                    if combined.Magnitude > 1 then combined = combined.Unit end
                    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                    bv.Velocity = combined * maxspeed
                    if flyFaceMode == "水平" then
                        local look = cam.CFrame.LookVector
                        local yaw = math.atan2(-look.X, -look.Z)
                        bg.CFrame = CFrame.new(anchorPos) * CFrame.Angles(0, yaw, 0)
                    else bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation end
                else
                    if flyMode == "悬空" then
                        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                        local hd = Vector3.new(moveDir.X, 0, moveDir.Z)
                        bv.Velocity = (hd.Magnitude > 0) and (hd.Unit * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation
                    elseif flyMode == "绝对锁高" then
                        bv.MaxForce = Vector3.new(9e9, 0, 9e9)
                        local hd = Vector3.new(moveDir.X, 0, moveDir.Z)
                        bv.Velocity = (hd.Magnitude > 0) and (hd.Unit * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation
                        local pos = curHrp.Position
                        curHrp.CFrame = CFrame.new(pos.X, startY, pos.Z) * (curHrp.CFrame - curHrp.Position)
                    else
                        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                        bv.Velocity = (moveDir.Magnitude > 0) and (moveDir * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = CFrame.new(anchorPos) * cam.CFrame.Rotation
                    end
                end
            end
        end
        removeFly()
    end)
    return true
end

local function resetHumanoidAfterFly()
    local char = player.Character
    if not char then return end

local function stopPhysAccel()
    if accelConn then accelConn:Disconnect() accelConn = nil end
    if accelObj then pcall(function() accelObj:Destroy() end) accelObj = nil end
    if accelAttach then pcall(function() accelAttach:Destroy() end) accelAttach = nil end
    accelLastMethod = nil
end
local function isPhysMode()
    return speedSubMode == "持续力" or speedSubMode == "物理速度" or speedSubMode == "强制位移"
end
local function startVectorForce()
    local char = player.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local attach = Instance.new("Attachment"); attach.Parent = hrp; accelAttach = attach
    local vf = Instance.new("VectorForce")
    vf.Attachment0 = attach; vf.RelativeTo = Enum.ActuatorRelativeTo.World
    vf.ApplyAtCenterOfMass = true; vf.Force = Vector3.new(0, 0, 0)
    vf.Parent = hrp; accelObj = vf; accelLastMethod = "VectorForce"
    accelConn = RunService.Heartbeat:Connect(function()
        if not speedModeEnabled then return end
        if not vf or not vf.Parent then return end
        local h = player.Character and player.Character:FindFirstChildWhichIsA("Humanoid")
        if not h then return end
        local dir = h.MoveDirection
        if dir.Magnitude > 0.1 then
            local dxz = Vector3.new(dir.X, 0, dir.Z).Unit
            local mag = lockedSpeed * accelForceScale
            vf.Force = Vector3.new(dxz.X * mag, 0, dxz.Z * mag)
        else vf.Force = Vector3.new(0, 0, 0) end
    end)
    return true
end
local function startBodyVelocity()
    local char = player.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(1e6, 0, 1e6); bv.P = 1250
    bv.Velocity = Vector3.new(0, 0, 0); bv.Parent = hrp
    accelObj = bv; accelLastMethod = "BodyVelocity"
    accelConn = RunService.Heartbeat:Connect(function()
        if not speedModeEnabled then return end
        if not bv or not bv.Parent then return end
        local h = player.Character and player.Character:FindFirstChildWhichIsA("Humanoid")
        if not h then return end
        local dir = h.MoveDirection
        if dir.Magnitude > 0.1 then
            local dxz = Vector3.new(dir.X, 0, dir.Z).Unit
            bv.Velocity = Vector3.new(dxz.X * lockedSpeed, 0, dxz.Z * lockedSpeed)
        else bv.Velocity = Vector3.new(0, 0, 0) end

end
local function startCFramePush()
    local char = player.Character
    if not char then return false end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    accelLastMethod = "CFrame"
    accelConn = RunService.RenderStepped:Connect(function(dt)
        if not speedModeEnabled then return end
        local c = player.Character
        if not c then return end
        local root = c:FindFirstChild("HumanoidRootPart")
        local h = c:FindFirstChildWhichIsA("Humanoid")
        if not root or not h then return end
        local dir = h.MoveDirection
        if dir.Magnitude > 0.1 then
            local dxz = Vector3.new(dir.X, 0, dir.Z).Unit
            root.CFrame = root.CFrame + dxz * lockedSpeed * dt
        end

    return true
end
local function startPhysAccel()
    stopPhysAccel()
    if speedSubMode == "持续力" then return startVectorForce()
    elseif speedSubMode == "物理速度" then return startBodyVelocity()
    elseif speedSubMode == "强制位移" then return startCFramePush() end
    return false
end

local function applySpeedMode(enable)
    if enable then
        if speedModeEnabled then return end
        forceDisableFly()
        local char = player.Character
        local currentWS = 16
        if char then
            local hum = char:FindFirstChildWhichIsA("Humanoid")
            if hum then currentWS = hum.WalkSpeed end
        end
        if currentWS <= 0 then currentWS = 16 end
        originalWalkSpeed = currentWS; lockedSpeed = currentWS
        speedModeEnabled = true
        if speedSubMode == "修改" then
            if speedModeConnection then speedModeConnection:Disconnect() end
            speedModeConnection = RunService.Heartbeat:Connect(function()
                if not speedModeEnabled then return end
                if speedSubMode ~= "修改" then return end

                if c then
                    local h = c:FindFirstChildWhichIsA("Humanoid")
                    if h then pcall(function() h.WalkSpeed = lockedSpeed end) end
                end
            end)

            local hum = char:FindFirstChildWhichIsA("Humanoid")

local function saveOriginalLighting()
    local Lighting = game:GetService("Lighting")
    originalLighting = {
        Brightness = Lighting.Brightness,
        Ambient = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        ColorShift_Bottom = Lighting.ColorShift_Bottom,
        ColorShift_Top = Lighting.ColorShift_Top,
        FogEnd = Lighting.FogEnd, FogStart = Lighting.FogStart,
        GlobalShadows = Lighting.GlobalShadows,
        EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
        EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
    }
    atmosphereInstance = Lighting:FindFirstChildOfClass("Atmosphere")
    if atmosphereInstance then
        originalAtmosphere = {
            Density = atmosphereInstance.Density,
            Offset = atmosphereInstance.Offset,
            Color = atmosphereInstance.Color,
            Decay = atmosphereInstance.Decay,
            Glare = atmosphereInstance.Glare,
            Haze = atmosphereInstance.Haze
        }
    end
end
local function restoreLighting()
    local Lighting = game:GetService("Lighting")
    for prop, val in pairs(originalLighting) do
        pcall(function() Lighting[prop] = val end)
    end
    if atmosphereInstance and atmosphereInstance.Parent then
        for prop, val in pairs(originalAtmosphere) do
            pcall(function() atmosphereInstance[prop] = val end)
        end
    end
    for _, conn in pairs(lightingFrozenConnections) do
        pcall(function() conn:Disconnect() end)
    end
    lightingFrozenConnections = {}
end
local function freezeLightingProperty(instance, propName, targetValue, isFrozen)
    local connKey = instance:GetFullName() .. "_" .. propName
    if lightingFrozenConnections[connKey] then
        lightingFrozenConnections[connKey]:Disconnect()
        lightingFrozenConnections[connKey] = nil
    end
    pcall(function() instance[propName] = targetValue end)
    if isFrozen then
        lightingFrozenConnections[connKey] = instance:GetPropertyChangedSignal(propName):Connect(function()
            pcall(function()
                if instance[propName] ~= targetValue then instance[propName] = targetValue end
            end)
        end)
    end
end
local function applyNightVision()
    local Lighting = game:GetService("Lighting")
    Lighting.Brightness = nightVisionBrightness
    if nightVisionSettings.ambient then
        freezeLightingProperty(Lighting, "Ambient", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "OutdoorAmbient", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "ColorShift_Bottom", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "ColorShift_Top", Color3.new(1, 1, 1), true)
    else
        if originalLighting.Ambient then freezeLightingProperty(Lighting, "Ambient", originalLighting.Ambient, false) end
        if originalLighting.OutdoorAmbient then freezeLightingProperty(Lighting, "OutdoorAmbient", originalLighting.OutdoorAmbient, false) end
        if originalLighting.ColorShift_Bottom then freezeLightingProperty(Lighting, "ColorShift_Bottom", originalLighting.ColorShift_Bottom, false) end
        if originalLighting.ColorShift_Top then freezeLightingProperty(Lighting, "ColorShift_Top", originalLighting.ColorShift_Top, false) end
    end
    if nightVisionSettings.shadows then freezeLightingProperty(Lighting, "GlobalShadows", false, true)
    else if originalLighting.GlobalShadows ~= nil then freezeLightingProperty(Lighting, "GlobalShadows", originalLighting.GlobalShadows, false) end end
    if nightVisionSettings.envAo then
        freezeLightingProperty(Lighting, "EnvironmentDiffuseScale", 0, true)
        freezeLightingProperty(Lighting, "EnvironmentSpecularScale", 0, true)
    else
        if originalLighting.EnvironmentDiffuseScale then freezeLightingProperty(Lighting, "EnvironmentDiffuseScale", originalLighting.EnvironmentDiffuseScale, false) end
        if originalLighting.EnvironmentSpecularScale then freezeLightingProperty(Lighting, "EnvironmentSpecularScale", originalLighting.EnvironmentSpecularScale, false) end
    end
    if nightVisionSettings.fog then
        freezeLightingProperty(Lighting, "FogEnd", 100000, true)
        freezeLightingProperty(Lighting, "FogStart", 0, true)
    else
        if originalLighting.FogEnd then freezeLightingProperty(Lighting, "FogEnd", originalLighting.FogEnd, false) end
        if originalLighting.FogStart then freezeLightingProperty(Lighting, "FogStart", originalLighting.FogStart, false) end
    end
    if atmosphereInstance and atmosphereInstance.Parent then
        if nightVisionSettings.atmosphere then
            freezeLightingProperty(atmosphereInstance, "Density", 0, true)
            freezeLightingProperty(atmosphereInstance, "Haze", 0, true)
        else
            if originalAtmosphere.Density then freezeLightingProperty(atmosphereInstance, "Density", originalAtmosphere.Density, false) end
            if originalAtmosphere.Haze then freezeLightingProperty(atmosphereInstance, "Haze", originalAtmosphere.Haze, false) end
        end
    end
end

        if nightVisionWindow then nightVisionWindow:Destroy() nightVisionWindow = nil end
        if nightVisionMaintainConnection then nightVisionMaintainConnection:Disconnect() nightVisionMaintainConnection = nil end
        restoreLighting()
        nightVisionEnabled = false
        tanchuangxiaoxi("已关闭透视", "透视")
        updateMainButtonText(); updateSpeedButtonText()
    end
end

local function makeDeltaDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragInput = false, nil
    local handleTouches = {}
    local function isStartInput(i)
        return i.UserInputType == Enum.UserInputType.Touch
            or i.UserInputType == Enum.UserInputType.MouseButton1
    end
    local function beginDrag(i) dragging = true; dragInput = i end
    handle.InputBegan:Connect(function(input)
        if not isStartInput(input) then return end
        handleTouches[input] = true
        if dragging then return end
        beginDrag(input)
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        local matched = (input == dragInput)
            or (dragInput ~= nil
                and dragInput.UserInputType == Enum.UserInputType.MouseButton1
                and input.UserInputType == Enum.UserInputType.MouseMovement)
        if not matched then return end
        if input.UserInputType ~= Enum.UserInputType.Touch
        and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
        local d = Vector2.new(input.Delta.X, input.Delta.Y)
        if d.Magnitude == 0 then return end
        frame.Position = frame.Position + UDim2.fromOffset(d.X, d.Y)
    end)
    UserInputService.InputEnded:Connect(function(input)
        if handleTouches[input] then handleTouches[input] = nil end
        if input ~= dragInput then return end
        if not isStartInput(input) then return end
        local nextInput = nil
        for t in pairs(handleTouches) do nextInput = t; break end
        if nextInput then beginDrag(nextInput)
        else dragging = false; dragInput = nil end
    end)
end

local function getBeaconAnchor(target)
    if not target then return nil, nil end
    local adornee, size = nil, nil
    if target:IsA("BasePart") then
        adornee = target
        size = target.Size
    elseif target:IsA("Model") or target:IsA("Folder") then
        local head = target:FindFirstChild("Head")
        local hrp = target:FindFirstChild("HumanoidRootPart")
        local primary = target:IsA("Model") and target.PrimaryPart or nil
        if head and head:IsA("BasePart") then adornee = head
        elseif hrp and hrp:IsA("BasePart") then adornee = hrp
        elseif primary then adornee = primary
        else adornee = target:FindFirstChildWhichIsA("BasePart") end
        if not adornee then return nil, nil end
        if target:IsA("Model") then
            local ok, cf, sz = pcall(function() return target:GetBoundingBox() end)
            if ok and cf and sz then size = sz else size = adornee.Size end
        else
            size = adornee.Size
        end
    end
    return adornee, size
end

local function destroyBeaconInstance(inst)
    if not inst then return end
    if inst.hl and inst.hl.Parent then inst.hl:Destroy() end
    if inst.billboard and inst.billboard.Parent then inst.billboard:Destroy() end
    if inst.beamModel and inst.beamModel.Parent then inst.beamModel:Destroy() end
end

local function updateBeaconInstance(inst, myPos)
    local target = inst.target
    if not target or not target.Parent then return false end

    local adornee = inst.adornee
    if not adornee or not adornee.Parent then
        local newAdornee, newSize = getBeaconAnchor(target)
        if not newAdornee then return false end
        inst.adornee = newAdornee
        adornee = newAdornee
        if newSize then inst.halfHeight = newSize.Y / 2 end
        if inst.billboard then
            inst.billboard.Parent = adornee
            inst.billboard.StudsOffset = Vector3.new(
                0, inst.halfHeight + (inst.config.beamHeight or 250) + 5, 0)
        end
    end

    local pos = adornee.Position
    local topPos = pos + Vector3.new(0, inst.halfHeight or 2, 0)

    if inst.beamSegs and #inst.beamSegs > 0 and inst.segHeight then
        for i, seg in ipairs(inst.beamSegs) do
            if seg and seg.Parent then
                local cy = topPos.Y + inst.segHeight * (i - 0.5)
                seg.CFrame = CFrame.new(topPos.X, cy, topPos.Z)
                           * CFrame.Angles(0, 0, math.rad(90))
            end
        end
    end

    if inst.distLabel and inst.distLabel.Parent then
        local dist = (pos - myPos).Magnitude
        if dist >= 1000 then
            inst.distLabel.Text = string.format("%.2f km", dist / 1000)
        else
            inst.distLabel.Text = string.format("%.1f m", dist)
        end
    end

    return true
end

-- [Delta修复] 分批遍历
local function findKeywordMatches(kwItem)
    local matches, count = {}, 0
    for _, obj in ipairs(workspace:GetDescendants()) do
        count = count + 1
        if count % 2000 == 0 then task.wait() end
        local n = obj.Name
        if n and n ~= "" then
            local ln = string.lower(n)
            local matched = false
            if kwItem.exact and kwItem.exact ~= "" and string.lower(kwItem.exact) == ln then
                matched = true
            elseif kwItem.prefix and kwItem.prefix ~= "" and string.find(ln, string.lower(kwItem.prefix), 1, true) then
                matched = true
            end
            if matched then
                table.insert(matches, obj)
            end
        end
    end
    return matches
end

local function collectBeaconTargets()
    local result = {}
    for _, kw in ipairs(userProtectKeywords) do
        if kw.beacon and kw.beacon.enabled then
            local matches = findKeywordMatches(kw)
            for _, obj in ipairs(matches) do
                result[obj] = kw.beacon
            end
        end
    end
    for obj, cfg in pairs(protectedBeaconMap) do
        if cfg and cfg.enabled and obj and obj.Parent then
            result[obj] = cfg
        end
    end
    return result
end

local function hasPeriodicScanEnabled()
    for _, kw in ipairs(userProtectKeywords) do
        if kw.beacon and kw.beacon.enabled and kw.beacon.periodicScan then
            return true, kw.beacon.scanInterval or 2.5
        end
    end
    return false, 0
end

local function syncBeacons()
    local desired = collectBeaconTargets()

    for target, inst in pairs(beaconInstances) do
        if not target.Parent or not desired[target] then
            destroyBeaconInstance(inst)
            beaconInstances[target] = nil
        end
    end

    for target, cfg in pairs(desired) do
        if target.Parent then
            local inst = beaconInstances[target]
            if not inst then
                local newInst = createBeaconInstance(target, cfg)
                if newInst then beaconInstances[target] = newInst end
            elseif inst.config ~= cfg then
                destroyBeaconInstance(inst)
                local newInst = createBeaconInstance(target, cfg)
                if newInst then beaconInstances[target] = newInst
                else beaconInstances[target] = nil end
            end
        end
    end
end

local beaconFrameConn = nil
local beaconScanConn = nil
local beaconLastScan = 0

local function startBeaconLoop()
    if beaconFrameConn then beaconFrameConn:Disconnect() end
    beaconFrameConn = RunService.RenderStepped:Connect(function()
        if isShuttingDown then return end
        local char = player.Character
        local myRoot = char and char:FindFirstChild("HumanoidRootPart")
        local myPos = myRoot and myRoot.Position or (camera and camera.CFrame.Position)
        if not myPos then return end
        for target, inst in pairs(beaconInstances) do
            local ok = updateBeaconInstance(inst, myPos)
            if not ok then
                destroyBeaconInstance(inst)
                beaconInstances[target] = nil
            end
        end
    end)

    if beaconScanConn then beaconScanConn:Disconnect() end
    beaconScanConn = RunService.Heartbeat:Connect(function()
        if isShuttingDown then return end
        local has, interval = hasPeriodicScanEnabled()
        if not has then return end
        local t = tick()
        if t - beaconLastScan < interval then return end
        beaconLastScan = t
        syncBeacons()
    end)
end

local function stopBeaconLoop()
    if beaconFrameConn then beaconFrameConn:Disconnect() beaconFrameConn = nil end
    if beaconScanConn then beaconScanConn:Disconnect() beaconScanConn = nil end
    for target, inst in pairs(beaconInstances) do
        destroyBeaconInstance(inst)
    end
    beaconInstances = {}
    beaconLastScan = 0
end

local function reposition()
    local y = startY
    for _, n in ipairs(notifs) do
        local f = n.frame
        TweenService:Create(f, TweenInfo.new(0.2), {Position = UDim2.new(1, -f.Size.X.Offset - 10, 0, y)}):Play()
        y = y + n.height + spacing
    end
end
local function remove(f)
    for i, n in ipairs(notifs) do
        if n.frame == f then table.remove(notifs, i) break end
    end
end
local function processReady()
    while #readyQueue > 0 do
        local n = table.remove(readyQueue, 1)
        if n and n.frame and n.frame.Parent then
            local f, sg = n.frame, n.sg
            local fadeOut = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            local t1 = TweenService:Create(f, fadeOut, {BackgroundTransparency = 1})
            TweenService:Create(n.title, fadeOut, {TextTransparency = 1}):Play()
            TweenService:Create(n.line, fadeOut, {BackgroundTransparency = 1}):Play()
            TweenService:Create(n.msg, fadeOut, {TextTransparency = 1}):Play()
            t1:Play(); t1.Completed:Wait()
            remove(f); sg:Destroy()
        end
    end
    reposition()
end

function createMenu(title, buttons, parentMenu)
    if _activeInputDialog then
        local d = _activeInputDialog
        _activeInputDialog = nil
        pcall(function() d:Destroy() end)
    end
    if _activeMenuDialog then
        local m = _activeMenuDialog
        _activeMenuDialog = nil
        pcall(function() m:Destroy() end)
    end

    local screenSize = getScreenSize()
    local menuWidth = math.min(350, screenSize.X * 0.8)
    local btnHeight, spacing, padding = 40, 5, 10
    local titleHeight, closeBtnHeight = 40, 40
    local contentHeight = #buttons * btnHeight + (#buttons - 1) * spacing
    local totalContentHeight = contentHeight
    local totalHeight = padding + titleHeight + totalContentHeight + padding + closeBtnHeight + padding
    local maxHeight = screenSize.Y * 0.8
    local needsScrolling = totalHeight > maxHeight
    if needsScrolling then
        totalHeight = maxHeight
        local sh = totalHeight - padding - titleHeight - padding - closeBtnHeight - padding
        contentHeight = math.min(contentHeight, sh)
    end
    local dialog = Instance.new("ScreenGui")
    _activeMenuDialog = dialog
    dialog.Parent = playerGui; dialog.IgnoreGuiInset = true; dialog.ResetOnSpawn = false
    dialog.Destroying:Connect(function()
        if _activeMenuDialog == dialog then _activeMenuDialog = nil end
    end)
    local bg = Instance.new("Frame")
    bg.Parent = dialog
    bg.Size = UDim2.new(0, menuWidth, 0, totalHeight)
    bg.Position = UDim2.new(0.5, -menuWidth/2, 0.5, -totalHeight/2)
    bg.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    bg.BackgroundTransparency = 0.2; bg.BorderSizePixel = 0
    bg.Active = true; bg.ClipsDescendants = true
    local corner = Instance.new("UICorner"); corner.Parent = bg; corner.CornerRadius = UDim.new(0, 8)
    local titleLabel = Instance.new("TextLabel")
    titleLabel.Parent = bg
    titleLabel.Size = UDim2.new(1, -20, 0, 30)
    titleLabel.Position = UDim2.new(0, 10, 0, 5)
    titleLabel.BackgroundTransparency = 1; titleLabel.Text = title
    titleLabel.TextColor3 = Color3.new(1, 1, 1); titleLabel.Font = Enum.Font.GothamBold
    titleLabel.TextSize = 18; titleLabel.TextXAlignment = Enum.TextXAlignment.Center
    local sf = Instance.new("ScrollingFrame")
    sf.Parent = bg; sf.Size = UDim2.new(1, -20, 0, contentHeight)
    sf.Position = UDim2.new(0, 10, 0, 40)
    sf.CanvasSize = UDim2.new(0, 0, 0, totalContentHeight)
    sf.ScrollBarThickness = 8; sf.BackgroundTransparency = 1; sf.BorderSizePixel = 0
    sf.ScrollingEnabled = needsScrolling
    sf.ScrollBarImageColor3 = Color3.fromRGB(150, 150, 150)
    local bc = Instance.new("Frame")
    bc.Parent = sf; bc.Size = UDim2.new(1, 0, 0, totalContentHeight)
    bc.BackgroundTransparency = 1
    local yPos = 0
    for _, btn in ipairs(buttons) do
        local button = Instance.new("TextButton")
        button.Parent = bc
        button.Size = UDim2.new(1, -10, 0, btnHeight)
        button.Position = UDim2.new(0, 0, 0, yPos)
        button.BackgroundColor3 = btn.bgColor or Color3.fromRGB(50, 50, 50)
        button.Text = btn.text; button.TextColor3 = Color3.new(1, 1, 1)
        button.Font = Enum.Font.Gotham; button.TextSize = 14
        button.TextWrapped = true; button.AutoButtonColor = true
        local btnC = Instance.new("UICorner"); btnC.Parent = button; btnC.CornerRadius = UDim.new(0, 4)

        if btn.onLongPress then
            local pressStart = 0
            local longTriggered = false
            button.MouseButton1Down:Connect(function()
                pressStart = tick()
                longTriggered = false
            end)
            button.MouseButton1Up:Connect(function()
                if longTriggered then return end
                if tick() - pressStart < 0.4 then
                    if btn.callback then btn.callback(dialog) end
                end
            end)
            task.spawn(function()
                while button.Parent do
                    task.wait(0.05)
                    if pressStart > 0 and not longTriggered and tick() - pressStart >= 0.4 then
                        longTriggered = true
                        pressStart = 0
                        if btn.onLongPress then btn.onLongPress(dialog) end
                    end
                end
            end)
        else
            button.MouseButton1Click:Connect(function()
                if btn.callback then btn.callback(dialog) end
            end)
        end

        yPos = yPos + btnHeight + spacing
    end
    local closeBtn = Instance.new("TextButton")
    closeBtn.Parent = bg
    closeBtn.Size = UDim2.new(1, -20, 0, 35)
    closeBtn.Position = UDim2.new(0, 10, 1, -45)
    closeBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
    closeBtn.Text = parentMenu and "返回上级" or "关闭"
    closeBtn.TextColor3 = Color3.new(1, 1, 1); closeBtn.Font = Enum.Font.Gotham
    closeBtn.TextSize = 14; closeBtn.AutoButtonColor = true
    local cc = Instance.new("UICorner"); cc.Parent = closeBtn; cc.CornerRadius = UDim.new(0, 4)
    closeBtn.MouseButton1Click:Connect(function()
        dialog:Destroy()
        if parentMenu then parentMenu() end
    end)
    return dialog
end

local function colorToHexSafe(c)
    local r = math.floor(math.clamp(c.R, 0, 1) * 255 + 0.5)
    local g = math.floor(math.clamp(c.G, 0, 1) * 255 + 0.5)
    local b = math.floor(math.clamp(c.B, 0, 1) * 255 + 0.5)
    return string.format("#%02X%02X%02X", r, g, b)
end

showKeywordManageMenu = function()
    local buttons = {}

    local protectedCount = 0
    for obj, _ in pairs(protectedList) do

    end
    table.insert(buttons, {
        text = "🎯 单目标保护 (" .. protectedCount .. ")  →",
        bgColor = Color3.fromRGB(180, 100, 60),
        callback = function(menu)
            menu:Destroy()
            if showSingleProtectMenu then showSingleProtectMenu() end
        end
    })

    if #userProtectKeywords == 0 then
        table.insert(buttons, {
            text = "（暂无关键词）",
            bgColor = Color3.fromRGB(60, 60, 60),
            callback = function() end
        })
    else
        table.insert(buttons, {
            text = "📋 复制所有前缀 (" .. #userProtectKeywords .. ")",
            bgColor = Color3.fromRGB(50, 130, 200),
            callback = function(menu)
                local result = {}
                for _, item in ipairs(userProtectKeywords) do
                    if item.prefix and item.prefix ~= "" then table.insert(result, item.prefix) end
                end
                if #result == 0 then tanchuangxiaoxi("没有可复制的前缀", "保护"); return end
                local text = table.concat(result, ",")
                if setclipboard then pcall(setclipboard, text); tanchuangxiaoxi("已复制: " .. text, "保护")
                else tanchuangxiaoxi("剪贴板不可用", "保护") end
                menu:Destroy()
            end
        })
        table.insert(buttons, {
            text = "📋 复制所有完整名 (" .. #userProtectKeywords .. ")",
            bgColor = Color3.fromRGB(70, 140, 180),
            callback = function(menu)
                local result = {}
                for _, item in ipairs(userProtectKeywords) do
                    if item.exact and item.exact ~= "" then table.insert(result, item.exact) end
                end
                if #result == 0 then tanchuangxiaoxi("没有可复制的完整名", "保护"); return end
                local text = table.concat(result, ",")
                if setclipboard then pcall(setclipboard, text); tanchuangxiaoxi("已复制: " .. text, "保护")
                else tanchuangxiaoxi("剪贴板不可用", "保护") end
                menu:Destroy()
            end
        })
        table.insert(buttons, {
            text = "📤 导出到文件 (" .. #userProtectKeywords .. ")",
            bgColor = Color3.fromRGB(200, 130, 0),
            callback = function(menu)
                menu:Destroy()
                if not isfolder("ProtectKeywords") then makefolder("ProtectKeywords") end
                showInputDialog("输入导出文件名", "keywords_" .. os.date("%Y%m%d_%H%M%S"), function(filename)
                    if not filename or filename == "" then return end
                    filename = filename:match("^%s*(.-)%s*$")
                    if filename == "" then return end
                    local filepath = "ProtectKeywords/" .. filename .. ".txt"
                    local function doWrite()
                        local text = serializeKeywords(userProtectKeywords)
                        writefile(filepath, text)
                        tanchuangxiaoxi("已导出: " .. filepath, "保护")
                    end
                    if isfile(filepath) then
                        createMenu("文件已存在", {
                            { text = "🔄 覆盖原文件", callback = function(subMenu) subMenu:Destroy(); doWrite() end },
                            { text = "✏️ 重命名", callback = function(subMenu)

                                    newName = newName:match("^%s*(.-)%s*$")
                                    if newName == "" then return end
                                    local newPath = "ProtectKeywords/" .. newName .. ".txt"

                                    else writefile(newPath, serializeKeywords(userProtectKeywords)); tanchuangxiaoxi("已导出: " .. newPath, "保护") end
                                end)
                            end },
                            { text = "❌ 取消", callback = function(subMenu) subMenu:Destroy() end }
                        }, nil)
                    else
                        doWrite()
                    end
                end)
            end
        })
        for i, item in ipairs(userProtectKeywords) do
            local idxCopy = i
            local label = item.prefix or item.exact or "?"
            local beaconMark = (item.beacon and item.beacon.enabled) and "💡 " or ""

                text = beaconMark .. "🔍 " .. label .. "（短按查看，长按信标）",

                        function() return targetItem.beacon or makeBeaconConfig({}) end,
                        function(cfg) targetItem.beacon = cfg end,
                        "关键词: " .. (targetItem.prefix or targetItem.exact or "?"),
                        showKeywordManageMenu
                    )
                end
            })
        end
    end
    createMenu("管理关键词 (" .. #userProtectKeywords .. ")", buttons, showProtectMenu)

showSingleProtectMenu = function()
    local buttons = {}

    local protectedCount = 0
    for obj, _ in pairs(protectedList) do
        if obj and obj.Parent then protectedCount = protectedCount + 1 end
    end

    table.insert(buttons, {
        text = "➕ 添加目标（按名字，逗号分隔批量）",
        bgColor = Color3.fromRGB(50, 130, 200),
        callback = function(menu)
            menu:Destroy()
            showInputDialog("输入对象名字（可用逗号分隔批量）", "", function(val)
                if not val or val == "" then return end
                local names = {}
                for n in string.gmatch(val, "([^,]+)") do
                    local trimmed = n:match("^%s*(.-)%s*$")
                    if trimmed ~= "" then table.insert(names, trimmed) end
                end
                if #names == 0 then
                    tanchuangxiaoxi("未检测到有效名字", "保护"); return
                end
                local added, scanned = 0, 0
                local nameSet = {}
                for _, n in ipairs(names) do nameSet[n] = true end
                for _, obj in ipairs(workspace:GetDescendants()) do
                    scanned = scanned + 1
                    if nameSet[obj.Name] and not protectedList[obj] then
                        protectedList[obj] = true
                        added = added + 1
                    end
                end
                tanchuangxiaoxi("已添加 " .. added .. " 个对象 (扫描 " .. scanned .. ")", "保护")
                if showSingleProtectMenu then showSingleProtectMenu() end
            end)
        end
    })

    table.insert(buttons, {
        text = "🎯 从准心拾取目标",

            menu:Destroy()

        end

        text = "📥 从文件导入单目标",
        bgColor = Color3.fromRGB(46, 139, 87),
        callback = function(menu)
            menu:Destroy()

                tanchuangxiaoxi("文件夹 ProtectKeywords 不存在，请先导出一次", "保护"); return
            end
            local files = listfiles("ProtectKeywords")
            local btnList = {}

                            subMenu:Destroy()
                            local content = readfile(file)
                            local names = {}
                            for line in string.gmatch(content, "[^\n]+") do
                                local firstComma = string.find(line, ",")
                                if firstComma then
                                    local exact = string.sub(line, firstComma + 1)
                                    if exact ~= "" then table.insert(names, exact)
                                    else
                                        local prefix = string.sub(line, 1, firstComma - 1)
                                        if prefix ~= "" then table.insert(names, prefix) end
                                    end
                                else
                                    if line ~= "" then table.insert(names, line) end
                                end
                            end

                            for _, n in ipairs(names) do nameSet[n] = true end
                            local added = 0
                            for _, obj in ipairs(workspace:GetDescendants()) do
                                if nameSet[obj.Name] and not protectedList[obj] then
                                    protectedList[obj] = true
                                    added = added + 1
                                end
                            end

                            if showSingleProtectMenu then showSingleProtectMenu() end
                        end
                    })
                end

-- ==================== 管理单目标 ====================
showSingleManageMenu = function()
    local buttons = {}

    local validTargets = {}
    for obj, _ in pairs(protectedList) do
        if obj and obj.Parent and obj.Name and obj.Name ~= "" then
            table.insert(validTargets, obj)
        else
            protectedList[obj] = nil
            protectedBeaconMap[obj] = nil
        end
    end
    local total = #validTargets

    if total == 0 then
        table.insert(buttons, {
            text = "（暂无单目标）",
            bgColor = Color3.fromRGB(60, 60, 60),
            callback = function() end
        })
        createMenu("📋 管理单目标 (0)", buttons, showSingleProtectMenu)
        return
    end

    table.insert(buttons, {
        text = "📋 复制所有名字 (" .. total .. ")",
        bgColor = Color3.fromRGB(50, 130, 200),
        callback = function(menu)
            menu:Destroy()
            local names = {}
            for _, obj in ipairs(validTargets) do
                if obj.Name and obj.Name ~= "" then table.insert(names, obj.Name) end
            end
            if #names == 0 then tanchuangxiaoxi("没有有效的名字可复制", "保护"); return end
            local text = table.concat(names, ",")
            if setclipboard then pcall(setclipboard, text); tanchuangxiaoxi("已复制 " .. #names .. " 个名字", "保护")
            else tanchuangxiaoxi("剪贴板不可用", "保护") end
        end
    })

    table.insert(buttons, {
        text = "📋 复制所有路径 (" .. total .. ")",
        bgColor = Color3.fromRGB(70, 140, 180),
        callback = function(menu)
            menu:Destroy()
            local paths = {}
            for _, obj in ipairs(validTargets) do
                local p = obj:GetFullName()
                if p and p ~= "" then table.insert(paths, p) end
            end
            if #paths == 0 then tanchuangxiaoxi("没有有效的路径可复制", "保护"); return end
            local text = table.concat(paths, ",")
            if setclipboard then pcall(setclipboard, text); tanchuangxiaoxi("已复制 " .. #paths .. " 条路径", "保护")
            else tanchuangxiaoxi("剪贴板不可用", "保护") end
        end
    })

    table.insert(buttons, {
        text = "📤 导出到文件 (" .. total .. ")",
        bgColor = Color3.fromRGB(200, 130, 0),
        callback = function(menu)
            menu:Destroy()
            if not isfolder("ProtectKeywords") then makefolder("ProtectKeywords") end
            local lines = {}
            for _, obj in ipairs(validTargets) do
                if obj.Name and obj.Name ~= "" then
                    local cfg = protectedBeaconMap[obj]
                    if cfg and cfg.enabled then
                        local c = cfg.color
                        table.insert(lines, table.concat({
                            obj.Name, "",
                            "1",
                            "1",
                            cfg.showHighlight and "1" or "0",
                            cfg.showBeam and "1" or "0",
                            cfg.showName and "1" or "0",
                            (cfg.customName or ""):gsub("|", "/"),
                            tostring(math.floor(c.R * 255 + 0.5)),
                            tostring(math.floor(c.G * 255 + 0.5)),
                            tostring(math.floor(c.B * 255 + 0.5)),
                            string.format("%.3f", cfg.beamWidth or 0.5),
                            string.format("%.3f", cfg.beamHeight or 250),
                            string.format("%.3f", cfg.hlTransparency or 0.9),
                            cfg.periodicScan and "1" or "0",
                            string.format("%.3f", cfg.scanInterval or 2.5),
                        }, "|"))
                    else
                        table.insert(lines, obj.Name .. ",")
                    end
                end
            end
            if #lines == 0 then tanchuangxiaoxi("没有有效的对象名字可导出", "保护"); return end
            local exportText = table.concat(lines, "\n")
            showInputDialog("输入导出文件名", "singles_" .. os.date("%Y%m%d_%H%M%S"), function(filename)
                if not filename or filename == "" then return end
                filename = filename:match("^%s*(.-)%s*$")
                if filename == "" then return end
                local filepath = "ProtectKeywords/" .. filename .. ".txt"
                local function doWrite()
                    writefile(filepath, exportText)
                    tanchuangxiaoxi("已导出 " .. #lines .. " 条到 " .. filepath, "保护")
                end
                if isfile(filepath) then
                    createMenu("文件已存在", {
                        { text = "🔄 覆盖原文件", callback = function(sm) sm:Destroy(); doWrite() end },
                        { text = "✏️ 重命名", callback = function(sm)
                            sm:Destroy()
                            showInputDialog("输入新文件名", filename .. "_1", function(newName)
                                if not newName or newName == "" then return end
                                newName = newName:match("^%s*(.-)%s*$")
                                if newName == "" then return end
                                local np = "ProtectKeywords/" .. newName .. ".txt"
                                if isfile(np) then tanchuangxiaoxi("该文件名依然存在！", "保护")
                                else writefile(np, exportText); tanchuangxiaoxi("已导出: " .. np, "保护") end
                            end)
                        end },
                        { text = "❌ 取消", callback = function(sm) sm:Destroy() end }
                    }, nil)
                else
                    doWrite()
                end
            end)
        end
    })

    local displayLimit = 100
    local index = 0
    for _, obj in ipairs(validTargets) do
        if index >= displayLimit then break end
        index = index + 1
        local objCopy = obj
        local displayName = obj.Name
        if #displayName > 30 then displayName = string.sub(displayName, 1, 27) .. "..." end
        local cfg = protectedBeaconMap[obj]
        local beaconMark = (cfg and cfg.enabled) and "💡 " or ""
        table.insert(buttons, {
            text = beaconMark .. "🎯 " .. displayName .. "（短按查看，长按信标）",
            bgColor = (cfg and cfg.enabled) and Color3.fromRGB(120, 140, 90) or Color3.fromRGB(60, 100, 140),
            callback = function(menu)
                menu:Destroy()
                if showSingleDetailDialog then showSingleDetailDialog(objCopy) end
            end,
            onLongPress = function(menu)
                menu:Destroy()
                if not objCopy or not objCopy.Parent then
                    tanchuangxiaoxi("对象已不存在", "保护"); return
                end
                showBeaconConfigDialog(
                    function() return protectedBeaconMap[objCopy] or makeBeaconConfig({}) end,
                    function(cfg2) protectedBeaconMap[objCopy] = cfg2 end,
                    "单目标: " .. objCopy.Name,
                    showSingleManageMenu
                )
            end
        })
    end

    if total > displayLimit then
        table.insert(buttons, {
            text = "（还有 " .. (total - displayLimit) .. " 个未显示）",
            bgColor = Color3.fromRGB(60, 60, 60),
            callback = function() end
        })
    end

    createMenu("📋 管理单目标 (" .. total .. ")", buttons, showSingleProtectMenu)
end

-- ==================== 自由相机菜单 ====================
showCameraMenu = function(parentMenu)
    local function buildCamMenu()
        local freezeText = camFreezeEnabled and "🧊 冻结解锁: 开启中" or "🧊 冻结解锁: 关闭"
        local noClipText = camNoClipEnabled and "👻 相机穿墙: 开启" or "👻 相机穿墙: 关闭"
        local snapText = camSnapshot and ("📸 重新拍摄 (上次 " .. os.date("%H:%M:%S", camSnapshot.takenAt) .. ")") or "📸 重新拍摄快照"
        local restoreText = camSnapshot and "↩️ 恢复到快照" or "↩️ 恢复到快照 (无)"
        createMenu("🎥 自由相机", {
            { text = freezeText, callback = function(sm)
                if camFreezeEnabled then stopCamFreeze() else startCamFreeze(true) end
                sm:Destroy(); buildCamMenu()
            end },
            { text = "🔓 单次解锁", callback = function(sm)
                doCamSingleUnlock(true); sm:Destroy(); buildCamMenu()
            end },
            { text = noClipText, callback = function(sm)
                if camNoClipEnabled then disableCamNoClip() else enableCamNoClip() end
                sm:Destroy(); buildCamMenu()
            end },
            { text = snapText, callback = function(sm)
                manualResnap(); sm:Destroy(); buildCamMenu()
            end },
            { text = restoreText, callback = function(sm)
                restoreFromCamSnapshot(); sm:Destroy(); buildCamMenu()
            end },
        }, parentMenu)
    end
    buildCamMenu()
end

            local ny = tonumber(boxY.Text) or 0
            local nz = tonumber(boxZ.Text) or 0
            if not previewBackup then
                previewBackup = { x = camOffsetX, y = camOffsetY, z = camOffsetZ }
            end
            previewToken = previewToken + 1
            local myToken = previewToken
            camOffsetX = nx; camOffsetY = ny; camOffsetZ = nz
            applyCameraOffset()
            tanchuangxiaoxi(string.format("预览 %.1fs 后自动还原", s), "相机偏移")
            task.delay(s, function()
                if previewToken ~= myToken then return end
                if previewBackup then
                    camOffsetX = previewBackup.x
                    camOffsetY = previewBackup.y
                    camOffsetZ = previewBackup.z
                    applyCameraOffset()
                    previewBackup = nil
                end
                if dialog.Parent then
                    boxX.Text = string.format("%.2f", camOffsetX)
                    boxY.Text = string.format("%.2f", camOffsetY)
                    boxZ.Text = string.format("%.2f", camOffsetZ)
                end
                tanchuangxiaoxi("预览结束，已还原", "相机偏移")
            end)
        end)
    end)

    return dialog
end

-- ==================== 移动向量 ====================
local function getMoveVector(dir, rootPart)
    local step = dir * moveStep
    if moveMode == "角色上下" then return rootPart.CFrame.UpVector * step
    elseif moveMode == "角色前后" then return rootPart.CFrame.LookVector * step
    elseif moveMode == "角色左右" then return -rootPart.CFrame.RightVector * step
    elseif moveMode == "屏幕上下" then
        local c = workspace.CurrentCamera
        if c then return c.CFrame.UpVector * step end
    elseif moveMode == "屏幕前后" then
        local c = workspace.CurrentCamera
        if c then return c.CFrame.LookVector * step end
    elseif moveMode == "屏幕左右" then
        local c = workspace.CurrentCamera
        if c then return -c.CFrame.RightVector * step end
    elseif moveMode == "水平上下" then return Vector3.new(0, step, 0)
    elseif moveMode == "水平前后(屏幕)" then
        local c = workspace.CurrentCamera
        if c then
            local l = c.CFrame.LookVector
            local h = Vector3.new(l.X, 0, l.Z)
            if h.Magnitude > 0 then return h.Unit * step end
        end
    elseif moveMode == "水平左右(屏幕)" then
        local c = workspace.CurrentCamera
        if c then
            local r = c.CFrame.RightVector
            local h = Vector3.new(r.X, 0, r.Z)
            if h.Magnitude > 0 then return h.Unit * step end
        end
    end
    return Vector3.new()
end

        local miniPressStartPos = nil

            return input.UserInputType == Enum.UserInputType.Touch
                or input.UserInputType == Enum.UserInputType.MouseButton1
        end
        local function beginMiniDrag(input) miniIsDragging = true; miniDragInput = input end
        UserInputService.InputBegan:Connect(function(input, _gpe)
            if not isMiniStartInput(input) then return end
            if not isPointInGui(miniBtn, input.Position.X, input.Position.Y) then return end
            miniHandleTouches[input] = true
            if miniIsDragging then return end
            beginMiniDrag(input)
            miniTotalDelta = 0
            miniPressStartPos = Vector2.new(input.Position.X, input.Position.Y)
        end)
        UserInputService.InputChanged:Connect(function(input)
            if not miniIsDragging then return end
            local matched = (input == miniDragInput)
                or (miniDragInput ~= nil
                    and miniDragInput.UserInputType == Enum.UserInputType.MouseButton1
                    and input.UserInputType == Enum.UserInputType.MouseMovement)
            if not matched then return end
            if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseMovement then return end
            local d = Vector2.new(input.Delta.X, input.Delta.Y)
            miniTotalDelta = miniTotalDelta + d.Magnitude
            if d.Magnitude == 0 then return end
            local cam = workspace.CurrentCamera
            if not cam then return end
            local vp3 = cam.ViewportSize
            local mw = mini.Size.X.Offset; local mh = mini.Size.Y.Offset
            miniX_var = clamp(miniX_var + d.X, 0, vp3.X - mw)
            miniY_var = clamp(miniY_var + d.Y, 0, vp3.Y - mh)
            mini.Position = UDim2.fromOffset(miniX_var, miniY_var)
        end)
        UserInputService.InputEnded:Connect(function(input)
            if miniHandleTouches[input] then miniHandleTouches[input] = nil end
            if not miniIsDragging or input ~= miniDragInput then return end
            if not isMiniStartInput(input) then return end
            local nextInput = nil
            for t in pairs(miniHandleTouches) do nextInput = t; break end
            if nextInput then beginMiniDrag(nextInput); return end
            local endPos = Vector2.new(input.Position.X, input.Position.Y)
            local straightDist = miniPressStartPos and (endPos - miniPressStartPos).Magnitude or 0
            if straightDist < 10 and miniTotalDelta < 15 then
                local vp2 = getScreenSize()
                local fw = Frame.Size.X.Offset; local fh = Frame.Size.Y.Offset

                Frame.Visible = true
                mini:Destroy(); miniWindow = nil
            end
            miniIsDragging = false; miniDragInput = nil; miniPressStartPos = nil
        end)
        miniWindow = mini
    end

showProtectMenu = function()
    local buttons = {}
    table.insert(buttons, {
        text = "🔑 管理保护关键词 (" .. #userProtectKeywords .. ")",
        bgColor = Color3.fromRGB(80, 140, 200),
        callback = function(menu)
            menu:Destroy()
            if showKeywordManageMenu then showKeywordManageMenu() end
        end
    })
    table.insert(buttons, {
        text = "🛡 默认保护分类",
        bgColor = Color3.fromRGB(100, 180, 120),
        callback = function(menu)
            menu:Destroy()
            if showDefaultProtectMenu then showDefaultProtectMenu() end
        end
    })
    table.insert(buttons, {
        text = "➕ 添加保护关键词",
        bgColor = Color3.fromRGB(60, 160, 200),
        callback = function(menu)
            menu:Destroy()
            showInputDialog("输入关键词（多个用逗号分隔）", "", function(input)
                if not input or input == "" then return end
                local added = 0
                for kw in string.gmatch(input, "([^,]+)") do
                    local trimmed = kw:match("^%s*(.-)%s*$")
                    if trimmed ~= "" then
                        local exists = false
                        for _, item in ipairs(userProtectKeywords) do
                            if (item.prefix == trimmed) or (item.exact == trimmed) then
                                exists = true; break
                            end
                        end
                        if not exists then
                            table.insert(userProtectKeywords, { prefix = trimmed, exact = nil })
                            added = added + 1
                        end
                    end
                end
                tanchuangxiaoxi("已添加 " .. added .. " 个关键词", "保护")
                if showProtectMenu then showProtectMenu() end
            end)
        end
    })
    table.insert(buttons, {
        text = "📥 从文件导入关键词",
        bgColor = Color3.fromRGB(120, 100, 180),
        callback = function(menu)
            menu:Destroy()
            if not isfolder("ProtectKeywords") then
                tanchuangxiaoxi("文件夹 ProtectKeywords 不存在", "保护"); return
            end
            local files = listfiles("ProtectKeywords")
            local btnList = {}
            for _, file in ipairs(files) do
                if string.sub(file, -4) == ".txt" then
                    local fileName = string.match(file, "([^/\\]+)%.txt$") or file
                    table.insert(btnList, {
                        text = "📄 " .. fileName,
                        bgColor = Color3.fromRGB(70, 130, 180),
                        callback = function(subMenu)
                            subMenu:Destroy()
                            local content = readfile(file)
                            local parsed = deserializeKeywords(content)
                            local added = 0
                            for _, item in ipairs(parsed) do
                                table.insert(userProtectKeywords, item)
                                added = added + 1
                            end
                            tanchuangxiaoxi("已从 [" .. fileName .. "] 导入 " .. added .. " 条", "保护")
                            afterBeaconChange()
                            if showProtectMenu then showProtectMenu() end
                        end
                    })
                end
            end
            if #btnList == 0 then tanchuangxiaoxi("暂无 .txt 文件", "保护")
            else createMenu("选择要导入的文件", btnList, showProtectMenu) end
        end
    })
    table.insert(buttons, {
        text = "🗑 清空所有关键词 (" .. #userProtectKeywords .. ")",
        bgColor = Color3.fromRGB(180, 50, 50),
        callback = function(menu)
            menu:Destroy()
            local n = #userProtectKeywords
            userProtectKeywords = {}
            tanchuangxiaoxi("已清空 " .. n .. " 条关键词", "保护")
            afterBeaconChange()
            if showProtectMenu then showProtectMenu() end
        end
    })
    createMenu("🛡 保护设置", buttons, nil)
end

-- ==================== 音量键 ====================
function setUIVisible(visible)
    uiVisible = visible
    Frame.Visible = uiVisible
    if miniWindow then miniWindow.Visible = uiVisible end
    tanchuangxiaoxi(uiVisible and "UI已显示" or "UI已隐藏", "音量键")
end

function enableVolumeKey()
    local hasVD = pcall(function() return Enum.KeyCode.VolumeDown end)
    local hasVU = pcall(function() return Enum.KeyCode.VolumeUp end)
    if not hasVD or not hasVU then
        tanchuangxiaoxi("您的设备不支持音量键", "提示"); return
    end
    if volumeKeyEnabled then return end
    volumeKeyConnection = UserInputService.InputBegan:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.VolumeDown then setUIVisible(false)
        elseif input.KeyCode == Enum.KeyCode.VolumeUp then setUIVisible(true) end
    end)
    volumeKeyEnabled = true
    tanchuangxiaoxi("音量键已启用", "设置")
end

function disableVolumeKey()
    if not volumeKeyEnabled then return end
    if volumeKeyConnection then volumeKeyConnection:Disconnect() volumeKeyConnection = nil end
    volumeKeyEnabled = false
    tanchuangxiaoxi("音量键已禁用", "设置")
end

-- ==================== 彻底关闭 ====================
function fullShutdown()
    isShuttingDown = true
    if isFlying then pcall(toggleFly, false) end
    if speedModeEnabled then pcall(applySpeedMode, false) end
    if noclipEnabled then pcall(disableNoclip) end
    if nightVisionEnabled then pcall(disableNightVision) end
    if SmartBuilding.isActive() then pcall(SmartBuilding.closeUI) end

    pcall(stopBeaconLoop)

    pcall(stopCamFreeze)
    pcall(disableCamNoClip)
    if camOffsetConn then camOffsetConn:Disconnect() camOffsetConn = nil end
    camOffsetX, camOffsetY, camOffsetZ = 0, 0, 0
    local char = player.Character
    if char then
        local hum = char:FindFirstChildWhichIsA("Humanoid")
        if hum then pcall(function() hum.CameraOffset = Vector3.zero end) end
    end

    if speedModeConnection then speedModeConnection:Disconnect() speedModeConnection = nil end
    if speedUpdateConnection then speedUpdateConnection:Disconnect() speedUpdateConnection = nil end
    if accelConn then accelConn:Disconnect() accelConn = nil end
    if _G._flyData then removeFly() end
    stopPhysAccel()
    tpwalking = false

    if volumeKeyConnection then volumeKeyConnection:Disconnect() volumeKeyConnection = nil end
    volumeKeyEnabled = false

    if _activeInputDialog then
        pcall(function() _activeInputDialog:Destroy() end)
        _activeInputDialog = nil
    end
    if _activeMenuDialog then
        pcall(function() _activeMenuDialog:Destroy() end)
        _activeMenuDialog = nil
    end
    isShuttingDown = false
end

main.Destroying:Connect(function()
    pcall(fullShutdown)
    if miniWindow then miniWindow:Destroy() miniWindow = nil end
    if noclipWindow then noclipWindow:Destroy() noclipWindow = nil end
    if nightVisionWindow then nightVisionWindow:Destroy() nightVisionWindow = nil end
end)
