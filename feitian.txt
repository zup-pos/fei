-- ==================== 主窗口拖拽逻辑 ====================
local isDragging = false
local dragTouch = nil
local lastTouchPos = nil
local latestTouchPos = nil

Frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch then
        isDragging = true
        dragTouch = input
        lastTouchPos = Vector2.new(input.Position.X, input.Position.Y)
        latestTouchPos = lastTouchPos
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isDragging and input == dragTouch and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
        latestTouchPos = Vector2.new(input.Position.X, input.Position.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input == dragTouch then
        isDragging = false
        dragTouch = nil
        lastTouchPos = nil
        latestTouchPos = nil
    end
end)

RunService.RenderStepped:Connect(function()
    if not isDragging or not lastTouchPos or not latestTouchPos then return end
    local delta = latestTouchPos - lastTouchPos
    lastTouchPos = latestTouchPos

    local newX = frameX + delta.X
    local newY = frameY + delta.Y

    local camera = workspace.CurrentCamera
    if not camera then return end
    local vp = camera.ViewportSize
    local frameW = Frame.Size.X.Offset
    local frameH = Frame.Size.Y.Offset

    newX = clamp(newX, 0, vp.X - frameW)
    newY = clamp(newY, 0, vp.Y - frameH)

    setFramePos(newX, newY)
end)

local function updateMainButtonText()
    local modeName = modeDisplayNames[modeIndex + 1]
    local state = false
    if modeIndex == 0 then state = isFlying
    elseif modeIndex == 1 then state = speedModeEnabled
    elseif modeIndex == 2 then state = noclipEnabled
    elseif modeIndex == 3 then state = buildingRemovalEnabled
    elseif modeIndex == 4 then state = nightVisionEnabled

    onof.Text = modeName .. (state and "(开启)" or "(关闭)")

        plus.Text = "增加"

local function updateSpeedButtonText()
    if modeIndex == 0 then
        speed.Text = tostring(speeds)
    elseif modeIndex == 1 then

        if speedModeEnabled then
            if speedSubMode == "加速" then
                speed.Text = string.format("加%.0f", lockedSpeed)

                speed.Text = string.format("%.1f", lockedSpeed)
            end

            local char = player.Character
            if char then
                local hum = char:FindFirstChildWhichIsA("Humanoid")
                if hum then speed.Text = string.format("%.1f", hum.WalkSpeed)
                else speed.Text = "0.0" end
            else speed.Text = "0.0" end

        if buildingRemovalMode == 1 then speed.Text = "墙体"
        elseif buildingRemovalMode == 2 then speed.Text = "墙+图"
        elseif buildingRemovalMode == 3 then speed.Text = "仅图" end
    elseif modeIndex == 4 then
        speed.Text = string.format("%.2f", nightVisionBrightness)
    end
end

-- ==================== 建筑移除核心功能 ====================
local function isWallForRemoval(part)
    if not part:IsA("BasePart") then return false end
    if part:IsA("Terrain") then return false end
    local nameLower = string.lower(part.Name)
    for _, keyword in ipairs(IGNORED_NAMES) do
        if string.find(nameLower, keyword) then return false end
    end
    if IGNORE_PLAYER and player and player.Character then
        if part:IsDescendantOf(player.Character) then return false end
    end
    if part:IsA("Seat") or part:IsA("VehicleSeat") then return false end
    return true
end

local function enableBuildingRemoval()
    if buildingRemovalEnabled then return end
    buildingRemovalEnabled = true
    local count = 0
    local allParts = workspace:GetDescendants()
    for _, obj in ipairs(allParts) do
        if isWallForRemoval(obj) then
            if not modifiedBuildingParts[obj] then
                modifiedBuildingParts[obj] = {
                    CanCollide = obj.CanCollide,
                    Transparency = obj.Transparency
                }
            end
            if buildingRemovalMode == 1 then obj.CanCollide = false
            elseif buildingRemovalMode == 2 then obj.CanCollide = false obj.Transparency = buildingTransparency
            elseif buildingRemovalMode == 3 then obj.Transparency = buildingTransparency end
            count = count + 1
        end
    end
    tanchuangxiaoxi("已处理 " .. count .. " 个建筑部件", "建筑移除")
    updateMainButtonText()
    updateSpeedButtonText()
end

local function disableBuildingRemoval()
    if not buildingRemovalEnabled then return end
    buildingRemovalEnabled = false
    local count = 0
    for part, data in pairs(modifiedBuildingParts) do
        pcall(function()
            if part and part.Parent then
                part.CanCollide = data.CanCollide
                part.Transparency = data.Transparency
                count = count + 1
            end
        end)
    end
    modifiedBuildingParts = {}
    tanchuangxiaoxi("已恢复 " .. count .. " 个建筑部件", "建筑移除")
    updateMainButtonText()
    updateSpeedButtonText()
end

-- ==================== 飞天辅助函数 ====================
local function removeFly()
    if _G._flyData then
        pcall(function() _G._flyData.bg:Destroy() end)
        pcall(function() _G._flyData.bv:Destroy() end)
        _G._flyData = nil
    end
end

local function applyFly()
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hum then return false end

    char.Animate.Disabled = true
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        track:AdjustSpeed(0)
    end
    for _, state in ipairs(VALID_HUMANOD_STATES) do
        pcall(function() hum:SetStateEnabled(state, false) end)
    end

    local isFreeMode = (flyMode == "自由")
    if isFreeMode then
        hum.PlatformStand = true
        hum.AutoRotate = false
    else
        pcall(function() hum:ChangeState(Enum.HumanoidStateType.Swimming); hum.PlatformStand = true end)
    end

    local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("HumanoidRootPart")
    if not torso then return false end
    local startY = torso.Position.Y

    local bg = Instance.new("BodyGyro")
    bg.P = 9e4
    bg.maxTorque = Vector3.new(9e9, 9e9, 9e9)
    bg.CFrame = torso.CFrame
    bg.Parent = torso

    local bv = Instance.new("BodyVelocity")
    bv.Velocity = Vector3.new(0, 0.1, 0)
    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bv.Parent = torso

    _G._flyData = { bg = bg, bv = bv, torso = torso, startY = startY }

    task.spawn(function()
        while isFlying and player.Character and hum and hum.Parent and hum.Health > 0 do
            RunService.Heartbeat:Wait()
            local camera = workspace.CurrentCamera
            if camera then
                local moveDir = hum.MoveDirection
                local maxspeed = 50 * speeds

                if isFreeMode then
                    local camLook = camera.CFrame.LookVector
                    local camRight = camera.CFrame.RightVector
                    local flatLook = Vector3.new(camLook.X, 0, camLook.Z)
                    local flatRight = Vector3.new(camRight.X, 0, camRight.Z)
                    if flatLook.Magnitude > 0.001 then flatLook = flatLook.Unit end
                    if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit end
                    local forwardInput = moveDir:Dot(flatLook)
                    local rightInput = moveDir:Dot(flatRight)
                    local combined = camLook * forwardInput + camRight * rightInput
                    if combined.Magnitude > 1 then combined = combined.Unit end
                    bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                    bv.Velocity = combined * maxspeed
                    bg.CFrame = CFrame.new(torso.Position) * camera.CFrame.Rotation
                else
                    if flyMode == "悬空" then
                        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                        local horizontalDir = Vector3.new(moveDir.X, 0, moveDir.Z)
                        bv.Velocity = (horizontalDir.Magnitude > 0) and (horizontalDir.Unit * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = camera.CFrame
                    elseif flyMode == "绝对锁高" then
                        bv.MaxForce = Vector3.new(9e9, 0, 9e9)
                        local horizontalDir = Vector3.new(moveDir.X, 0, moveDir.Z)
                        bv.Velocity = (horizontalDir.Magnitude > 0) and (horizontalDir.Unit * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = camera.CFrame
                        local pos = torso.Position
                        torso.CFrame = CFrame.new(pos.X, startY, pos.Z) * (torso.CFrame - torso.Position)
                    else
                        bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                        bv.Velocity = (moveDir.Magnitude > 0) and (moveDir * maxspeed) or Vector3.new(0,0,0)
                        bg.CFrame = camera.CFrame
                    end
                end
            end
        end
        removeFly()
    end)
    return true
end

local function resetHumanoidAfterFly()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hum then return end

    for _, state in ipairs(VALID_HUMANOD_STATES) do
        pcall(function() hum:SetStateEnabled(state, true) end)
    end
    hum.PlatformStand = false
    hum.AutoRotate = true
    hum:ChangeState(Enum.HumanoidStateType.Freefall)
    hum:ChangeState(Enum.HumanoidStateType.Running)
    if char:FindFirstChild("Animate") then char.Animate.Disabled = false end
end

-- ==================== 【新增】物理加速核心函数 ====================
local function removePhysAcceleration()
    if accelConnection then
        accelConnection:Disconnect()
        accelConnection = nil
    end
    if accelLinearVelocity then
        pcall(function() accelLinearVelocity:Destroy() end)
        accelLinearVelocity = nil
    end
    if accelAttachment then
        pcall(function() accelAttachment:Destroy() end)
        accelAttachment = nil
    end
    accelEnabled = false
end

local function startPhysAcceleration(targetSpeed)
    local char = player.Character
    if not char then return false end

    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    if not hrp or not hum then return false end

    removePhysAcceleration()

    local attach = Instance.new("Attachment")
    attach.Name = "AccelAttachment"
    attach.Parent = hrp
    accelAttachment = attach

    -- Plane 模式：只约束水平面，Y 轴交给重力
    local lv = Instance.new("LinearVelocity")
    lv.Name = "AccelLinearVelocity"
    lv.Attachment0 = attach
    lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Plane
    lv.PrimaryAxis = Vector3.new(0, 1, 0)
    lv.PlaneVelocity = Vector2.new(0, 0)
    lv.MaxForce = accelMaxForce
    lv.RelativeTo = Enum.ActuatorRelativeTo.World
    lv.Parent = hrp

    accelLinearVelocity = lv
    accelEnabled = true

    accelConnection = RunService.Heartbeat:Connect(function()
        if not accelEnabled then return end
        if not lv or not lv.Parent then
            if accelConnection then accelConnection:Disconnect() accelConnection = nil end
            accelEnabled = false
            return
        end
        local c = player.Character
        if not c then return end
        local h = c:FindFirstChildWhichIsA("Humanoid")
        if not h then return end

        local moveDir = h.MoveDirection
        -- 每帧读取最新的 lockedSpeed，用户调整后立即生效
        local spd = lockedSpeed

        if moveDir.Magnitude > 0.1 then
            local dirXZ = Vector3.new(moveDir.X, 0, moveDir.Z)
            if dirXZ.Magnitude > 0.001 then
                dirXZ = dirXZ.Unit
                lv.PlaneVelocity = Vector2.new(dirXZ.X * spd, dirXZ.Z * spd)
            end
        else
            lv.PlaneVelocity = Vector2.new(0, 0)
        end
    end)

    return true
end

-- ==================== 关闭辅助函数 ====================
local function forceDisableSpeedMode()
    if speedModeEnabled then
        -- 若是加速子模式，先清理物理
        if speedSubMode == "加速" and accelEnabled then
            removePhysAcceleration()
        end

        if speedModeConnection then speedModeConnection:Disconnect() speedModeConnection = nil end

        local char = player.Character
        if char then
            local hum = char:FindFirstChildWhichIsA("Humanoid")
            if hum then pcall(function() hum.WalkSpeed = originalSpeed end) end
        end

        -- 【加速记忆】关闭时快照
        if lockedSpeed and lockedSpeed > 0 then
            if speedSubMode == "加速" then
                savedAccelSpeed = lockedSpeed
            else
                savedSpeed = lockedSpeed
            end
        end

        lockedSpeed = originalSpeed
        speedModeEnabled = false
        task.wait()
        return true
    end
    return false
end

local function forceDisableFly()
    if isFlying then
        removeFly()
        resetHumanoidAfterFly()
        stopTpwalking()
        isFlying = false
        task.wait()
        return true
    end
    return false
end

    if enable then
        local speedWasClosed = forceDisableSpeedMode()

        isFlying = true
        updateMainButtonText()
        stopTpwalking()
        if speedWasClosed then tanchuangxiaoxi("飞天已开启，移速已关闭", "飞天模式")
        else tanchuangxiaoxi("已开启飞天 (" .. flyMode .. ")", "飞天") end
        applyFly()
    else
        if not isFlying then return end
        isFlying = false
        updateMainButtonText()
        stopTpwalking()
        tanchuangxiaoxi("已关闭飞天", "飞天")
        removeFly()
        resetHumanoidAfterFly()
    end
    updateSpeedButtonText()
    updateSpeedDisplayLoop()
end

-- ==================== 移速模式（含加速子模式） ====================
local function applySpeedMode(enable)
    if enable then
        local flyWasClosed = forceDisableFly()
        local char = player.Character
        if char then
            local hum = char:FindFirstChildWhichIsA("Humanoid")
            if hum then originalSpeed = hum.WalkSpeed else originalSpeed = 16 end
        else originalSpeed = 16 end
        if originalSpeed <= 0 then originalSpeed = 16 end

        -- 【记忆机制】根据子模式读取对应的记忆值
        if speedSubMode == "加速" then
            if savedAccelSpeed and savedAccelSpeed > 0 then
                lockedSpeed = savedAccelSpeed
            else
                lockedSpeed = accelSpeed  -- 首次使用默认值
            end
        else
            if savedSpeed and savedSpeed > 0 then
                lockedSpeed = savedSpeed
            else
                lockedSpeed = originalSpeed
            end
        end

        speedModeEnabled = true

        if speedSubMode == "加速" then
            -- 加速子模式：启动物理推力
            local ok = startPhysAcceleration(lockedSpeed)
            if not ok then
                speedModeEnabled = false
                tanchuangxiaoxi("角色未就绪，无法启动加速", "移速模式")
                updateMainButtonText()
                updateSpeedButtonText()
                return
            end
        else
            -- 修改子模式：锁定 WalkSpeed
            if speedModeConnection then speedModeConnection:Disconnect() end
            speedModeConnection = RunService.Heartbeat:Connect(function()
                if not speedModeEnabled then return end
                if speedSubMode ~= "修改" then return end
                local c = player.Character
                if c then
                    local h = c:FindFirstChildWhichIsA("Humanoid")
                    if h then pcall(function() h.WalkSpeed = lockedSpeed end) end
                end
            end)
        end

        local label = (speedSubMode == "加速") and "加速" or "移速"
        if flyWasClosed then
            tanchuangxiaoxi(label .. "已开启，飞天已关闭", "移速模式")
        else
            tanchuangxiaoxi("已开启" .. label .. "模式，目标: " .. string.format("%.1f", lockedSpeed), "移速模式")
        end
    else
        -- 关闭
        if speedModeConnection then speedModeConnection:Disconnect() speedModeConnection = nil end
        if accelEnabled then removePhysAcceleration() end

        local char = player.Character
        if char then
            local hum = char:FindFirstChildWhichIsA("Humanoid")
            if hum then pcall(function() hum.WalkSpeed = originalSpeed end) end
        end

        -- 【记忆机制】关闭时按子模式分别快照
        if lockedSpeed and lockedSpeed > 0 then
            if speedSubMode == "加速" then
                savedAccelSpeed = lockedSpeed
            else
                savedSpeed = lockedSpeed
            end
        end

        lockedSpeed = originalSpeed
        speedModeEnabled = false
        tanchuangxiaoxi("已关闭移速模式", "移速模式")
    end
    updateMainButtonText()
    updateSpeedButtonText()
    updateSpeedDisplayLoop()
end

-- ==================== 透视基础 ====================
local function saveOriginalLighting()
    local Lighting = game:GetService("Lighting")
    originalLighting = {
        Brightness = Lighting.Brightness,
        Ambient = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        ColorShift_Bottom = Lighting.ColorShift_Bottom,
        ColorShift_Top = Lighting.ColorShift_Top,
        FogEnd = Lighting.FogEnd,
        FogStart = Lighting.FogStart,
        GlobalShadows = Lighting.GlobalShadows,
        EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
        EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
    }
    atmosphereInstance = Lighting:FindFirstChildOfClass("Atmosphere")
    if atmosphereInstance then
        originalAtmosphere = {
            Density = atmosphereInstance.Density,
            Offset = atmosphereInstance.Offset,
            Color = atmosphereInstance.Color,
            Decay = atmosphereInstance.Decay,
            Glare = atmosphereInstance.Glare,
            Haze = atmosphereInstance.Haze
        }
    end
end

local function restoreLighting()
    local Lighting = game:GetService("Lighting")
    for prop, val in pairs(originalLighting) do
        pcall(function() Lighting[prop] = val end)
    end
    if atmosphereInstance and atmosphereInstance.Parent then
        for prop, val in pairs(originalAtmosphere) do
            pcall(function() atmosphereInstance[prop] = val end)
        end
    end
    for _, conn in pairs(lightingFrozenConnections) do
        pcall(function() conn:Disconnect() end)
    end
    lightingFrozenConnections = {}
end

local function freezeLightingProperty(instance, propName, targetValue, isFrozen)
    local connKey = instance:GetFullName() .. "_" .. propName
    if lightingFrozenConnections[connKey] then
        lightingFrozenConnections[connKey]:Disconnect()
        lightingFrozenConnections[connKey] = nil
    end
    pcall(function() instance[propName] = targetValue end)
    if isFrozen then
        lightingFrozenConnections[connKey] = instance:GetPropertyChangedSignal(propName):Connect(function()
            pcall(function()
                if instance[propName] ~= targetValue then instance[propName] = targetValue end
            end)
        end)
    end
end

local function applyNightVision()
    local Lighting = game:GetService("Lighting")
    Lighting.Brightness = nightVisionBrightness
    if nightVisionSettings.ambient then
        freezeLightingProperty(Lighting, "Ambient", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "OutdoorAmbient", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "ColorShift_Bottom", Color3.new(1, 1, 1), true)
        freezeLightingProperty(Lighting, "ColorShift_Top", Color3.new(1, 1, 1), true)
    else
        if originalLighting.Ambient then freezeLightingProperty(Lighting, "Ambient", originalLighting.Ambient, false) end
        if originalLighting.OutdoorAmbient then freezeLightingProperty(Lighting, "OutdoorAmbient", originalLighting.OutdoorAmbient, false) end
        if originalLighting.ColorShift_Bottom then freezeLightingProperty(Lighting, "ColorShift_Bottom", originalLighting.ColorShift_Bottom, false) end
        if originalLighting.ColorShift_Top then freezeLightingProperty(Lighting, "ColorShift_Top", originalLighting.ColorShift_Top, false) end
    end
    if nightVisionSettings.shadows then freezeLightingProperty(Lighting, "GlobalShadows", false, true)
    else if originalLighting.GlobalShadows ~= nil then freezeLightingProperty(Lighting, "GlobalShadows", originalLighting.GlobalShadows, false) end end
    if nightVisionSettings.envAo then
        freezeLightingProperty(Lighting, "EnvironmentDiffuseScale", 0, true)
        freezeLightingProperty(Lighting, "EnvironmentSpecularScale", 0, true)
    else
        if originalLighting.EnvironmentDiffuseScale then freezeLightingProperty(Lighting, "EnvironmentDiffuseScale", originalLighting.EnvironmentDiffuseScale, false) end
        if originalLighting.EnvironmentSpecularScale then freezeLightingProperty(Lighting, "EnvironmentSpecularScale", originalLighting.EnvironmentSpecularScale, false) end
    end
    if nightVisionSettings.fog then
        freezeLightingProperty(Lighting, "FogEnd", 100000, true)
        freezeLightingProperty(Lighting, "FogStart", 0, true)
    else
        if originalLighting.FogEnd then freezeLightingProperty(Lighting, "FogEnd", originalLighting.FogEnd, false) end
        if originalLighting.FogStart then freezeLightingProperty(Lighting, "FogStart", originalLighting.FogStart, false) end
    end
    if atmosphereInstance and atmosphereInstance.Parent then
        if nightVisionSettings.atmosphere then
            freezeLightingProperty(atmosphereInstance, "Density", 0, true)
            freezeLightingProperty(atmosphereInstance, "Haze", 0, true)
        else
            if originalAtmosphere.Density then freezeLightingProperty(atmosphereInstance, "Density", originalAtmosphere.Density, false) end
            if originalAtmosphere.Haze then freezeLightingProperty(atmosphereInstance, "Haze", originalAtmosphere.Haze, false) end
        end
    end
end

local function enableNoclip()
    if not player.Character then return end
    if noclipEnabled then return end

    noclipEnabled = true
    applyNoclip()

    local char, hrp, hum = getCharacterParts()
    if hum then
        if noclipStateConnection then noclipStateConnection:Disconnect() end
        noclipStateConnection = hum.StateChanged:Connect(function(oldState, newState)
            if not noclipEnabled then return end
            if newState == Enum.HumanoidStateType.Jumping or
               newState == Enum.HumanoidStateType.Freefall or
               newState == Enum.HumanoidStateType.FallingDown or
               newState == Enum.HumanoidStateType.Landed then
                applyNoclip()
            end
        end)
    end

    if noclipConnection then noclipConnection:Disconnect() end
    noclipConnection = RunService.Stepped:Connect(function()
        if noclipEnabled then applyNoclip() end
    end)

    if noclipWindow then noclipWindow:Destroy() end
    noclipWindow = createNoclipWindow()

    updateMainButtonText()
    updateSpeedButtonText()
end

    savedFeatureStates.speed = speedModeEnabled
    savedFeatureStates.noclip = noclipEnabled
    savedFeatureStates.building = buildingRemovalEnabled
end

local function disableAffectedFeatures()
    if isFlying then toggleFly(false) end
    if speedModeEnabled then applySpeedMode(false) end
    if noclipEnabled then disableNoclip() end
    if buildingRemovalEnabled then disableBuildingRemoval() end
end

    task.spawn(function()
        if savedFeatureStates.fly then toggleFly(true) task.wait(0.1) end
        if savedFeatureStates.speed then applySpeedMode(true) task.wait(0.1) end
        if savedFeatureStates.noclip then enableNoclip() task.wait(0.1) end
        if savedFeatureStates.building then enableBuildingRemoval() end
    end)
end

-- ==================== 角色重生处理 ====================
local function onCharacterAdded(char)
    task.wait(0.7)
    if char:FindFirstChild("Animate") then char.Animate.Disabled = false end

    if deathMode == 2 then restoreFeatureStates()
    elseif deathMode == 3 then disableAffectedFeatures()
    elseif deathMode == 4 then disableAffectedFeatures() restoreFeatureStates() end

    if nightVisionEnabled then enableNightVision() end
    stopTpwalking()
    updateSpeedButtonText()
    updateMainButtonText()
    updateSpeedDisplayLoop()

    local hum = char:WaitForChild("Humanoid", 3)
    if hum then
        hum.Died:Connect(function()
            if deathMode == 1 then disableAffectedFeatures()
            elseif deathMode == 2 then saveFeatureStates() disableAffectedFeatures()
            elseif deathMode == 3 then -- 死亡时不关
            elseif deathMode == 4 then saveFeatureStates() end
        end)
    end
end

player.CharacterRemoving:Connect(function(character)
    if deathMode == 1 then disableAffectedFeatures()
    elseif deathMode == 2 then saveFeatureStates() disableAffectedFeatures() end
end)

player.CharacterAdded:Connect(onCharacterAdded)
if player.Character then onCharacterAdded(player.Character) end

local function reposition()
    local y = startY
    for _, n in ipairs(notifs) do
        local f = n.frame
        TweenService:Create(f, TweenInfo.new(0.2), {Position = UDim2.new(1, -f.Size.X.Offset - 10, 0, y)}):Play()
        y = y + n.height + spacing
    end
end

local function remove(f)
    for i, n in ipairs(notifs) do
        if n.frame == f then table.remove(notifs, i) break end
    end
end

local function processReady()
    while #readyQueue > 0 do
        local n = table.remove(readyQueue, 1)
        if n and n.frame and n.frame.Parent then
            local f, sg = n.frame, n.sg
            local fadeOut = TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
            local t1 = TweenService:Create(f, fadeOut, {BackgroundTransparency = 1})
            TweenService:Create(n.title, fadeOut, {TextTransparency = 1}):Play()
            TweenService:Create(n.line, fadeOut, {BackgroundTransparency = 1}):Play()
            TweenService:Create(n.msg, fadeOut, {TextTransparency = 1}):Play()
            t1:Play()
            t1.Completed:Wait()
            remove(f)
            sg:Destroy()
        end
    end
    reposition()
end

function createMenu(title, buttons, parentMenu)

local function setUIVisible(visible)
    uiVisible = visible
    Frame.Visible = uiVisible
    if miniWindow then miniWindow.Visible = uiVisible end
    tanchuangxiaoxi(uiVisible and "UI已显示" or "UI已隐藏", "音量键")
end

local function enableVolumeKey()
    local hasVolumeDown = pcall(function() return Enum.KeyCode.VolumeDown end)
    local hasVolumeUp = pcall(function() return Enum.KeyCode.VolumeUp end)
    if not hasVolumeDown or not hasVolumeUp then
        tanchuangxiaoxi("您的设备不支持音量键控制", "提示")
        return
    end
    if volumeKeyEnabled then return end
    volumeKeyConnection = UserInputService.InputBegan:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.VolumeDown then setUIVisible(false)
        elseif input.KeyCode == Enum.KeyCode.VolumeUp then setUIVisible(true) end
    end)
    volumeKeyEnabled = true
    tanchuangxiaoxi("音量键控制已启用：减隐藏，加显示", "设置")
end

local function disableVolumeKey()
    if not volumeKeyEnabled then return end
    if volumeKeyConnection then volumeKeyConnection:Disconnect() volumeKeyConnection = nil end
    volumeKeyEnabled = false
    tanchuangxiaoxi("音量键控制已禁用", "设置")
end

-- ==================== 辅助函数：根据当前模式获取移动向量 ====================
local function getMoveVector(dir, rootPart)
    local step = dir * moveStep
    if moveMode == "角色上下" then
        return rootPart.CFrame.UpVector * step
    elseif moveMode == "角色前后" then
        return rootPart.CFrame.LookVector * step
    elseif moveMode == "角色左右" then
        return -rootPart.CFrame.RightVector * step
    elseif moveMode == "屏幕上下" then
        local camera = workspace.CurrentCamera
        if camera then return camera.CFrame.UpVector * step end
    elseif moveMode == "屏幕前后" then
        local camera = workspace.CurrentCamera
        if camera then return camera.CFrame.LookVector * step end
    elseif moveMode == "屏幕左右" then
        local camera = workspace.CurrentCamera
        if camera then return -camera.CFrame.RightVector * step end
    elseif moveMode == "水平上下" then
        return Vector3.new(0, step, 0)
    elseif moveMode == "水平前后(屏幕)" then
        local camera = workspace.CurrentCamera
        if camera then
            local look = camera.CFrame.LookVector
            local horizontal = Vector3.new(look.X, 0, look.Z)
            if horizontal.Magnitude > 0 then return horizontal.Unit * step end
        end
    elseif moveMode == "水平左右(屏幕)" then
        local camera = workspace.CurrentCamera
        if camera then
            local right = camera.CFrame.RightVector
            local horizontal = Vector3.new(right.X, 0, right.Z)
            if horizontal.Magnitude > 0 then return horizontal.Unit * step end
        end
    end
    return Vector3.new()
end

-- ==================== 按钮长按逻辑 ====================
-- 上升按钮
do
    local holding = false
    local longPressTask = nil

    local function startLongPress()
        if not holding then return end
        local interval = longPressSpeed
        while holding do
            local char = player.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local rootPart = char.HumanoidRootPart
                local delta = getMoveVector(1, rootPart)
                rootPart.CFrame = rootPart.CFrame + delta
            end
            task.wait(interval)
            interval = math.max(0.001, interval * 0.9)
        end
    end

    up.MouseButton1Down:Connect(function()
        if holding then return end
        holding = true
        local char = player.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local rootPart = char.HumanoidRootPart
            local delta = getMoveVector(1, rootPart)
            rootPart.CFrame = rootPart.CFrame + delta
        end
        longPressTask = task.delay(0.3, function()
            if holding then startLongPress() end
        end)
    end)

    local function stopPress()
        if holding then
            holding = false
            if longPressTask then task.cancel(longPressTask) longPressTask = nil end
        end
    end

    up.MouseButton1Up:Connect(stopPress)
    up.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            stopPress()
        end
    end)
end

-- 减速 / 减少 按钮（【本次】同时适配修改/加速子模式）
do
    local holding = false
    local longPressTask = nil
    local MIN_SPEED = 0.1
    local MIN_LOCKED = 0.1
    local MIN_ACCEL = 10    -- 加速模式最小目标速度

    local function getMinLocked()
        return (speedSubMode == "加速") and MIN_ACCEL or MIN_LOCKED
    end

    local function decreaseFlySpeed()
        local current = tonumber(speeds) or 0
        if current > incStep then current = current - incStep
        elseif current > MIN_SPEED then current = MIN_SPEED
        else
            speed.Text = "已达最小速度"
            task.wait(1)
            speed.Text = tostring(speeds)
            return false
        end
        speeds = current
        speed.Text = tostring(speeds)
        return true
    end

    local function decreaseLockedSpeed()
        local minLock = getMinLocked()
        if lockedSpeed > incStep then lockedSpeed = lockedSpeed - incStep
        elseif lockedSpeed > minLock then lockedSpeed = minLock
        else
            speed.Text = "已达最小速度"
            task.wait(1)
            updateSpeedButtonText()
            return false
        end
        return true
    end

    local function decreaseNightVision()
        nightVisionBrightness = clamp(nightVisionBrightness - 0.05, 0.01, 100)
        nightVisionBrightness = math.floor(nightVisionBrightness * 100 + 0.5) / 100
        if nightVisionEnabled then applyNightVision() end
        updateSpeedButtonText()
        return true
    end

    local function startLongPress()
        if not holding then return end
        local interval = longPressSpeed
        while holding do
            local success = false
            if modeIndex == 0 then success = decreaseFlySpeed()
            elseif modeIndex == 1 and speedModeEnabled then
                success = decreaseLockedSpeed()
                if success then updateSpeedButtonText() end
            elseif modeIndex == 4 then success = decreaseNightVision() end
            if not success then break end
            task.wait(interval)
            interval = math.max(0.001, interval * 0.9)
        end
    end

    mine.MouseButton1Down:Connect(function()
        if holding then return end
        if modeIndex == 1 and not speedModeEnabled then return end
        holding = true
        if modeIndex == 0 then decreaseFlySpeed()
        elseif modeIndex == 1 then
            decreaseLockedSpeed()
            updateSpeedButtonText()
        elseif modeIndex == 4 then decreaseNightVision() end
        longPressTask = task.delay(0.3, function()
            if holding then startLongPress() end
        end)
    end)

    local function stopPress()
        if holding then
            holding = false
            if longPressTask then task.cancel(longPressTask) longPressTask = nil end
        end
    end

    mine.MouseButton1Up:Connect(stopPress)
    mine.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            stopPress()
        end
    end)
end

                                tanchuangxiaoxi("移速子模式已切换至: " .. speedSubMode, "移速模式")
                            end)
                        end
                    }
                )
            elseif modeIndex == 3 then
                showInputDialog("输入透明度 (0-1)", tostring(buildingTransparency), function(val)
                    local num = tonumber(val)
                    if num and num >= 0 and num <= 1 then
                        buildingTransparency = num
                        if buildingRemovalEnabled and (buildingRemovalMode == 2 or buildingRemovalMode == 3) then
                            disableBuildingRemoval()
                            enableBuildingRemoval()
                        end
                        updateSpeedButtonText()
                        tanchuangxiaoxi("贴图透明度快捷调整为 " .. num, "透明度设置")
                    else tanchuangxiaoxi("请输入0到1之间的数字", "错误") end
                end)
            elseif modeIndex == 4 then
                showBrightnessDialog()
            end
            holding = false
        end
    end

    speed.MouseButton1Up:Connect(onRelease)
    speed.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            onRelease()
        end
    end)
end

-- ==================== 主按钮长按/单击逻辑 ====================
do
    local holding = false
    local longPressTask = nil
    local isLongPress = false

    onof.MouseButton1Down:Connect(function()
        if holding then return end
        holding = true
        isLongPress = false

        longPressTask = task.delay(0.3, function()
            if holding then
                isLongPress = true
                modeIndex = (modeIndex + 1) % 5
                updateMainButtonText()
                updateSpeedButtonText()
                updateSpeedDisplayLoop()
                tanchuangxiaoxi("已切换至" .. modeDisplayNames[modeIndex + 1] .. "模式", "模式切换")
                holding = false
                longPressTask = nil
            end
        end)
    end)

    local function onUp()
        if holding then
            if longPressTask then task.cancel(longPressTask) longPressTask = nil end
            if not isLongPress then
                if modeIndex == 0 then toggleFly(not isFlying)
                elseif modeIndex == 1 then
                    speedModeEnabled = not speedModeEnabled
                    applySpeedMode(speedModeEnabled)
                elseif modeIndex == 2 then
                    if noclipEnabled then disableNoclip() else enableNoclip() end
                elseif modeIndex == 3 then
                    if buildingRemovalEnabled then disableBuildingRemoval() else enableBuildingRemoval() end
                elseif modeIndex == 4 then
                    if nightVisionEnabled then disableNightVision() else enableNightVision() end
                end
            end
            holding = false
        end
    end

    onof.MouseButton1Up:Connect(onUp)
    onof.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            onUp()
        end
    end)
end

-- ==================== 隐藏按钮（迷你窗口 - 双修复版） ====================
do
    local holding = false
    local longPressTask = nil
    local isLongPress = false

    hide.MouseButton1Down:Connect(function()
        if holding then return end
        holding = true
        isLongPress = false

        longPressTask = task.delay(0.3, function()
            if holding then
                isLongPress = true
                showMainMenu()
                holding = false
                longPressTask = nil
            end
        end)
    end)

    local function onUp()
        if holding then
            if longPressTask then task.cancel(longPressTask) longPressTask = nil end
            if not isLongPress then
                Frame.Visible = false

                local vp = getScreenSize()
                local miniX = clamp(frameX + 132, 0, vp.X - 58)
                local miniY = clamp(frameY, 0, vp.Y - 28)

                local mini = Instance.new("Frame")
                mini.Name = "MiniUI"
                mini.Parent = main
                mini.BackgroundColor3 = Color3.fromRGB(163, 255, 137)
                mini.BorderColor3 = Color3.fromRGB(103, 221, 213)
                mini.Size = UDim2.new(0, 58, 0, 28)
                mini.Position = UDim2.fromOffset(miniX, miniY)
                mini.Active = true

                local miniBtn = Instance.new("TextButton")
                miniBtn.Name = "MiniButton"
                miniBtn.Parent = mini
                miniBtn.Size = UDim2.new(1, 0, 1, 0)
                miniBtn.BackgroundColor3 = Color3.fromRGB(255, 249, 74)
                miniBtn.Text = "UI"
                miniBtn.TextColor3 = Color3.fromRGB(0, 0, 0)
                miniBtn.TextScaled = true
                miniBtn.Font = Enum.Font.SourceSans
                miniBtn.Active = true

                local miniX_var = miniX
                local miniY_var = miniY

                local miniIsDragging = false
                local miniDragTouch = nil
                local miniLastTouchPos = nil
                local miniLatestTouchPos = nil
                local miniTotalDelta = 0
                local miniPressStartPos = nil

                miniBtn.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.Touch then
                        miniIsDragging = true
                        miniDragTouch = input
                        miniLastTouchPos = Vector2.new(input.Position.X, input.Position.Y)
                        miniLatestTouchPos = miniLastTouchPos
                        miniTotalDelta = 0
                        miniPressStartPos = Vector2.new(input.Position.X, input.Position.Y)
                    end
                end)

                UserInputService.InputChanged:Connect(function(input)
                    if miniIsDragging and input == miniDragTouch and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
                        local newPos = Vector2.new(input.Position.X, input.Position.Y)
                        miniTotalDelta = miniTotalDelta + (newPos - miniLatestTouchPos).Magnitude
                        miniLatestTouchPos = newPos
                    end
                end)

                UserInputService.InputEnded:Connect(function(input)
                    if input == miniDragTouch then
                        local endPos = Vector2.new(input.Position.X, input.Position.Y)
                        local straightDist = miniPressStartPos and (endPos - miniPressStartPos).Magnitude or 0

                        if straightDist < 10 and miniTotalDelta < 15 then
                            local vp2 = getScreenSize()
                            local frameW = Frame.Size.X.Offset
                            local frameH = Frame.Size.Y.Offset
                            local miniW = mini.Size.X.Offset
                            local miniH = mini.Size.Y.Offset

                            local newX = miniX_var - (frameW - miniW) / 2
                            local newY = miniY_var - (frameH - miniH) / 2

                            newX = clamp(newX, 0, vp2.X - frameW)
                            newY = clamp(newY, 0, vp2.Y - frameH)

                            setFramePos(newX, newY)
                            Frame.Visible = true
                            mini:Destroy()
                            miniWindow = nil
                        end
                        miniIsDragging = false
                        miniDragTouch = nil
                        miniLastTouchPos = nil
                        miniLatestTouchPos = nil
                        miniPressStartPos = nil
                    end
                end)

                local miniRenderConn
                miniRenderConn = RunService.RenderStepped:Connect(function()
                    if not mini or not mini.Parent then
                        if miniRenderConn then miniRenderConn:Disconnect() miniRenderConn = nil end
                        return
                    end
                    if not miniIsDragging or not miniLastTouchPos or not miniLatestTouchPos then return end

                    local delta = miniLatestTouchPos - miniLastTouchPos
                    miniLastTouchPos = miniLatestTouchPos

                    local newX = miniX_var + delta.X
                    local newY = miniY_var + delta.Y

                    local camera = workspace.CurrentCamera
                    if not camera then return end
                    local vp3 = camera.ViewportSize
                    local miniW = mini.Size.X.Offset
                    local miniH = mini.Size.Y.Offset

                    newX = clamp(newX, 0, vp3.X - miniW)
                    newY = clamp(newY, 0, vp3.Y - miniH)

                    miniX_var = newX
                    miniY_var = newY
                    mini.Position = UDim2.fromOffset(newX, newY)
                end)

                miniWindow = mini
            end
            holding = false
        end
    end

    hide.MouseButton1Up:Connect(onUp)
    hide.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            onUp()
        end
    end)
end

    textBox.Position = UDim2.new(0.2, 0, 0, 50)
    textBox.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
    textBox.TextColor3 = Color3.new(1, 1, 1)
    textBox.PlaceholderText = "亮度 (0.01~100)"
    textBox.Text = string.format("%.2f", nightVisionBrightness)
    textBox.Font = Enum.Font.Gotham
    textBox.TextSize = 14
    textBox.ClearTextOnFocus = false

    local sliderBg = Instance.new("Frame")
    sliderBg.Parent = bg
    sliderBg.Size = UDim2.new(0.8, 0, 0, 4)
    sliderBg.Position = UDim2.new(0.1, 0, 0, 120)
    sliderBg.BackgroundColor3 = Color3.fromRGB(80, 80, 80)

    sliderCorner.Parent = sliderBg

    local knob = Instance.new("TextButton")
    knob.Size = UDim2.new(0, 20, 0, 20)
    local minBright = 0.01
    local maxBright = 100
    local percent = (nightVisionBrightness - minBright) / (maxBright - minBright)
    knob.Position = UDim2.new(percent, -10, 0, -8)
    knob.BackgroundColor3 = Color3.fromRGB(200, 200, 200)
    knob.Text = ""
    knob.Parent = sliderBg
    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = knob

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    local conn
    conn = RunService.RenderStepped:Connect(function()
        if dragging then
            local mousePos = UserInputService:GetMouseLocation()
            local absPos = sliderBg.AbsolutePosition
            local absSize = sliderBg.AbsoluteSize.X
            local relX = clamp(mousePos.X - absPos.X, 0, absSize)
            local currentPercent = relX / absSize
            local newBrightness = minBright + currentPercent * (maxBright - minBright)
            newBrightness = math.floor(newBrightness * 100) / 100
            nightVisionBrightness = newBrightness
        end
        local syncPercent = (nightVisionBrightness - minBright) / (maxBright - minBright)
        knob.Position = UDim2.new(syncPercent, -10, 0, -8)
        if not textBox:IsFocused() then
            textBox.Text = string.format("%.2f", nightVisionBrightness)
        end
        if modeIndex == 4 then updateSpeedButtonText() end
    end)

    textBox.FocusLost:Connect(function()
        local num = tonumber(textBox.Text)
        if num then
            num = clamp(num, minBright, maxBright)
            num = math.floor(num * 100) / 100
            textBox.Text = string.format("%.2f", num)
            nightVisionBrightness = num
            if nightVisionEnabled then applyNightVision() end
        else textBox.Text = string.format("%.2f", nightVisionBrightness) end
    end)

    local buttonFrame = Instance.new("Frame")
    buttonFrame.Parent = bg
    buttonFrame.Size = UDim2.new(1, -20, 0, 40)
    buttonFrame.Position = UDim2.new(0, 10, 1, -50)
    buttonFrame.BackgroundTransparency = 1

    local cancelBtn = Instance.new("TextButton")
    cancelBtn.Parent = buttonFrame
    cancelBtn.Size = UDim2.new(0.4, -5, 1, 0)
    cancelBtn.Position = UDim2.new(0.1, 0, 0, 0)
    cancelBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
    cancelBtn.Text = "取消"
    cancelBtn.TextColor3 = Color3.new(1, 1, 1)
    cancelBtn.Font = Enum.Font.GothamBold
    cancelBtn.TextSize = 14
    local cancelCorner = Instance.new("UICorner")
    cancelCorner.CornerRadius = UDim.new(0, 6)
    cancelCorner.Parent = cancelBtn

    local confirmBtn = Instance.new("TextButton")
    confirmBtn.Parent = buttonFrame
    confirmBtn.Size = UDim2.new(0.4, -5, 1, 0)
    confirmBtn.Position = UDim2.new(0.5, 5, 0, 0)
    confirmBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
    confirmBtn.Text = "确认"
    confirmBtn.TextColor3 = Color3.new(1, 1, 1)
    confirmBtn.Font = Enum.Font.GothamBold
    confirmBtn.TextSize = 14
    local confirmCorner = Instance.new("UICorner")
    confirmCorner.CornerRadius = UDim.new(0, 6)
    confirmCorner.Parent = confirmBtn

    cancelBtn.MouseButton1Click:Connect(function()
        if conn then conn:Disconnect() end
        if inputEndConn then inputEndConn:Disconnect() end
        dialog:Destroy()
    end)

    confirmBtn.MouseButton1Click:Connect(function()
        local num = tonumber(textBox.Text)
        if num then
            num = clamp(num, minBright, maxBright)
            num = math.floor(num * 100) / 100
            nightVisionBrightness = num
            if nightVisionEnabled then applyNightVision() end
            tanchuangxiaoxi("透视亮度已设为 " .. tostring(num), "亮度设置")
        end
        if conn then conn:Disconnect() end
        if inputEndConn then inputEndConn:Disconnect() end
        dialog:Destroy()
    end)
end
