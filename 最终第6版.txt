local function isPointInGui(g, x, y)
    if not g or not g.Parent then return false end
    local tl, sz = g.AbsolutePosition, g.AbsoluteSize

local function bindSafeClick(btn, callback)
    local pressInput, pressPos
    btn.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.Touch
            and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
        if input.UserInputState ~= Enum.UserInputState.Begin then return end
        pressInput = input
        pressPos = Vector2.new(input.Position.X, input.Position.Y)
    end)
    btn.InputEnded:Connect(function(input)
        if input ~= pressInput then return end
        local pos = Vector2.new(input.Position.X, input.Position.Y)
        local d = pressPos and (pos - pressPos).Magnitude or 999
        local wasInside = isPointInGui(btn, pos.X, pos.Y)
        pressInput = nil
        pressPos = nil
        if wasInside and d < 30 then
            local ok, err = pcall(callback)
            if not ok then warn("[bindSafeClick] " .. tostring(err)) end
        end
    end)
end

local function isProtected(target)
    if not target then return false end
    if state.protectedTargets[target] then return true end
    local cur = target
    local depth = 0
    while cur and cur ~= workspace and depth < 30 do
        if state.protectedTargets[cur] then return true end
        cur = cur.Parent
        depth = depth + 1
    end
    return false
end

local function isProtectedTree(target)
    if not target then return false end
    if isProtected(target) then return true end
    if target:IsA("Model") then
        local ok, descendants = pcall(function() return target:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                if state.protectedTargets[d] then return true end
            end
        end
    end
    return false
end

local function extractKeyword(name)
    if not name or name == "" then return "" end
    local s = name
    s = s:gsub("%s*%(%d+%)%s*$", "")
    s = s:gsub("[_%s%-]+%d+%s*$", "")
    s = s:gsub("%d+$", "")
    s = s:gsub("[_%s%-]+$", "")
    s = s:match("^%s*(.-)%s*$") or s
    if s == "" then s = name end
    return string.lower(s)
end

}

local function isGenericKeyword(kw)
    if not kw or kw == "" then return true end
    if #kw < 3 then return true end
    return GENERIC_WORDS[kw] == true
end

local function tryExtractKeyword(name)
    local k = extractKeyword(name)
    if k == "" then return nil end
    if isGenericKeyword(k) then return nil end
    return k
end

local function findBestKeyword(root)
    if not root then return nil end
    local kw = tryExtractKeyword(root.Name)
    if kw then return kw end
    local ok, descendants = pcall(function() return root:GetDescendants() end)
    if ok and descendants then
        for _, d in ipairs(descendants) do
            if d:IsA("BasePart") then
                local k2 = tryExtractKeyword(d.Name)
                if k2 then return k2 end
            end
        end
    end
    local fallback = extractKeyword(root.Name)
    if fallback == "" or isGenericKeyword(fallback) then return nil end
    return fallback
end

local function sanitizeNote(val)
    if val == nil then return "" end
    val = tostring(val)
    val = val:gsub("[\r\n]+", " ")
    if #val > 80 then val = string.sub(val, 1, 80) end
    return val
end

local function isVisibleForRaycast(part)
    if not part or not part:IsA("BasePart") then return false end
    if part:IsA("Terrain") then return false end
    if part.Transparency >= 0.95 then return false end
    if part.LocalTransparencyModifier >= 0.95 then return false end
    local n = string.lower(part.Name)
    if string.find(n, "hitbox", 1, true) then return false end
    if string.find(n, "trigger", 1, true) then return false end
    if string.find(n, "zone", 1, true) then return false end
    if string.find(n, "invisible", 1, true) then return false end
    if string.find(n, "collision", 1, true) then return false end
    return true
end

local function resolveHitPart(part, hitPos)
    if not part or not part:IsA("BasePart") then return nil end
    if part:IsA("Terrain") then return nil end
    if isVisibleForRaycast(part) then return part end
    local cur = part
    local depth = 0
    while cur and cur ~= workspace and depth < 5 do
        if cur:IsA("Model") then
            local bestPart = nil
            local bestDist = math.huge
            local count = 0
            local ok, descendants = pcall(function() return cur:GetDescendants() end)
            if ok and descendants then
                for _, d in ipairs(descendants) do
                    count = count + 1
                    if count > 200 then break end
                    if d:IsA("BasePart") and not d:IsA("Terrain") and isVisibleForRaycast(d) then
                        local dist = (d.Position - hitPos).Magnitude
                        if dist < bestDist then bestDist = dist; bestPart = d end
                    end
                end
            end
            if bestPart then return bestPart end
        end
        cur = cur.Parent
        depth = depth + 1
    end
    return nil
end

local function isVisiblePart(part)
    if not part:IsA("BasePart") then return false end

    local n = string.lower(part.Name)

    if string.find(n, "trigger", 1, true) then return false end
    if string.find(n, "zone", 1, true) then return false end
    if string.find(n, "invisible", 1, true) then return false end
    return true
end

local function getModelCenter(target)
    if target:IsA("BasePart") then return target.Position end
    if target:IsA("Model") then
        local hrp = target:FindFirstChild("HumanoidRootPart")
        if hrp and hrp:IsA("BasePart") then return hrp.Position end
        local sum = Vector3.new(0, 0, 0)
        local n = 0
        local ok, descendants = pcall(function() return target:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                if d:IsA("BasePart") and isVisiblePart(d) then
                    sum = sum + d.Position; n = n + 1
                end
            end
        end
        if n > 0 then return sum / n end
        local ok2, cf = pcall(function() return target:GetBoundingBox() end)
        if ok2 and cf then return cf.Position end
    end
    return nil
end

local function getBBox2D(target)
    local parts = {}
    if target:IsA("BasePart") then parts = { target }
    elseif target:IsA("Model") then
        local ok, descendants = pcall(function() return target:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                if d:IsA("BasePart") and isVisiblePart(d) then
                    table.insert(parts, d)
                end
            end
        end
    end
    if #parts == 0 then return nil end
    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local any = false
    for _, part in ipairs(parts) do
        local cf, sz = part.CFrame, part.Size
        local hx, hy, hz = sz.X / 2, sz.Y / 2, sz.Z / 2
        local corners = {
            cf * Vector3.new(-hx, -hy, -hz), cf * Vector3.new(hx, -hy, -hz),
            cf * Vector3.new(-hx, hy, -hz), cf * Vector3.new(hx, hy, -hz),
            cf * Vector3.new(-hx, -hy, hz), cf * Vector3.new(hx, -hy, hz),
            cf * Vector3.new(-hx, hy, hz), cf * Vector3.new(hx, hy, hz),
        }
        for _, v in ipairs(corners) do
            local sp, on = camera:WorldToScreenPoint(v)
            if on and sp.Z > 0 then
                any = true
                local safeX = math.clamp(sp.X, -10000, 10000)
                local safeY = math.clamp(sp.Y, -10000, 10000)
                if safeX < minX then minX = safeX end
                if safeX > maxX then maxX = safeX end
                if safeY < minY then minY = safeY end
                if safeY > maxY then maxY = safeY end
            end
        end
    end
    if not any then return nil end
    return minX, minY, maxX, maxY
end

function getTargetInCenter(includeProtected)
    local vp = camera.ViewportSize
    local ray = camera:ViewportPointToRay(vp.X / 2, vp.Y / 2)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filterList = {}
    if player.Character then table.insert(filterList, player.Character) end
    if not includeProtected then
        for t in pairs(state.protectedTargets) do
            if t then table.insert(filterList, t) end
        end
    end
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and not isKeywordDrawEnabled(entry.keyword) then
            table.insert(filterList, target)
        end
    end
    params.FilterDescendantsInstances = filterList
    params.IgnoreWater = true
    local origin = ray.Origin
    local direction = ray.Direction.Unit
    local remainingDist = 1000
    local attempts = 0
    while attempts < 15 and remainingDist > 0.5 do
        attempts = attempts + 1
        local r = workspace:Raycast(origin, direction * remainingDist, params)
        if not r then break end
        local part = r.Instance
        if part and part:IsA("BasePart") then
            local resolved = resolveHitPart(part, r.Position)
            if resolved then return resolved end
        end
        local dist = (r.Position - origin).Magnitude
        if dist <= 0 then break end
        origin = r.Position + direction * 0.1
        remainingDist = remainingDist - dist - 0.1
    end
    return nil
end

local TOUCH_MATCH_THRESHOLD = 150

local function isStartInput(i)
    return i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch
end

titleBar.InputBegan:Connect(function(input)
    if not isStartInput(input) then return end
    if input.UserInputState ~= Enum.UserInputState.Begin then return end
    dragging = true; dragInput = input; dragLastPos = input.Position
    dragStartScreen = Vector2.new(input.Position.X, input.Position.Y)
    dragStartFrameX = frameX; dragStartFrameY = frameY
end)

addConn(UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    local matched = false
    if input == dragInput then matched = true
    elseif dragInput and dragInput.UserInputType == Enum.UserInputType.MouseButton1
        and input.UserInputType == Enum.UserInputType.MouseMovement then matched = true
    elseif dragInput and dragInput.UserInputType == Enum.UserInputType.Touch
        and input.UserInputType == Enum.UserInputType.Touch then
        if dragLastPos then
            local d = (input.Position - dragLastPos).Magnitude
            if d < TOUCH_MATCH_THRESHOLD then matched = true end
        end
    end
    if not matched then return end
    dragLastPos = input.Position
    local cur = Vector2.new(input.Position.X, input.Position.Y)
    local d = cur - dragStartScreen
    local newX, newY = clampFramePos(dragStartFrameX + d.X, dragStartFrameY + d.Y)
    setFramePos(newX, newY)
end))

addConn(UserInputService.InputEnded:Connect(function(input)
    if not dragging then return end
    if input == dragInput then
        dragging = false; dragInput = nil; dragLastPos = nil; dragStartScreen = nil

    end
end))

local function parseColorInput(str)
    if type(str) ~= "string" then return nil end
    str = str:gsub("^%s+", ""):gsub("%s+$", "")
    if str == "" then return nil end
    local hex6 = str:match("^#?(%x%x%x%x%x%x)$")
    if hex6 then
        return Color3.new(
            tonumber(hex6:sub(1,2), 16) / 255,
            tonumber(hex6:sub(3,4), 16) / 255,
            tonumber(hex6:sub(5,6), 16) / 255)
    end
    local hex3 = str:match("^#(%x%x%x)$")
    if hex3 then
        return Color3.new(
            tonumber(hex3:sub(1,1):rep(2), 16) / 255,
            tonumber(hex3:sub(2,2):rep(2), 16) / 255,
            tonumber(hex3:sub(3,3):rep(2), 16) / 255)
    end
    local lower = str:lower():gsub("%s+", "")
    if NAMED_COLORS[lower] then
        local c = NAMED_COLORS[lower]
        return Color3.fromRGB(c[1], c[2], c[3])
    end
    local nums = {}
    for n in str:gmatch("%-?%d+%.?%d*") do
        table.insert(nums, tonumber(n))
    end
    if #nums == 3 then
        local r, g, b = nums[1], nums[2], nums[3]
        if r <= 1 and g <= 1 and b <= 1 and r >= 0 and g >= 0 and b >= 0 then
            return Color3.new(math.clamp(r, 0, 1), math.clamp(g, 0, 1), math.clamp(b, 0, 1))
        end
        return Color3.fromRGB(
            math.clamp(math.floor(r), 0, 255),
            math.clamp(math.floor(g), 0, 255),
            math.clamp(math.floor(b), 0, 255))
    end
    return nil
end

-- 收集目标所有 BasePart（在 registerDrawEntry 之前定义）
local function collectParts(target)
    local parts = {}
    if target:IsA("Model") then
        local ok, desc = pcall(function() return target:GetDescendants() end)
        if ok and desc then
            for _, d in ipairs(desc) do
                if d:IsA("BasePart") then table.insert(parts, d) end
            end
        end
    elseif target:IsA("BasePart") then
        table.insert(parts, target)
    end
    return parts
end

-- 冻结：记录 part 原始 Anchored
local freezeAnchoredSaved = {}

local collisionRefCount = {}
local collisionOriginal = {}

local function restoreCollisionsFrom(entry)
    if not entry or not entry.affectedParts then return end
    local affected = entry.affectedParts
    entry.affectedParts = nil
    for part in pairs(affected) do
        local count = collisionRefCount[part]
        if count then
            count = count - 1
            if count <= 0 then
                collisionRefCount[part] = nil
                local orig = collisionOriginal[part]
                collisionOriginal[part] = nil
                if part and part.Parent and orig ~= nil then
                    pcall(function() part.CanCollide = orig end)
                end
            else
                collisionRefCount[part] = count
            end
        end
    end
end

local function applyCollisionRemovalFor(root, entry)
    if not root or not entry then return end
    entry.affectedParts = entry.affectedParts or {}
    local function processPart(d)
        if not d:IsA("BasePart") then return end
        if d:IsA("Terrain") then return end
        if entry.affectedParts[d] then return end
        entry.affectedParts[d] = true
        if collisionRefCount[d] == nil then
            collisionRefCount[d] = 0
            collisionOriginal[d] = d.CanCollide
        end
        collisionRefCount[d] = collisionRefCount[d] + 1
        pcall(function() d.CanCollide = false end)
    end
    if root:IsA("Model") then
        local ok, descendants = pcall(function() return root:GetDescendants() end)
        if ok and descendants then
            for _, d in ipairs(descendants) do
                processPart(d)
            end
        end
    else
        processPart(root)
    end
end

local function destroyEntry(entry)
    if not entry then return end
    if entry.frozenParts then
        for part in pairs(entry.frozenParts) do
            local orig = freezeAnchoredSaved[part]
            if orig ~= nil then
                if part and part.Parent then
                    pcall(function() part.Anchored = orig end)
                end
                freezeAnchoredSaved[part] = nil
            end
        end
        entry.frozenParts = nil
    end
    restoreCollisionsFrom(entry)
    -- 不再需要destroyGroup，池由全局管理
    if entry.label then pcall(function() entry.label:Destroy() end) end
end

local function registerDrawEntry(root, entry)
    drawGroups[root] = entry
    if state.collisionRemovalOn then
        applyCollisionRemovalFor(root, entry)
    end
    if state.freezeOn and isKeywordDrawEnabled(entry.keyword) then
        entry.frozenParts = entry.frozenParts or {}
        local parts = collectParts(root)
        for _, d in ipairs(parts) do
            if freezeAnchoredSaved[d] == nil then
                freezeAnchoredSaved[d] = d.Anchored
            end
            entry.frozenParts[d] = true
            pcall(function() d.Anchored = true end)
        end
    end
end

local function deleteKeywordGroup(kw, delay)
    if not kw or kw == "" then return end
    pauseScanAndResume(delay or 1.5)
    if isSingleKeyword(kw) then
        local removeTarget = nil
        for target, entry in pairs(drawGroups) do
            if entry.keyword == kw then removeTarget = target; break end
        end
        if removeTarget then
            destroyEntry(drawGroups[removeTarget])
            drawGroups[removeTarget] = nil
        end
        state.keywordNotes[kw] = nil
        state.keywordDrawEnabled[kw] = nil
        state.keywordStyle[kw] = nil
        local newLogs = {}
        for _, l in ipairs(state.logs) do
            if l.keyword ~= kw then table.insert(newLogs, l) end
        end
        state.logs = newLogs
        updateLogBtn()
        return
    end
    local removeTargets = {}
    for target, entry in pairs(drawGroups) do
        if entry.keyword == kw then table.insert(removeTargets, target) end
    end
    for _, target in ipairs(removeTargets) do
        local entry = drawGroups[target]
        if entry then destroyEntry(entry); drawGroups[target] = nil end
    end
    state.keywords[kw] = nil
    state.keywordNotes[kw] = nil
    state.keywordDrawEnabled[kw] = nil
    state.keywordStyle[kw] = nil
    local newLogs = {}
    for _, l in ipairs(state.logs) do
        if l.keyword ~= kw then table.insert(newLogs, l) end
    end
    state.logs = newLogs
    updateLogBtn()
end

local function getBlackholeTargetCF()
    local vp = camera.ViewportSize
    local gs = state.globalStyle
    local screenX = vp.X / 2 + gs.blackholeOffsetX
    local screenY = vp.Y / 2 + gs.blackholeOffsetY
    local ray = camera:ViewportPointToRay(screenX, screenY)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filterList = {}
    if player.Character then table.insert(filterList, player.Character) end
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            table.insert(filterList, target)
        end
    end
    params.FilterDescendantsInstances = filterList
    params.IgnoreWater = true
    local hit = workspace:Raycast(ray.Origin, ray.Direction * gs.blackholeFarDistance, params)
    local camCF = camera.CFrame
    local rightVec = camCF.RightVector
    local upVec = camCF.UpVector
    local forwardVec = ray.Direction.Unit
    local finalPos
    if hit then
        finalPos = hit.Position
            + rightVec * gs.blackholeHitOffsetX
            + upVec * gs.blackholeHitOffsetY
            - forwardVec * gs.blackholeHitBackDist
    else
        finalPos = ray.Origin + forwardVec * gs.blackholeEmptyDistance
    end
    return CFrame.new(finalPos)
end

local function blackholeSnapOnce()
    if not state.drawOn then return 0 end
    if next(drawGroups) == nil then return 0 end
    local targetCF = getBlackholeTargetCF()
    local savedMap = {}
    local touched = {}

    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            entry.frozenParts = entry.frozenParts or {}
            local parts = collectParts(target)
            for _, d in ipairs(parts) do
                if savedMap[d] == nil then savedMap[d] = d.Anchored end
                entry.frozenParts[d] = true
                pcall(function() d.Anchored = true end)
                table.insert(touched, d)
            end
        end
    end
    if #touched == 0 then return 0 end
    task.wait(0.05)
    local movedCount = 0
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            if target:IsA("Model") then
                local centerPos = getModelCenter(target)
                if centerPos then
                    local delta = targetCF.Position - centerPos
                    local oldPivot = target:GetPivot()
                    pcall(function() target:PivotTo(oldPivot + delta) end)
                else
                    pcall(function() target:PivotTo(targetCF) end)
                end
                movedCount = movedCount + 1
            elseif target:IsA("BasePart") then
                pcall(function() target.CFrame = targetCF end); movedCount = movedCount + 1
            end
        end
    end
    task.wait(0.1)
    for part, orig in pairs(savedMap) do
        if part and part.Parent then
            pcall(function() part.Anchored = orig end)
        end
    end
    return movedCount
end

local function blackholeSnapContinuous()
    if not state.drawOn then return 0 end
    if next(drawGroups) == nil then return 0 end
    local targetCF = getBlackholeTargetCF()
    local movedCount = 0
    for target, entry in pairs(drawGroups) do
        local ok, parent = pcall(function() return target.Parent end)
        if ok and parent and isKeywordDrawEnabled(entry.keyword) and not isProtectedTree(target) then
            if target:IsA("Model") then
                local centerPos = getModelCenter(target)
                if centerPos then
                    local delta = targetCF.Position - centerPos
                    local oldPivot = target:GetPivot()
                    pcall(function() target:PivotTo(oldPivot + delta) end)
                else
                    pcall(function() target:PivotTo(targetCF) end)
                end
                movedCount = movedCount + 1
            elseif target:IsA("BasePart") then
                pcall(function() target.CFrame = targetCF end); movedCount = movedCount + 1
            end
            if state.freezeOn then
                entry.frozenParts = entry.frozenParts or {}
                local parts = collectParts(target)
                for _, d in ipairs(parts) do
                    if freezeAnchoredSaved[d] == nil then
                        freezeAnchoredSaved[d] = d.Anchored
                    end
                    entry.frozenParts[d] = true
                    if d.Anchored ~= true then
                        pcall(function() d.Anchored = true end)
                    end
                end
            end
        end
    end
    return movedCount
end

local logPressInput = nil
local logPressPos = nil
local logPressTime = 0
local logLongFired = false
local LOG_LONG_DELAY = 0.55

logBtn.InputBegan:Connect(function(input)
    if input.UserInputType ~= Enum.UserInputType.Touch
        and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
    if input.UserInputState ~= Enum.UserInputState.Begin then return end
    logPressInput = input
    logPressPos = Vector2.new(input.Position.X, input.Position.Y)
    logPressTime = tick()
    logLongFired = false
end)

task.spawn(function()
    while logBtn.Parent do
        task.wait(0.05)
        if logPressInput and not logLongFired and tick() - logPressTime >= LOG_LONG_DELAY then
            logLongFired = true
            pauseScanAndResume(2)
            for t, entry in pairs(drawGroups) do destroyEntry(entry) end
            state.logs = {}
            state.keywords = {}
            state.keywordNotes = {}
            state.keywordDrawEnabled = {}
            state.protectedTargets = {}
            state.keywordStyle = {}
            state.targetCamps = {}
            drawGroups = {}
            updateBlockedCamps()
            updateLogBtn()
            updateProtectBtn()
            addLog("日志已全部清空")
            if logWindow and logWindow.Parent then logWindow:Destroy(); logWindow = nil end
        end
    end
end)

logBtn.InputEnded:Connect(function(input)
    if input ~= logPressInput then return end
    local pos = Vector2.new(input.Position.X, input.Position.Y)
    local d = logPressPos and (pos - logPressPos).Magnitude or 999
    local wasInside = isPointInGui(logBtn, pos.X, pos.Y)
    local wasLong = logLongFired
    logPressInput = nil
    logPressPos = nil
    if wasLong or not wasInside or d >= 30 then return end
    if logWindow and logWindow.Parent then
        logWindow:Destroy()
        logWindow = nil
    else
        openLogWindow()
    end
end)

local function countKeywordOverrides(kw)
    local n = 0
    local ks = state.keywordStyle[kw]
    if ks then
        for _, v in pairs(ks) do
            if v ~= nil then n = n + 1 end
        end
    end
    return n
end

local EXPORT_FOLDER = "TargetTracer/Keywords"
local HAS_FS = (type(writefile) == "function"
    and type(readfile) == "function" and type(listfiles) == "function"
    and type(makefolder) == "function" and type(isfolder) == "function"
    and type(isfile) == "function" and type(delfile) == "function")

local function colorToTable(c)
    if typeof(c) == "Color3" then return { r = c.R, g = c.G, b = c.B } end
    return nil
end
local function tableToColor(t)
    if type(t) == "table" and type(t.r) == "number"
        and type(t.g) == "number" and type(t.b) == "number" then
        return Color3.new(math.clamp(t.r, 0, 1), math.clamp(t.g, 0, 1), math.clamp(t.b, 0, 1))
    end
    return nil
end
local function sanitizeFileName(name)
    if not name or name == "" then return nil end
    name = tostring(name):gsub("[/\\:*?\"<>|]", "_")
    name = name:gsub("%.json$", "")
    name = name:gsub("^%s+", ""):gsub("%s+$", "")
    if #name == 0 then return nil end
    if #name > 40 then name = string.sub(name, 1, 40) end
    return name
end
local function getFilePath(name) return EXPORT_FOLDER .. "/" .. name .. ".json" end
local function ensureExportFolder()
    if not HAS_FS then return false end
    return pcall(function()
        if not isfolder(EXPORT_FOLDER) then makefolder(EXPORT_FOLDER) end
    end)
end
local function countKeywords()
    local n = 0
    for _ in pairs(state.keywords) do n = n + 1 end
    return n
end
local function buildExportData()
    local data = { version = 1, exportedAt = os.date("%Y-%m-%d %H:%M:%S"), keywords = {} }
    for kw, _ in pairs(state.keywords) do
        local entry = {
            note = state.keywordNotes[kw] or "",
            drawEnabled = isKeywordDrawEnabled(kw),
        }
        local ks = state.keywordStyle[kw]
        if ks then
            local styleOut = {}
            for k, v in pairs(ks) do
                if typeof(v) == "Color3" then
                    local c = colorToTable(v)
                    if c then styleOut[k] = c end
                elseif type(v) == "number" or type(v) == "boolean" or type(v) == "string" then
                    styleOut[k] = v
                end
            end
            if next(styleOut) then entry.style = styleOut end
        end
        data.keywords[kw] = entry
    end
    return data
end
local function writeExportFile(name, data)
    if not HAS_FS then return false, "执行器不支持文件操作" end
    if not ensureExportFolder() then return false, "无法创建文件夹" end
    local path = getFilePath(name)
    local jsonStr
    local ok1 = pcall(function() jsonStr = HttpService:JSONEncode(data) end)
    if not ok1 or not jsonStr then return false, "编码失败" end
    local ok2, err2 = pcall(function() writefile(path, jsonStr) end)
    if not ok2 then return false, "写入失败: " .. tostring(err2) end
    return true
end
local function readExportFile(name)
    if not HAS_FS then return nil, "执行器不支持文件操作" end
    local path = getFilePath(name)
    local ok1, content = pcall(function() return readfile(path) end)
    if not ok1 or not content then return nil, "读取失败" end
    local ok2, decoded = pcall(function() return HttpService:JSONDecode(content) end)
    if not ok2 or type(decoded) ~= "table" then return nil, "解析失败" end
    return decoded
end
local function listExportFiles()
    local files = {}
    if not HAS_FS then return files end
    pcall(function()
        if not isfolder(EXPORT_FOLDER) then return end
        for _, p in ipairs(listfiles(EXPORT_FOLDER)) do
            local fname = p:match("([^/\\]+)%.json$")
            if fname then table.insert(files, { name = fname, path = p }) end
        end
    end)
    table.sort(files, function(a, b) return a.name < b.name end)
    return files
end
local function deleteExportFile(name)
    if not HAS_FS then return false end
    return pcall(function()
        local path = getFilePath(name)
        if isfile(path) then delfile(path) end
    end)
end
local function fileExists(name)
    if not HAS_FS then return false end
    local ok, result = pcall(function() return isfile(getFilePath(name)) end)
    return ok and result
end
local function applyImportData(data)
    pauseScanAndResume(1)
    for t, entry in pairs(drawGroups) do destroyEntry(entry) end
    state.keywords = {}
    state.keywordNotes = {}
    state.keywordDrawEnabled = {}
    state.keywordStyle = {}
    drawGroups = {}
    local added, skipped = 0, 0
    if type(data) ~= "table" or type(data.keywords) ~= "table" then return added, skipped end
    for kw, entry in pairs(data.keywords) do
        local ok = false
        if type(kw) == "string" and kw ~= "" and type(entry) == "table" then
            ok = pcall(function()
                state.keywords[kw] = true
                if type(entry.note) == "string" then state.keywordNotes[kw] = sanitizeNote(entry.note) end
                if type(entry.drawEnabled) == "boolean" then state.keywordDrawEnabled[kw] = entry.drawEnabled end
                if type(entry.style) == "table" then
                    local styleIn = {}
                    for k, v in pairs(entry.style) do
                        if type(v) == "table" then
                            local c = tableToColor(v)
                            if c then styleIn[k] = c end
                        elseif type(v) == "number" or type(v) == "boolean" or type(v) == "string" then
                            styleIn[k] = v
                        end
                    end
                    if next(styleIn) then state.keywordStyle[kw] = styleIn end
                end
            end)
        end
        if ok then added = added + 1
        else
            skipped = skipped + 1
            state.keywords[kw] = nil
            state.keywordNotes[kw] = nil
            state.keywordDrawEnabled[kw] = nil
            state.keywordStyle[kw] = nil
        end
    end
    return added, skipped
end

local function renameExportFile(oldName, onDone)
    showInputDialog("输入新文件名（不含扩展名）", oldName, function(raw)
        local newName = sanitizeFileName(raw)
        if not newName then
            addLog("文件名无效")
            return
        end

        if newName == oldName then
            showInfoDialog("无法重命名", "不能覆盖自己，文件名未改变。", function()
                renameExportFile(oldName, onDone)
            end)
            return
        end

        if fileExists(newName) then
            showExportConflictDialog(newName, function(action)
                if action == "cancel" then
                    return
                elseif action == "rename" then
                    renameExportFile(oldName, onDone)
                elseif action == "overwrite" then
                    local data = readExportFile(oldName)
                    if not data then
                        showInfoDialog("重命名失败", "原文件不存在，可能已被删除。", function()
                            if onDone then onDone() end
                        end)
                        return
                    end
                    local ok, werr = writeExportFile(newName, data)
                    if not ok then
                        addLog("重命名失败: " .. tostring(werr))
                        if onDone then onDone() end
                        return
                    end
                    local delOk = deleteExportFile(oldName)
                    if delOk then
                        addLog("已用 " .. oldName .. ".json 覆盖 " .. newName .. ".json")
                    else
                        addLog("已覆盖，但旧文件删除失败，请手动清理: " .. oldName .. ".json")
                    end
                    if onDone then onDone() end
                end
            end)
            return
        end

        local data = readExportFile(oldName)
        if not data then
            showInfoDialog("重命名失败", "原文件不存在，可能已被删除。", function()
                if onDone then onDone() end
            end)
            return
        end
        local ok, werr = writeExportFile(newName, data)
        if not ok then
            addLog("重命名失败: " .. tostring(werr))
            if onDone then onDone() end
            return
        end
        local delOk = deleteExportFile(oldName)
        if delOk then
            addLog("已重命名: " .. oldName .. ".json → " .. newName .. ".json")
        else
            addLog("已重命名，但旧文件删除失败，请手动清理: " .. oldName .. ".json")
        end
        if onDone then onDone() end
    end)
end

    local function applyExpandedState()
        for _, sec in ipairs(sections) do
            if sec.isStatic then
                sec.expanded = true
                sec.content.Visible = true
                sec.arrow.Text = "▌"
            else
                local want = state.settingsSectionExpanded[sec.key] == true
                sec.expanded = want
                sec.content.Visible = want
                sec.arrow.Text = want and "▼" or "▶"
            end
        end
        rebuildLayout()
    end

    local function toggleSection(key)
        local target
        for _, sec in ipairs(sections) do
            if sec.key == key then target = sec; break end
        end
        if not target or target.isStatic then return end
        if target.expanded then
            state.settingsSectionExpanded[key] = false
        else
            for _, s in ipairs(sections) do
                if not s.isStatic then
                    state.settingsSectionExpanded[s.key] = (s.key == key)
                end
            end
        end
        applyExpandedState()
    end

        local handle = Instance.new("Frame")
        handle.Size = UDim2.new(0, 16, 0, 16)
        handle.AnchorPoint = Vector2.new(0.5, 0.5)
        handle.Position = UDim2.new(0, 0, 0.5, 0)
        handle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        handle.BorderSizePixel = 0
        handle.ZIndex = 3
        handle.Parent = track
        local hs = Instance.new("UIStroke"); hs.Color = Color3.fromRGB(0, 0, 0); hs.Thickness = 1.5; hs.Parent = handle
        local hc = Instance.new("UICorner"); hc.CornerRadius = UDim.new(1, 0); hc.Parent = handle
        local function ratioToValue(r) return sliderMin + r * (sliderMax - sliderMin) end
        local function valueToRatio(v)
            if sliderMax == sliderMin then return 0 end
            return math.clamp((v - sliderMin) / (sliderMax - sliderMin), 0, 1)
        end
        local function refresh()
            local v = getter()
            if v == nil then btn.Text = "—" else btn.Text = tostring(v) end
            local r = valueToRatio(v or 0)
            handle.Position = UDim2.new(r, 0, 0.5, 0)
            fill.Size = UDim2.new(r, 0, 1, 0)

            local cur = getter()

                cur and tostring(cur) or "",
                function(val)
                    val = tostring(val or ""):gsub("%s+", "")

                        setter(n)
                        refreshAllDrawingStyles()
                        addLog(label .. " 已设为 " .. n)
                        refresh()
                    else
                        addLog(label .. " 输入无效（范围 " .. inputMin .. " ~ " .. inputMax .. "）")
                    end
                end)
        end)
        local dragging = false
        local function updateFromX(screenX)
            local absPos = track.AbsolutePosition.X
            local absSize = track.AbsoluteSize.X
            if absSize <= 0 then return end
            local ratio = math.clamp((screenX - absPos) / absSize, 0, 1)
            local v = ratioToValue(ratio)
            if (sliderMax - sliderMin) > 10 then v = math.floor(v + 0.5)
            else v = math.floor(v * 100 + 0.5) / 100 end
            setter(v)
            refreshAllDrawingStyles()
            refresh()
        end
        track.InputBegan:Connect(function(input)
            if input.UserInputType ~= Enum.UserInputType.Touch
                and input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
            if input.UserInputState ~= Enum.UserInputState.Begin then return end
            dragging = true
            updateFromX(input.Position.X)
        end)
        local c1 = UserInputService.InputChanged:Connect(function(input)
            if not dragging then return end
            if input.UserInputType == Enum.UserInputType.MouseMovement
                or input.UserInputType == Enum.UserInputType.Touch then
                updateFromX(input.Position.X)
            end
        end)
        local c2 = UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch
                or input.UserInputType == Enum.UserInputType.MouseButton1 then
                dragging = false
            end
        end)
        table.insert(settingsSliderConns, c1)
        table.insert(settingsSliderConns, c2)
    end

local function buildKws()
    local kws, hasAny = {}, false
    for kw, _ in pairs(state.keywords) do kws[kw] = true; hasAny = true end
    return hasAny and kws or nil
end

local function matchKeywordByName(name, kws)
    if not name or name == "" then return nil end
    if keywordMatchCache[name] ~= nil then
        local cachedResult = keywordMatchCache[name]
        if cachedResult == false then return nil end
        return cachedResult
    end

    local extracted = extractKeyword(name)
    if extracted == "" then
        keywordMatchCache[name] = false
        return nil
    end
    if kws[extracted] then
        keywordMatchCache[name] = extracted
        return extracted
    end
    if isGenericKeyword(extracted) or #extracted < 4 then
        keywordMatchCache[name] = false
        return nil
    end
    for kw, _ in pairs(kws) do
        if not isGenericKeyword(kw) and #kw >= 4 then
            if string.find(extracted, kw, 1, true) then
                keywordMatchCache[name] = kw
                return kw
            end
        end
    end
    keywordMatchCache[name] = false
    return nil
end

local function matchRoot(root, inst, kws)
    local cur = inst
    local depth = 0
    while cur and cur ~= workspace and depth < 6 do
        local m = matchKeywordByName(cur.Name, kws)
        if m then return m end
        cur = cur.Parent
        depth = depth + 1
    end
    return nil
end

local function quickNearbyScan()
    local kws = buildKws()
    if not kws then return 0 end
    keywordMatchCache = {}

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local origin = hrp and hrp.Position or camera.CFrame.Position
    
    local playerChars = {}
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl.Character then playerChars[pl.Character] = true end
    end

    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if char then params.FilterDescendantsInstances = { char } end
    params.MaxParts = NEARBY_MAX_PARTS
    local ok, parts = pcall(function() return workspace:GetPartBoundsInRadius(origin, NEARBY_RADIUS, params) end)
    if not ok or not parts then return 0 end
    
    local added = 0
    local seenRoot = {}
    for _, inst in ipairs(parts) do
        if inst:IsA("BasePart") and not inst:IsA("Terrain") then
            local root = findModelRoot(inst)
            if root and not seenRoot[root] and not drawGroups[root] and not isProtectedTree(root) then
                seenRoot[root] = true
                if not playerChars[root] then
                    local matchedKw = matchRoot(root, inst, kws)
                    if matchedKw then
                        registerDrawEntry(root, {
                            g = createGroup(matchedKw),
                            label = createNameLabel(matchedKw),
                            name = root.Name, keyword = matchedKw,
                        })
                        added = added + 1
                    end
                end
            end
        end
    end
    return added
end

local function tryAddRoot(root)
    if isScanPaused then return false end
    if not root then return false end
    local ok, parent = pcall(function() return root.Parent end)
    if not ok or not parent then return false end
    if drawGroups[root] then return false end
    if isProtectedTree(root) then return false end
    local kws = buildKws()
    if not kws then return false end
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local origin = hrp and hrp.Position or camera.CFrame.Position
    local center
    if root:IsA("Model") then
        local ok2, box = pcall(function() return root:GetBoundingBox() end)
        if ok2 and box then center = box.Position end
    elseif root:IsA("BasePart") then center = root.Position end
    if center and (center - origin).Magnitude > state.globalStyle.maxDrawDistance then return false end
    local inst = root:IsA("BasePart") and root or root:FindFirstChildWhichIsA("BasePart")
    if not inst then
        local ok3, descendants = pcall(function() return root:GetDescendants() end)
        if ok3 and descendants then
            for _, d in ipairs(descendants) do
                if d:IsA("BasePart") then inst = d; break end
            end
        end
    end
    if not inst then return false end
    for _, pl in ipairs(Players:GetPlayers()) do
        local ok4, isDesc = pcall(function()
            return pl.Character and (pl.Character == root
                or root:IsDescendantOf(pl.Character)
                or pl.Character:IsDescendantOf(root))
        end)
        if ok4 and isDesc then return false end
    end
    local matchedKw = matchRoot(root, inst, kws)
    if not matchedKw then return false end
    registerDrawEntry(root, {
        g = createGroup(matchedKw),
        label = createNameLabel(matchedKw),
        name = root.Name, keyword = matchedKw,
    })
    return true
end

startScanByKeywords = function()
    if isScanPaused then return end
    local kws = buildKws()
    if not kws then return end
    if scanQueue then scanQueue = nil end
    keywordMatchCache = {}

    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local origin = hrp and hrp.Position or camera.CFrame.Position
    local params = OverlapParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    if char then params.FilterDescendantsInstances = { char } end
    params.MaxParts = 20000
    
    local ok, parts = pcall(function() return workspace:GetPartBoundsInRadius(origin, state.scanRadius, params) end)
    if not ok or not parts then return end

    local playerChars = {}
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl.Character then playerChars[pl.Character] = true end
    end

    scanQueue = { 
        parts = parts, 
        index = 1, 
        added = 0, 
        seenRoot = {}, 
        kws = kws,
        playerChars = playerChars
    }
end

local function processScanQueue()
    if not scanQueue then return end
    local parts = scanQueue.parts
    local kws = scanQueue.kws
    local playerChars = scanQueue.playerChars or {}
    if not kws or not parts then scanQueue = nil; return end

    local startTime = os.clock()
    local stopIdx = scanQueue.index
    local added = scanQueue.added or 0

    for i = scanQueue.index, #parts do
        if os.clock() - startTime > SCAN_TIME_BUDGET then
            break
        end
        stopIdx = i
        local inst = parts[i]
        if inst and inst.Parent and inst:IsA("BasePart") and not inst:IsA("Terrain") then
            local root = findModelRoot(inst)
            if root and not scanQueue.seenRoot[root] and not drawGroups[root] and not isProtectedTree(root) then
                if not playerChars[root] then
                    local matchedKw = matchRoot(root, inst, kws)
                    if matchedKw then
                        scanQueue.seenRoot[root] = true
                        registerDrawEntry(root, {
                            g = createGroup(matchedKw),
                            label = createNameLabel(matchedKw),
                            name = root.Name, keyword = matchedKw,
                        })
                        added = added + 1
                    end
                end
            end
        end
    end

    scanQueue.index = stopIdx + 1
    scanQueue.added = added

    if scanQueue.index > #parts then
        if added >= 2 then 
            addLog("扫描补充 " .. added .. " 个目标") 
        end
        keywordMatchCache = {}
        scanQueue = nil
    end
end

function addTarget(target, autoScan)
    if not target then return false end
    local root = findModelRoot(target)
    if not root then return false end
    if isProtectedTree(root) then addLog("该目标已被保护"); return false end
    if drawGroups[root] then addLog("已存在: " .. root.Name); return false end
    local keyword = findBestKeyword(root)
    if not keyword then
        addLog("该目标名为通用词无法自动提取关键词: " .. root.Name)
        return false
    end
    state.keywords[keyword] = true
    registerDrawEntry(root, {
        g = createGroup(keyword),
        label = createNameLabel(keyword),
        name = root.Name, keyword = keyword,
    })
    addLog("添加: " .. root.Name .. " [关键词: " .. keyword .. "]", keyword, root)
    if autoScan == true then
        local nearbyAdded = quickNearbyScan()
        if nearbyAdded > 0 then addLog("近距离扫描补充 " .. nearbyAdded .. " 个目标") end
        startScanByKeywords()
    end
    return true
end

local function addSingleTarget(target)
    if not target then return false end

    return true
end

function protectTarget(target)
    if not target then return false end
    if state.protectedTargets[target] then
        addLog("已在保护列表: " .. target.Name)
        return false
    end
    state.protectedTargets[target] = { note = "" }
    local check = target
    while check and check ~= workspace do
        if drawGroups[check] then
            destroyEntry(drawGroups[check])
            drawGroups[check] = nil
            break
        end
        check = check.Parent
    end
    updateProtectBtn()
    updateBlockedCamps()
    addLog("已加入单目标保护: " .. target.Name)
    return true
end

function unprotectTarget(target)
    if not target then return false end
    if not state.protectedTargets[target] then
        addLog("不在保护列表: " .. target.Name)
        return false
    end
    state.protectedTargets[target] = nil
    updateProtectBtn()
    updateBlockedCamps()
    addLog("已取消保护: " .. target.Name)
    return true
end
