// ==UserScript==
// @name         V26.0 双节点强化探测版
// @namespace    http://tampermonkey.net/
// @version      26.0
// @description  主页面逻辑完美保留。中转页强化探测，同时兼容第一步和最后一步的Continue按钮，防息屏，带可视日志。
// @author       AI
// @match        *://*/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

    if (window.top !== window.self) return;
    if (window.location.href.includes('challenges.cloudflare.com')) return;

    const isMainPage = document.title.includes('Get Delta Key') ||
                       (document.body.innerText.includes('Complete Tasks to Continue') && document.body.innerText.match(/~\d+\s*sec/i));

    // ================= 1. 屏幕常亮 (防息屏) =================
    let wakeLock = null;
    async function requestWakeLock() {
        try {
            if ('wakeLock' in navigator && document.visibilityState === 'visible') {
                wakeLock = await navigator.wakeLock.request('screen');
                console.log('[WakeLock] 屏幕常亮已激活');
            }
        } catch (err) {}
    }
    requestWakeLock();
    document.addEventListener('visibilitychange', async () => {
        if (document.visibilityState === 'visible' && (wakeLock === null || wakeLock.released)) await requestWakeLock();
    });
    setInterval(async () => {
        if (wakeLock === null || wakeLock.released) await requestWakeLock();
    }, 30000);

    // ================= 2. 悬浮窗日志系统 =================
    let logMessages = [];
    let logContent = null;
    let panel = null;

    function createLogWindow() {
        if (document.getElementById('v26-log-panel')) return;
        panel = document.createElement('div');
        panel.id = 'v26-log-panel';
        Object.assign(panel.style, {
            position: 'fixed', bottom: '20px', right: '5px', width: '200px', height: '160px',
            backgroundColor: 'rgba(0, 0, 0, 0.9)', color: '#00ff00', border: '1px solid #00ffff',
            borderRadius: '6px', zIndex: '2147483647', display: 'flex', flexDirection: 'column',
            overflow: 'hidden', fontFamily: 'monospace', fontSize: '10px', boxShadow: '0 2px 10px rgba(0,0,0,0.8)', touchAction: 'none'
        });
        panel.innerHTML = `
            <div id="v26-log-header" style="background: #003333; padding: 3px 5px; display: flex; justify-content: space-between; align-items: center; cursor: move; user-select: none;">
                <span style="color: #00ffff; font-weight: bold; font-size: 10px;">📜 日志 V26.0</span>
                <button id="v26-copy-btn" style="background: #00ffff; border: none; border-radius: 3px; padding: 1px 4px; cursor: pointer; font-weight: bold; font-size: 9px;">📋 复制</button>
            </div>
            <div id="v26-log-content" style="flex: 1; overflow-y: auto; padding: 4px; word-break: break-all; line-height: 1.3;">初始化...</div>
        `;
        document.body.appendChild(panel);

        logContent = document.getElementById('v26-log-content');
        const copyBtn = document.getElementById('v26-copy-btn');

        let isDragging = false, startX, startY, initialLeft, initialTop;
        const header = document.getElementById('v26-log-header');
        const startDrag = (e) => { if (e.target.tagName.toLowerCase() === 'button') return; isDragging = true; const cx = e.touches ? e.touches[0].clientX : e.clientX; const cy = e.touches ? e.touches[0].clientY : e.clientY; startX = cx; startY = cy; const rect = panel.getBoundingClientRect(); initialLeft = rect.left; initialTop = rect.top; };
        const doDrag = (e) => { if (!isDragging) return; e.preventDefault(); const cx = e.touches ? e.touches[0].clientX : e.clientX; const cy = e.touches ? e.touches[0].clientY : e.clientY; panel.style.bottom = 'auto'; panel.style.right = 'auto'; panel.style.left = `${initialLeft + (cx - startX)}px`; panel.style.top = `${initialTop + (cy - startY)}px`; };
        const stopDrag = () => { isDragging = false; };
        header.addEventListener('touchstart', startDrag, { passive: false }); header.addEventListener('touchmove', doDrag, { passive: false }); header.addEventListener('touchend', stopDrag);
        header.addEventListener('mousedown', startDrag); document.addEventListener('mousemove', doDrag); document.addEventListener('mouseup', stopDrag);

        copyBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            const text = logMessages.join('\n');
            const textArea = document.createElement('textarea');
            textArea.value = text; textArea.style.position = 'fixed'; textArea.style.left = '-9999px';
            document.body.appendChild(textArea); textArea.focus(); textArea.select();
            try { document.execCommand('copy'); copyBtn.innerText = '✅ 已复制'; setTimeout(() => copyBtn.innerText = '📋 复制', 2000); } catch (err) { alert('复制失败，请手动长按复制。'); }
            document.body.removeChild(textArea);
        });
    }

    function log(msg, isHighlight = false) {
        const time = new Date().toLocaleTimeString('zh-CN', { hour12: false });
        const entry = `[${time}] ${msg}`;
        logMessages.push(entry);
        if (logMessages.length > 60) logMessages.shift();
        if (logContent) {
            logContent.innerText = logMessages.join('\n');
            logContent.scrollTop = logContent.scrollHeight;
            if (isHighlight) logContent.style.color = '#00ffff';
        }
        console.log(entry);
    }

    const forceClick = (element) => {
        if (!element) return false;
        try {
            if (typeof element.click === 'function') element.click();
            element.dispatchEvent(new MouseEvent('mousedown', { bubbles: true, cancelable: true, view: window }));
            element.dispatchEvent(new MouseEvent('mouseup', { bubbles: true, cancelable: true, view: window }));
            element.dispatchEvent(new PointerEvent('click', { bubbles: true, cancelable: true }));
            element.dispatchEvent(new TouchEvent('touchstart', { bubbles: true, cancelable: true }));
            element.dispatchEvent(new TouchEvent('touchend', { bubbles: true, cancelable: true }));
            return true;
        } catch (e) { return false; }
    };

    // ================= 3. 强化版中转页逻辑 (双节点兼容) =================
    async function runTransitPage() {
        log('检测到非主页面，启动强化探测...', true);
        let retryCount = 0;

        while (retryCount < 300) {
            await sleep(1000);
            retryCount++;

            // 【核心修改】获取所有文本为 Continue 的按钮（无论 ID）
            let allContinueBtns = Array.from(document.querySelectorAll('button, div, a')).filter(el => {
                const text = el.textContent.trim();
                // 必须完全等于 Continue，且不能包含等待或加载字样
                return (text === 'Continue' || text === '继续') && !text.includes('wait') && !text.includes('loading');
            });

            let clickedAny = false;

            for (let btn of allContinueBtns) {
                const style = window.getComputedStyle(btn);
                const isDisabled = btn.disabled || btn.getAttribute('disabled') !== null;
                const isPointerNone = style.pointerEvents === 'none';
                const isVisible = style.display !== 'none' && style.visibility !== 'hidden' && style.opacity !== '0';

                // 排除纯暗色的背景（未激活状态常为暗色）
                const isDarkBg = style.backgroundColor.includes('rgb(29, 43, 66)') || style.backgroundColor.includes('rgba(0, 0, 0, 0)');

                if (!isDisabled && !isPointerNone && isVisible && !isDarkBg) {
                    log(`✅ 探测到有效按钮 (${btn.tagName})，执行点击！`, true);
                    forceClick(btn);
                    if (btn.parentElement) forceClick(btn.parentElement);
                    clickedAny = true;
                }
            }

            if (clickedAny) {
                // 点击后等待 2 秒，检查页面是否跳转
                const beforeUrl = window.location.href;
                await sleep(2000);
                if (window.location.href !== beforeUrl) {
                    log('🎉 页面已成功跳转！', true);
                    return;
                } else {
                    log('⚠️ 点击后暂未跳转，继续尝试...');
                }
            } else {
                if (retryCount % 5 === 0) log('⏳ 等待 Continue 按钮出现或变亮...');
            }
        }
        log('⚠️ 中转页监控超时');
    }

    // ================= 4. 主页面逻辑 =================
    async function runMainPage() {
        log('检测到主页面，开始执行任务');

        let retries = 0, taskCards = [];
        const getTaskCards = () => {
            const allElements = Array.from(document.querySelectorAll('*'));
            const timeElements = allElements.filter(el => /~\d+\s*sec/i.test(el.textContent) && el.children.length === 0);
            const uniqueCards = [];
            timeElements.forEach(timeEl => {
                let container = timeEl;
                for (let i = 0; i < 2; i++) { if (container.parentElement) container = container.parentElement; }
                if (!uniqueCards.includes(container)) uniqueCards.push(container);
            });
            return uniqueCards;
        };

        while (retries < 20) {
            taskCards = getTaskCards();
            if (taskCards.length > 0) break;
            log(`等待任务加载... (${retries+1}s)`); await sleep(1000); retries++;
        }
        if (taskCards.length === 0) { log('❌ 加载超时，请刷新网页'); return; }

        let tasksToProcess = taskCards.slice(0, 4);
        for (let i = 0; i < tasksToProcess.length; i++) {
            let clickable = tasksToProcess[i].querySelector('a') || tasksToProcess[i].querySelector('div[class*="cursor"]') || tasksToProcess[i];
            forceClick(clickable);
            log(`已点击第 ${i+1} 个任务`);
            if (i === 0) await sleep(500); else if (i < 3) await sleep(100);
        }

        log('任务点击完毕，等待人机验证出现...');
        let waitVerifyCount = 0;
        while (waitVerifyCount < 30) {
            let verifyTitle = Array.from(document.querySelectorAll('*')).find(el => /CONFIRM YOU ARE\s*HUMAN/i.test(el.textContent) && el.children.length === 0);
            if (verifyTitle) {
                forceClick(verifyTitle);
                await sleep(500);
                if (verifyTitle.parentElement && verifyTitle.parentElement.parentElement) forceClick(verifyTitle.parentElement.parentElement);
                log('👉 请手动勾选“请验证您是真人”！', true);
                break;
            }
            waitVerifyCount++;
            await sleep(1000);
        }

        // 主任务循环
        let maxWait = 300;
        let rewardClicked = false;
        let finalClicked = false;

        while (maxWait > 0 && !finalClicked) {
            maxWait--;
            await sleep(1000);

            const taskDivs = Array.from(document.querySelectorAll('div.task'));
            let doneCount = 0;
            taskDivs.forEach(div => { if (div.className.includes('done')) doneCount++; });
            const allDone = doneCount > 0 && doneCount >= taskDivs.length;

            if (taskDivs.length > 0) {
                if (allDone) {
                    log(`🎉 所有任务已完成 (${doneCount}/${taskDivs.length})，准备领奖！`, true);

                    if (!rewardClicked) {
                        let claimSpan = Array.from(document.querySelectorAll('span')).find(el => el.textContent.toLowerCase().includes('claim reward'));
                        if (claimSpan) {
                            let targetBtn = claimSpan;
                            for (let i = 0; i < 3; i++) { if (targetBtn.parentElement) targetBtn = targetBtn.parentElement; }

                            const style = window.getComputedStyle(targetBtn);
                            if (style.pointerEvents !== 'none' && parseFloat(style.opacity) >= 0.8 && !style.backgroundColor.includes('128, 128, 128')) {
                                log('✅ Claim Reward 已激活，执行点击！', true);
                                forceClick(claimSpan);
                                forceClick(targetBtn);
                                rewardClicked = true;

                                await sleep(2000);
                                let stillExist = Array.from(document.querySelectorAll('span')).some(el => el.textContent.toLowerCase().includes('claim reward'));
                                if (stillExist) { rewardClicked = false; } else { log('🎉 奖励领取成功！', true); }
                            }
                        }
                    }

                    if (rewardClicked) {
                        let finalBtns = Array.from(document.querySelectorAll('button')).filter(el => el.textContent.trim() === 'Continue');
                        for (let btn of finalBtns) {
                            const style = window.getComputedStyle(btn);
                            if (style.pointerEvents !== 'none' && parseFloat(style.opacity) >= 0.8 && !style.backgroundColor.includes('128, 128, 128')) {
                                log('✅ 最终 Continue 已激活，执行点击！', true);
                                forceClick(btn);
                                finalClicked = true;
                                break;
                            }
                        }
                        if (finalClicked) { await sleep(2000); log('🎉 所有任务流程全部完成！', true); }
                    }
                } else {
                    if (maxWait % 5 === 0) log(`⏳ 等待全部任务打勾... (当前进度: ${doneCount}/${taskDivs.length})`);
                }
            }
        }
        if (!finalClicked) log('⚠️ 等待超时，请手动检查页面状态。');
    }

    // ================= 5. 启动入口 =================
    function init() {
        createLogWindow();
        if (isMainPage) {
            log('日志系统已启动 (V26.0 主页面)');
            runMainPage();
        } else {
            log('日志系统已启动 (V26.0 中转页)');
            runTransitPage();
        }
    }

    if (document.body) init(); else window.addEventListener('load', init);
})();
