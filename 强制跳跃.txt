local dragging = false
local dragInput = nil
local dragStart = nil
local startPos = nil
local handleTouches = {}

local function isDragStartInput(input)
    return input.UserInputType == Enum.UserInputType.MouseButton1 
        or input.UserInputType == Enum.UserInputType.Touch
end

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    
    local matched = (input == dragInput)
        or (dragInput and dragInput.UserInputType == Enum.UserInputType.MouseButton1
            and input.UserInputType == Enum.UserInputType.MouseMovement)

    if not matched then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement 
        and input.UserInputType ~= Enum.UserInputType.Touch then return end

    local delta = input.Position - dragStart
    if delta.Magnitude == 0 then return end
    
    mainFrame.Position = UDim2.new(
        startPos.X.Scale, startPos.X.Offset + delta.X,
        startPos.Y.Scale, startPos.Y.Offset + delta.Y
    )
end)

UserInputService.InputEnded:Connect(function(input)
    if handleTouches[input] then 
        handleTouches[input] = nil 
    end
    
    if not dragging or input ~= dragInput then return end
    if not isDragStartInput(input) then return end

    local nextInput = nil
    for t in pairs(handleTouches) do nextInput = t; break end

    if nextInput then 
        dragging = true
        dragInput = nextInput
        dragStart = nextInput.Position
        startPos = mainFrame.Position
    else
        dragging = false
        dragInput = nil
        dragStart = nil
        startPos = nil
    end
end)

titleBar.InputBegan:Connect(function(input)
    if not isDragStartInput(input) then return end
    handleTouches[input] = true
    
    if dragging then return end
    
    dragging = true
    dragInput = input
    dragStart = input.Position
    startPos = mainFrame.Position
end)

local isAutoJumping = false
local heartbeatConn

local function doJump()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildWhichIsA("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    
    if hum and hrp then
        pcall(function()
            hum.Jump = true
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        
        pcall(function()
            local currentVel = hrp.AssemblyLinearVelocity
            if currentVel.Y < 10 then
                hrp.AssemblyLinearVelocity = Vector3.new(currentVel.X, 50, currentVel.Z)
            end
        end)
    end
end

jumpBtn.MouseButton1Click:Connect(function()
    doJump()
    jumpBtn.BackgroundColor3 = Color3.fromRGB(100, 200, 120)
    task.delay(0.1, function() jumpBtn.BackgroundColor3 = Color3.fromRGB(60, 160, 90) end)
end)

jumpBtn.MouseButton1Down:Connect(function()
    doJump()
end)

autoJumpBtn.MouseButton1Click:Connect(function()
    isAutoJumping = not isAutoJumping
    if isAutoJumping then
        autoJumpBtn.Text = "连跳: 开"
        autoJumpBtn.BackgroundColor3 = Color3.fromRGB(60, 160, 90)
    else
        autoJumpBtn.Text = "连跳: 关"
        autoJumpBtn.BackgroundColor3 = Color3.fromRGB(80, 80, 100)
    end
end)

heartbeatConn = RunService.Heartbeat:Connect(function()
    if isAutoJumping then
        doJump()
    end
end)

exitBtn.MouseButton1Click:Connect(function()
    isAutoJumping = false
    if heartbeatConn then
        heartbeatConn:Disconnect()
        heartbeatConn = nil
    end
    screenGui:Destroy()
end)

player.CharacterAdded:Connect(function()
    isAutoJumping = false
    if autoJumpBtn and autoJumpBtn.Parent then
        autoJumpBtn.Text = "连跳: 关"
        autoJumpBtn.BackgroundColor3 = Color3.fromRGB(80, 80, 100)
    end
end)
